DECLARE
    TYPE ColumnArray IS TABLE OF VARCHAR2(100);
    column_names ColumnArray;
    input_query    VARCHAR2(4000) := 'SELECT FORM_OF_LC FORM_OF_LC, APPL_ID APPL_ID7, APPL_NM APPL_NM3, APPL_ADD APPL_ADD2, BENE_NM_ADD BENE_NM_ADD2, NEW_LC_AMT LC_AMT FROM INLC_MASTER';
    table_name     VARCHAR2(100) := 'INLC_MASTER';
    valid_query    VARCHAR2(4000) := 'SELECT ';
    col_list       VARCHAR2(4000) := '';
    token          VARCHAR2(1000);
    pos            NUMBER;

BEGIN
    -- Retrieve column names into the array
    SELECT COLUMN_NAME
    BULK COLLECT INTO column_names
    FROM ALL_TAB_COLUMNS
    WHERE TABLE_NAME = UPPER(table_name);

    -- Extract the column list from the input query
    input_query := REGEXP_REPLACE(input_query, '.*SELECT\s+', '', 1, 1, 'i');
    input_query := REGEXP_REPLACE(input_query, 'FROM.*', '');

    -- Loop through each column in the input_query
    LOOP
        -- Find the next comma
        pos := INSTR(input_query, ',');

        -- If no more commas, get the remaining part
        IF pos = 0 THEN
            token := TRIM(input_query);
        ELSE
            token := TRIM(SUBSTR(input_query, 1, pos - 1));
            input_query := SUBSTR(input_query, pos + 1);
        END IF;

        DECLARE
            col_name      VARCHAR2(100);
            alias_name    VARCHAR2(100);
            column_exists  BOOLEAN := FALSE;

        BEGIN
            -- Extract column name and alias, if any
            col_name := TRIM(REGEXP_SUBSTR(token, '^[^ ]+'));
            alias_name := TRIM(REGEXP_SUBSTR(token, '\s+(\S+)$'));

            -- If no alias found, set alias_name to NULL
            IF alias_name = col_name THEN
                alias_name := NULL; -- No alias
            ELSE
                alias_name := TRIM(REPLACE(alias_name, ' ', ''));
            END IF;

            -- Check if column exists in the column_names array
            FOR i IN 1 .. column_names.COUNT LOOP
                IF UPPER(column_names(i)) = UPPER(col_name) THEN
                    column_exists := TRUE;
                    EXIT;
                END IF;
            END LOOP;

            -- If column exists, add it to the valid_query
            IF column_exists THEN
                IF col_list IS NOT NULL THEN
                    col_list := col_list || ', ';
                END IF;
                col_list := col_list || col_name;

                -- If there's an alias, include it
                IF alias_name IS NOT NULL THEN
                    col_list := col_list || ' ' || alias_name;
                END IF;
            END IF;

        END;

        -- Exit the loop if we processed the last column
        EXIT WHEN pos = 0;
    END LOOP;

    -- Construct the final valid query
    valid_query := valid_query || col_list || ' FROM ' || table_name;

    -- Output the valid query
    DBMS_OUTPUT.PUT_LINE(valid_query);
END;
