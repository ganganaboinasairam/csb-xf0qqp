    <title>Snake Game</title>
    <style>
        body {
            text-align: center;
        }

        .geeks {
            font-size: 40px;
            font-weight: bold;
            color: green;
        }

        #buttons {
            margin-top: 20px;
        }

        #buttons button {
            padding: 10px 20px;
            font-size: 20px;
            margin: 10px;
        }

        #board {
            display: none; /* Hide the board initially */
            border: 2px solid black; /* Add a border to the board */
        }
    </style>
</head>

<body>
    <h1>Snake Game</h1>
    <div id='score'>Score: 0</div>
    <div id='Hscore'>High Score: 0</div>
    <canvas id="board"></canvas>

    <div id="buttons">
        <button id="easyBtn">Easy</button>
        <button id="hardBtn">Hard</button>
    </div>

    <script>
        let blockSize = 10;
        let total_row = 50;
        let total_col = 50;
        let board;
        let context;

        let snakeX = blockSize * 5;
        let snakeY = blockSize * 5;

        let speedX = 0;
        let speedY = 0;
        let speed = 10;

        let snakeBody = [];
        let foodX;
        let foodY;

        let gameOver = false;
        let currentScore = 0;
        let isEasyMode = false;

        window.onload = function () {
            board = document.getElementById("board");
            board.height = total_row * blockSize;
            board.width = total_col * blockSize;
            context = board.getContext("2d");

            let highScore = sessionStorage.getItem("Hscore") || 0;
            document.getElementById("Hscore").innerText = "High Score: " + highScore;

            // Event listeners for Easy and Hard buttons
            document.getElementById("easyBtn").addEventListener("click", function () {
                setEasyMode();
            });

            document.getElementById("hardBtn").addEventListener("click", function () {
                setHardMode();
            });
        }

        function update() {
            if (gameOver) {
                return;
            }

            context.fillStyle = "green";
            context.fillRect(0, 0, board.width, board.height);

            context.fillStyle = "yellow";
            context.fillRect(foodX, foodY, blockSize, blockSize);

            if (snakeX === foodX && snakeY === foodY) {
                snakeBody.push([foodX, foodY]);
                currentScore++;
                document.getElementById("score").innerText = "Score: " + currentScore;

                let highScore = sessionStorage.getItem("Hscore") || 0;
                if (currentScore > highScore) {
                    sessionStorage.setItem("Hscore", currentScore);
                    document.getElementById("Hscore").innerText = "High Score: " + currentScore;
                }

                placeFood();
            }

            for (let i = snakeBody.length - 1; i > 0; i--) {
                snakeBody[i] = snakeBody[i - 1];
            }
            if (snakeBody.length) {
                snakeBody[0] = [snakeX, snakeY];
            }

            snakeX += speedX * blockSize;
            snakeY += speedY * blockSize;

            if (!isEasyMode) {
                if (snakeX < 0 || snakeX >= total_col * blockSize || snakeY < 0 || snakeY >= total_row * blockSize) {
                    endGame();
                }
            } else {
                // Snake can pass through walls in easy mode
                if (snakeX < 0) snakeX = total_col * blockSize - blockSize;
                if (snakeX >= total_col * blockSize) snakeX = 0;
                if (snakeY < 0) snakeY = total_row * blockSize - blockSize;
                if (snakeY >= total_row * blockSize) snakeY = 0;
            }

            context.fillStyle = "black";
            context.fillRect(snakeX, snakeY, blockSize, blockSize);
            for (let i = 0; i < snakeBody.length; i++) {
                context.fillRect(snakeBody[i][0], snakeBody[i][1], blockSize, blockSize);
            }

            for (let i = 0; i < snakeBody.length; i++) {
                if (snakeX === snakeBody[i][0] && snakeY === snakeBody[i][1]) {
                    endGame();
                }
            }
        }

        function endGame() {
            gameOver = true;
            alert("Game Over");
            currentScore = 0;
            document.getElementById("score").innerText = "Score: 0";
            gameOver = false;
            snakeX = blockSize * 5;
            snakeY = blockSize * 5;
            speedX = 0;
            speedY = 0;
            snakeBody = [];
            placeFood();
        }

        function changeDirection(e) {
            if (e.code === "ArrowUp" && speedY !== 1) {
                speedX = 0;
                speedY = -1;
            }
            else if (e.code === "ArrowDown" && speedY !== -1) {
                speedX = 0;
                speedY = 1;
            }
            else if (e.code === "ArrowLeft" && speedX !== 1) {
                speedX = -1;
                speedY = 0;
            }
            else if (e.code === "ArrowRight" && speedX !== -1) {
                speedX = 1;
                speedY = 0;
            }
        }

        function placeFood() {
            foodX = Math.floor(Math.random() * total_col) * blockSize;
            foodY = Math.floor(Math.random() * total_row) * blockSize;

            for (let i = 0; i < snakeBody.length; i++) {
                if (foodX === snakeBody[i][0] && foodY === snakeBody[i][1]) {
                    placeFood();
                    return;
                }
            }
        }

        // Function to set easy mode
        function setEasyMode() {
            isEasyMode = true;
            speed = 8;  // Slower speed in easy mode
            document.getElementById("board").style.display = "block"; // Show the board
            setInterval(update, 1000 / speed);
        }

        // Function to set hard mode
        function setHardMode() {
            isEasyMode = false;
            speed = 15;  // Faster speed in hard mode
            document.getElementById("board").style.display = "block"; // Show the board
            setInterval(update, 1000 / speed);
        }
    </script>
</body>

</html>
