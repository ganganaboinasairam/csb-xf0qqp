// script.js

const textDisplay = document.getElementById('text-display');
const textInput = document.getElementById('text-input');
const startBtn = document.getElementById('start-btn');
const timeDisplay = document.getElementById('time-display');
const wpmDisplay = document.getElementById('wpm-display');
const accuracyDisplay = document.getElementById('accuracy-display');

const sampleText = "This is the sample text that you need to type quickly and accurately.";
let startTime, endTime;
let timer = null;

function startTest() {
    textInput.value = "";
    textInput.disabled = false;
    textInput.focus();
    
    startTime = new Date();
    timer = setInterval(updateTime, 1000);

    startBtn.disabled = true;
}

function endTest() {
    clearInterval(timer);
    endTime = new Date();
    
    const timeTaken = (endTime - startTime) / 1000; // in seconds
    const wordsTyped = textInput.value.split(" ").length;
    const wpm = Math.round((wordsTyped / timeTaken) * 60);

    const accuracy = calculateAccuracy();
    
    timeDisplay.textContent = timeTaken.toFixed(1);
    wpmDisplay.textContent = wpm;
    accuracyDisplay.textContent = accuracy.toFixed(1);
    
    textInput.disabled = true;
    startBtn.disabled = false;
}

function updateTime() {
    const currentTime = new Date();
    const timeElapsed = (currentTime - startTime) / 1000; // in seconds
    timeDisplay.textContent = timeElapsed.toFixed(1);
}

function calculateAccuracy() {
    const inputText = textInput.value;
    const inputChars = inputText.split("");
    const sampleChars = sampleText.split("");
    
    let correctChars = 0;
    for (let i = 0; i < inputChars.length; i++) {
        if (inputChars[i] === sampleChars[i]) correctChars++;
    }
    return (correctChars / sampleChars.length) * 100;
}

// Event listeners
startBtn.addEventListener('click', startTest);
textInput.addEventListener('input', () => {
    if (textInput.value === sampleText) endTest();
});
