  <script>
    function convertSQL() {
      const sqlInput = document.getElementById('sqlInput').value;

      // Clean up the input SQL: remove extra whitespace and line breaks
      const sanitizedInput = sqlInput.replace(/\s+/g, ' ').trim();

      // Extract table name, columns, and values using regex
      const tableNameMatch = sanitizedInput.match(/INSERT INTO\s+([^\s]+)\s+([^)]+)\s+VALUES\s+(.+)/i);
      
      if (!tableNameMatch) {
        document.getElementById('output').textContent = 'Invalid SQL format';
        return;
      }

      const tableName = tableNameMatch[1].trim();
      const columns = tableNameMatch[2].split(',').map(col => col.trim());
      const valuesString = tableNameMatch[3].trim();

      // Split values, accounting for commas inside quotes
      const values = [];
      let currentVal = '';
      let insideQuotes = false;

      for (let i = 0; i < valuesString.length; i++) {
        const char = valuesString[i];

        if (char === "'" || char === '"') {
          insideQuotes = !insideQuotes;
        }

        if (char === ',' && !insideQuotes) {
          values.push(currentVal.trim());
          currentVal = '';
        } else {
          currentVal += char;
        }
      }
      // Push the last value
      values.push(currentVal.trim());

      // Check if columns and values lengths match
      if (columns.length !== values.length) {
        document.getElementById('output').textContent = 'Column and value counts do not match';
        return;
      }

      // Create key-value pairs
      const result = {};
      for (let i = 0; i < columns.length; i++) {
        result[columns[i]] = values[i];
      }

      document.getElementById('output').textContent = `Table: ${tableName}\n` + JSON.stringify(result, null, 2);
    }
  </script>
