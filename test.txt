    <input type="text" id="tableName" placeholder="Enter table name"><br><br>

    <label for="xmlInput">XML Input:</label><br>
    <textarea id="xmlInput" rows="10" cols="50" placeholder="Enter XML content"></textarea><br><br>

    <button onclick="generateSQL()">Generate SQL Query</button>

    <h4>Generated SQL Query:</h4>
    <pre id="sqlOutput"></pre>

    <script>
        function generateSQL() {
            // Get table name and XML input
            const tableName = document.getElementById('tableName').value.trim();
            const xmlInput = document.getElementById('xmlInput').value.trim();

            // Parse XML string to DOM
            const parser = new DOMParser();
            const xmlDoc = parser.parseFromString(`<root>${xmlInput}</root>`, 'text/xml');

            // Prepare the select query parts
            let selectFields = [];

            // Get each child element of the XML
            const fields = xmlDoc.documentElement.children;
            for (let i = 0; i < fields.length; i++) {
                const field = fields[i];
                const alias = field.tagName;
                const column = field.textContent.trim();

                // Format as "column alias"
                selectFields.push(`${column} ${alias}`);
            }

            // Join the fields to form the select part of the SQL query
            const selectPart = selectFields.join(", ");
            const sqlQuery = `SELECT ${selectPart} FROM ${tableName};`;

            // Display the generated SQL query
            document.getElementById('sqlOutput').textContent = sqlQuery;
        }
    </script>
