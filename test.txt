  <pre id="output"></pre>

  <script>
    function convertSQL() {
      const sqlInput = document.getElementById('sqlInput').value;

      // Extract table name, columns, and values
      const tableNameMatch = sqlInput.match(/INSERT INTO\s+([^\s]+)\s+/i);
      const columnsMatch = sqlInput.match(/([^)]+)\s+VALUES/i);
      const valuesMatch = sqlInput.match(/VALUES\s+([^)]+)/i);

      if (!tableNameMatch || !columnsMatch || !valuesMatch) {
        document.getElementById('output').textContent = 'Invalid SQL format';
        return;
      }

      const tableName = tableNameMatch[1];
      const columns = columnsMatch[1].split(',').map(col => col.trim());
      const values = valuesMatch[1].split(/,(?![^']*'|[^"]*")/).map(val => val.trim());

      if (columns.length !== values.length) {
        document.getElementById('output').textContent = 'Column and value counts do not match';
        return;
      }

      // Create key-value pairs
      const result = {};
      for (let i = 0; i < columns.length; i++) {
        result[columns[i]] = values[i];
      }

      document.getElementById('output').textContent = `Table: ${tableName}\n` + JSON.stringify(result, null, 2);
    }
  </script>
