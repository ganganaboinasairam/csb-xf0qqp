function tenor_days(){
    try {
		var td = getFldValue("TENOR_DAYS");
		var tt = getFldValue("NATURE_OF_BILLS");
			
		if(tt == "2"){
			updateTenorMatDate();
		}
		if(tt == "1" && td!=0){
			showMsg("Tenor Days should be zero for Sight Bills");
			return false;
		}else if(tt == "2" && td == "0"){
			showMsg("Tenor Days cannot be zero for Usance Bills");	
			return false;
		}
		else if(tt=="2" && td > 1095){
			showMsg("Tenor Days should not be greater than 1095 days");
			return false;
		
		}
		return true;
      	} catch (e) {
        showExcpt("IIND_Reg_NonLC_Function", e);
    }
}

				TodayDate = Matmon +"/"+ Matdate +"/"+Matyear;

function submitTab() {    

    var bill_amt = document.getElementById("BILL_AMT").value;
	document.getElementById("TEMP_BILL_AMT").value = bill_amt;
	
	
    if (!undertakingsTick()) {
        if ($('#Undertakings').attr('class') != 'accordion active') {
            $('#Undertakings').trigger("click");
        }
        alert("Please accept Terms & Conditions");
		//start of ir 24060022
        return false;
    }
	if(document.getElementById('tick5').style.display=="none"){ 
		alert("Document Upload is mandatory to proceed further"); 
		//end of ir 24060022
        return false;
    }
    var allFieldsValid = true; 
    $('input[class$="_M"], select[class$="_M"], textarea[class$="_M"]').each(function() {
        var value = $(this).val().trim();
        
        if (value === '' || value === '0') {
            allFieldsValid = false; 
            return false; 
        }
    });
    if (!allFieldsValid) {
        alert("Fill Mandatory Fields!");
        $('#prim_confirm').hide();
        return false;
    }
    if (!isCancel) {
        var myWindow = openWin("../Default/SCRN/IIND_RegisterBill_PreviewPage.jsp?", "width=100", "height=100", "fullscreen='yes'");
        myWindow.resizeTo("2000", "2000");
        myWindow.focus();
        return true;
    }
    alert("Fill Mandatory Fields!");
    $('#prim_confirm').hide();
    return false;
}

   	}
