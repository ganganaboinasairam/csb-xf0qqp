String identifier = pr.getProperty("filename"); // "_signed" from properties file
this.list = new ArrayList<>();
Vector<ChannelSftp.LsEntry> fileList = sftpChannel.ls("*.xml");

for (ChannelSftp.LsEntry entry : fileList) {
    if (entry.getFilename().endsWith(identifier)) {
        this.list.add(entry.getFilename());
    }
}

logg.info("Filtered list Value : [" + this.list + "] sftpChannel : [" + sftpChannel + "]");
