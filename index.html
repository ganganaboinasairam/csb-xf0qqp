import java.io.FileOutputStream; import java.util.Base64; public class Base64ToPDF { public static void main(String[] args) { // Replace this with your Base64 string String base64String = "H4sIAAAAAAAA...."; // Your actual Base64 content // Specify the output file path String outputFilePath = "output.pdf"; try (FileOutputStream fos = new FileOutputStream(outputFilePath)) { // Decode the Base64 string byte[] decodedBytes = Base64.getDecoder().decode(base64String); // Write the decoded bytes to a file fos.write(decodedBytes); System.out.println("PDF file created successfully: " + outputFilePath); } catch (Exception e) { e.printStackTrace(); } } }
