function getMainFormTable() { var form = document.getElementById("MAINFORM"); var table = document.createElement("table"); table.border = "1"; // Header var header = table.insertRow(); header.insertCell().innerHTML = "Field Title"; header.insertCell().innerHTML = "Field ID"; header.insertCell().innerHTML = "Field Value"; header.insertCell().innerHTML = "Field Length"; var fields = form.querySelectorAll("input, select, textarea"); fields.forEach(function(field) { var value = ""; // Radio / Checkbox if (field.type === "radio" || field.type === "checkbox") { if (!field.checked) { return; } value = field.value || ""; } else { value = field.value || ""; } // Ignore fields without a value if (value.trim() === "") { return; } /* * Field title: * 1. title attribute * 2. aria-label * 3. associated label * 4. name * 5. id */ var title = field.getAttribute("title") || field.getAttribute("aria-label") || ""; if (!title && field.id) { var label = form.querySelector( 'label[for="' + field.id + '"]' ); if (label) { title = label.innerText.trim(); } } if (!title) { title = field.name || field.id || ""; } /* * Field length: * maxlength if available, * otherwise actual value length */ var maxLength = field.getAttribute("maxlength"); var fieldLength; if (maxLength !== null && maxLength !== "") { fieldLength = maxLength; } else { fieldLength = value.length; } // Add row var row = table.insertRow(); row.insertCell().innerText = title; row.insertCell().innerText = field.id || ""; row.insertCell().innerText = value; row.insertCell().innerText = fieldLength; }); return table; }
