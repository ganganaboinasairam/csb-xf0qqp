private String processRequest(Request request, Response response) throws Exception { try { // 1. Read request body. String rawRequest = request.body(); if (rawRequest == null || rawRequest.trim().isEmpty()) { response.status(400); response.type("application/json"); return new JSONObject().put("error", "Request body is empty").toString(); } // 2. Parse incoming JSON. com.cs.esp.org.json.JSONObject requestJson = new com.cs.esp.org.json.JSONObject(rawRequest); // 3. Load target API configuration. String configPath = ASPathConst.getUserDirPath() + File.separator + "properties" + File.separator + "apitest.properties"; String targetUrl; try (InputStream input = new FileInputStream(configPath)) { Properties properties = new Properties(); properties.load(input); targetUrl = properties.getProperty("API_TEST_URL"); } if (targetUrl == null || targetUrl.trim().isEmpty()) { response.status(500); response.type("application/json"); return new JSONObject().put("error", "Target API URL is not configured").toString(); } // 4. Extract target URL from the incoming request. String url = requestJson.optString("EIS_URL", "").trim(); if (url.isEmpty()) { response.status(400); response.type("application/json"); return new JSONObject().put("error", "EIS_URL is missing").toString(); } /* * 5. Extract EIS_REQUEST. The inner request may contain property names such as * "\"SOURCE_ID\"" instead of "SOURCE_ID". */ Object eisRequestValue = requestJson.opt("EIS_REQUEST"); if (!(eisRequestValue instanceof com.cs.esp.org.json.JSONObject)) { response.status(400); response.type("application/json"); return new JSONObject().put("error", "EIS_REQUEST must be a JSON object").toString(); } com.cs.esp.org.json.JSONObject actualRequest = (com.cs.esp.org.json.JSONObject) eisRequestValue; // 6. Normalize property names recursively. actualRequest = normalizeJsonObject(actualRequest); // 7. Get RRN from EIS_REQUEST, falling back to outer level. String requestReference = actualRequest.optString("REQUEST_REFERENCE_NUMBER", requestJson.optString("REQUEST_REFERENCE_NUMBER", "")); if (requestReference.trim().isEmpty()) { response.status(400); response.type("application/json"); return new JSONObject().put("error", "REQUEST_REFERENCE_NUMBER is missing").toString(); } // 8. Log the normalized request for debugging. System.out.println("Target URL: " + url); System.out.println("Request Reference: " + requestReference); System.out.println("Normalized EIS_REQUEST: " + actualRequest); // 9. Forward request to the existing REST client. RestClient restClient = new RestClient(); //String apiResponse = restClient.SendPostRequest(actualRequest, requestReference, url, null, null, null, false); // Convert to standard JSON with double-quoted keys. String jsonRequest = new org.json.JSONObject( actualRequest.toString() ).toString(); // Parse into standard org.json JSONObject. org.json.JSONObject serverRequest = new org.json.JSONObject(jsonRequest); // Log the exact JSON representation. System.out.println("Server Request: " + serverRequest.toString(2)); // Send the request using the standard JSON object. String apiResponse = restClient.SendPostRequest( serverRequest, requestReference, url, null, null, null, false ); // 10. Return the target API response. response.status(200); response.type("application/json"); return apiResponse; } catch (org.json.JSONException e) { response.status(400); response.type("application/json"); return new JSONObject().put("error", "Invalid JSON request").toString(); } catch (Exception e) { LogUtil.error(e); response.status(500); response.type("application/json"); return new JSONObject().put("error", "Failed to process API request").toString(); } }
