package com.chat.client; import java.io.BufferedReader; import java.io.IOException; import java.io.InputStreamReader; import java.io.OutputStream; import java.net.HttpURLConnection; import java.net.URL; import java.net.URLEncoder; import java.nio.file.Files; import java.nio.file.Paths; import javax.servlet.ServletException; import javax.servlet.annotation.WebServlet; import javax.servlet.http.HttpServlet; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletResponse; import org.json.JSONObject; import org.slf4j.Logger; import org.slf4j.LoggerFactory; import com.chat.config.AsymmetricSignatureUtil; /** * ChatClientServlet * - Proxies signed HTTP requests to the chat server * - Offers GET /chatClient/getSignedWsUrl to return a signed WebSocket URL for direct browser connections * * NOTE: server-side WebSocket proxying (connectWS) REMOVED. */ @WebServlet("/chatClient/*") public class ChatClientServlet extends HttpServlet { private static final Logger log = LoggerFactory.getLogger(ChatClientServlet.class); private static final String CLIENT_ID = "example-client"; private static final String PRIVATE_KEY_PATH = "keys/example-client.pem"; private static final String CHAT_SERVER_BASE_URL = "http://localhost:8080"; // Adjust as needed private static final String CHAT_SERVER_WS_URL = "ws://localhost:8080/chatServer"; // Adjust as needed private String privateKeyBase64; @Override public void init() throws ServletException { try { // Load private key (PEM) into a base64 string (strip headers) String privateKeyPem = new String(Files.readAllBytes(Paths.get(PRIVATE_KEY_PATH))) .replaceAll("\\n", "") .replaceAll("\\r", "") .replace("-----BEGIN PRIVATE KEY-----", "") .replace("-----END PRIVATE KEY-----", "") .trim(); privateKeyBase64 = privateKeyPem; log.info("Loaded private key for client: {}", CLIENT_ID); } catch (Exception e) { throw new ServletException("Failed to load private key", e); } } @Override protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException { String pathInfo = req.getPathInfo(); if (pathInfo == null || pathInfo.equals("/")) { resp.sendError(HttpServletResponse.SC_BAD_REQUEST, "Invalid path"); return; } String endpoint = pathInfo.substring(1); // Remove leading / if ("getSignedWsUrl".equals(endpoint)) { // Return a signed WebSocket URL for the browser to connect directly handleGetSignedWsUrl(req, resp); } else { // Proxy other GET requests to chat server (signed) handleHttpRequest("GET", endpoint, req, resp); } } @Override protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException { String pathInfo = req.getPathInfo(); if (pathInfo == null || pathInfo.equals("/")) { resp.sendError(HttpServletResponse.SC_BAD_REQUEST, "Invalid path"); return; } String endpoint = pathInfo.substring(1); // Proxy POSTs to chat server (signed) handleHttpRequest("POST", endpoint, req, resp); } /** * Proxy signed HTTP request to chat server. * Supports GET and POST (reads body for POST). */ private void handleHttpRequest(String method, String endpoint, HttpServletRequest req, HttpServletResponse resp) throws IOException { try { // Read request body if POST String body = null; if ("POST".equalsIgnoreCase(method)) { BufferedReader reader = req.getReader(); body = reader.lines().reduce("", (a, b) -> a + b); } // Generate timestamp String timestamp = String.valueOf(System.currentTimeMillis()); // Create canonical string String canonicalString = AsymmetricSignatureUtil.createCanonicalString(method, "/" + endpoint, body, timestamp); // Sign the canonical string String signature = AsymmetricSignatureUtil.sign(canonicalString, privateKeyBase64); // Forward request to chat server String query = req.getQueryString() == null ? "" : "?" + req.getQueryString(); URL url = new URL(CHAT_SERVER_BASE_URL + "/" + endpoint + query); HttpURLConnection conn = (HttpURLConnection) url.openConnection(); conn.setRequestMethod(method); conn.setRequestProperty("X-Timestamp", timestamp); conn.setRequestProperty("X-Signature", signature); conn.setRequestProperty("X-Client-ID", CLIENT_ID); if (body != null) { conn.setDoOutput(true); try (OutputStream os = conn.getOutputStream()) { os.write(body.getBytes("UTF-8")); } } // Read response
