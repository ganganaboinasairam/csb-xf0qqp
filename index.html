const versionLabels = JSON.parse(vesion_lables); const crStatusArray = JSON.parse(CR_Status); // Convert array to lookup object const crStatus = {}; crStatusArray.forEach(obj => { const key = Object.keys(obj)[0]; crStatus[key] = obj[key]; }); function checkDependencies(crNo) { const currentRevision = versionLabels[crNo]; if (!currentRevision) { alert("CR not found"); return; } const blocked = []; Object.entries(versionLabels).forEach(([cr, rev]) => { // Dependency if (compareRevision(rev, currentRevision) <= 0) { if (crStatus[cr] === "WBT APPROVED") { blocked.push({ cr: cr, revision: rev, status: crStatus[cr] }); } } }); if (blocked.length === 0) { alert("All dependencies are OK."); } else { console.table(blocked); alert("Found " + blocked.length + " dependency(s) in WBT APPROVED."); } } function compareRevision(r1, r2) { const a = r1.split(".").map(Number); const b = r2.split(".").map(Number); const len = Math.max(a.length, b.length); for (let i = 0; i < len; i++) { const x = a[i] || 0; const y = b[i] || 0; if (x > y) return 1; if (x < y) return -1; } return 0; }
