private static void generateExcel(File[] files, String outputPath) throws Exception { Workbook workbook = new XSSFWorkbook(); Sheet sheet = workbook.createSheet("Evidence"); CreationHelper helper = workbook.getCreationHelper(); Drawing drawing = sheet.createDrawingPatriarch(); // Set column widths (0 to 6) for (int c = 0; c <= 6; c++) { sheet.setColumnWidth(c, 30 * 256); } for (int i = 0; i < files.length; i++) { File file = files[i]; byte[] imageBytes = Files.readAllBytes(file.toPath()); int pictureIdx = workbook.addPicture(imageBytes, Workbook.PICTURE_TYPE_PNG); // Grid logic int col = (i % 2 == 0) ? 0 : 5; // left / right int row = (i / 2) * 10; // block of 10 rows per pair // Ensure rows exist and set height for 10-row block for (int r = row; r < row + 10; r++) { Row tempRow = sheet.getRow(r); if (tempRow == null) tempRow = sheet.createRow(r); tempRow.setHeightInPoints(34); } // Anchor image ClientAnchor anchor = helper.createClientAnchor(); anchor.setCol1(col); anchor.setRow1(row); Picture pict = drawing.createPicture(anchor, pictureIdx); pict.resize(); } FileOutputStream fos = new FileOutputStream(outputPath); workbook.write(fos); fos.close(); }
