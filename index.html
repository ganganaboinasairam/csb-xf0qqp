import java.util.ArrayDeque; import java.util.Deque; import java.util.Set; public class IRContext { private final Deque irStack = new ArrayDeque<>(); private final Set disabledIRs; public IRContext(Set disabledIRs) { this.disabledIRs = disabledIRs; } public void enter(String ir) { irStack.push(ir); } public void exit(String ir) { if (!irStack.isEmpty() && irStack.peek().equals(ir)) { irStack.pop(); } } /** * Disable ONLY when the disabled IR is the current active IR */ public boolean shouldDisable() { return !irStack.isEmpty() && disabledIRs.contains(irStack.peek()); } }
