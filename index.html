private void flush(List buffer, List out, FileType type) { if (buffer.isEmpty()) return; String indent = CommentUtil.indent(buffer.get(0)); out.add(CommentUtil.startMarker(tag, indent, type)); if (buffer.size() == 1) { // Single line → single comment out.add(CommentUtil.commentLine(buffer.get(0).trim(), type)); } else { if (type == FileType.JS) { // JS multi-line block out.add(indent + "/*"); for (String l : buffer) { out.add(l); } out.add(indent + "*/"); } else { // XML / JSP multi-line block (SINGLE COMMENT) out.add(indent + ""); } } out.add(CommentUtil.endMarker(tag, indent, type)); buffer.clear(); }
