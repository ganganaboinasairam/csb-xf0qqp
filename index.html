package logExtractor; import java.io.*; import java.time.LocalTime; import java.time.format.DateTimeFormatter; import java.util.regex.*; import javax.servlet.annotation.WebServlet; import javax.servlet.http.*; @WebServlet("/extract-log") public class LogExtractorServlet extends HttpServlet { private static final Pattern TIMESTAMP_PATTERN = Pattern.compile( "\\[(\\d{4}-\\d{2}-\\d{2} " + "\\d{2}:\\d{2}:\\d{2}\\.\\d{3})\\]" ); private static final DateTimeFormatter LOG_TIME_FORMAT = DateTimeFormatter.ofPattern( "HH:mm:ss.SSS" ); @Override protected void doPost(HttpServletRequest request, HttpServletResponse response) throws IOException { response.setContentType("text/plain"); try { StringBuilder jsonBuilder = new StringBuilder(); BufferedReader requestReader = request.getReader(); String reqLine; while ((reqLine = requestReader.readLine()) != null) { jsonBuilder.append(reqLine); } String json = jsonBuilder.toString(); String inputFilePath = extractJsonValue( json, "inputFilePath" ); String outputFilePath = extractJsonValue( json, "outputFilePath" ); String fromTimeStr = extractJsonValue( json, "fromTime" ); String toTimeStr = extractJsonValue( json, "toTime" ); int linesPerFile = Integer.parseInt( extractJsonValue( json, "linesPerFile" ) ); LocalTime fromTime = LocalTime.parse( fromTimeStr ); LocalTime toTime = LocalTime.parse( toTimeStr ); File outputFile = new File(outputFilePath); String outputDir = outputFile.getParent(); String baseName = outputFile.getName(); if (baseName.contains(".")) { baseName = baseName.substring( 0, baseName.lastIndexOf(".") ); } BufferedReader fileReader = new BufferedReader( new FileReader( inputFilePath ) ); int currentFileNumber = 1; int currentLineCount = 0; int matchedBlocks = 0; BufferedWriter writer = createWriter( outputDir, baseName, currentFileNumber ); String line; StringBuilder blockBuffer = new StringBuilder(); boolean insideBlock = false; boolean shouldWriteBlock = false; while ((line = fileReader.readLine()) != null) { if (line.contains("[start]")) { insideBlock = true; shouldWriteBlock = false; blockBuffer.setLength(0); } if (insideBlock) { blockBuffer .append(line) .append("\n"); } if (line.contains("[INFO]")) { LocalTime logTime = extractTime(line); if (logTime != null) { if (!logTime.isBefore(fromTime) && !logTime.isAfter(toTime)) { shouldWriteBlock = true; } } } if (line.contains("[end]")) { insideBlock = false; if (shouldWriteBlock) { String block = blockBuffer.toString(); int blockLines = block.split("\n").length; if (currentLineCount + blockLines > linesPerFile) { writer.close(); currentFileNumber++; currentLineCount = 0; writer = createWriter( outputDir, baseName, currentFileNumber ); } writer.write(block); writer.newLine(); currentLineCount += blockLines; matchedBlocks++; } } } writer.close(); fileReader.close(); response.getWriter().write( "Extraction completed successfully\n" + "Matched Blocks : " + matchedBlocks + "\nOutput Files Created : " + currentFileNumber ); } catch (Exception e) { e.printStackTrace(); response.getWriter().write( "ERROR : " + e.getMessage() ); } } private BufferedWriter createWriter( String outputDir, String baseName, int fileNumber) throws IOException { return new BufferedWriter( new FileWriter( outputDir + File.separator + baseName + "_" + fileNumber + ".log" ) ); } private LocalTime extractTime( String line) { try { Matcher matcher = TIMESTAMP_PATTERN .matcher(line); if (matcher.find()) { String fullTimestamp = matcher.group(1); String timePart = fullTimestamp.substring(11); return LocalTime.parse( timePart, LOG_TIME_FORMAT ); } } catch (Exception e) { e.printStackTrace(); } return null; } private String extractJsonValue( String json, String key) { String search = "\"" + key + "\":"; int start = json.indexOf(search); if (start == -1) { return ""; } start += search.length(); while (json.charAt(start) == ' ' || json.charAt(start) == '"') { start++; } int end = start; while (end < json.length() && json.charAt(end) != '"' && json.charAt(end) != ',' && json.charAt(end) != '}') { end++; } return json .substring(start, end) .trim(); } }
