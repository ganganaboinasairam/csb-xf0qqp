private static void moveContents(Path source, Path destination) { if (Files.exists(source)) { try (DirectoryStream stream = Files.newDirectoryStream(source)) { for (Path entry : stream) { Path targetPath = destination.resolve(entry.getFileName()); try { if (Files.isDirectory(entry)) { if (!Files.exists(targetPath)) { Files.createDirectory(targetPath); } moveContents(entry, targetPath); // Recursive call for directories Files.delete(entry); // Remove empty folder after contents are moved } else { Files.move(entry, targetPath); System.out.println("Moved: " + entry.toAbsolutePath() + " to " + targetPath.toAbsolutePath()); } } catch (IOException e) { // Skip the file if it's in use or any other error occurs System.err.println("Skipped (locked/in use): " + entry.toAbsolutePath()); } } } catch (IOException e) { System.err.println("Failed to read source directory: " + source.toAbsolutePath()); e.printStackTrace(); } } else { System.err.println("Source folder does not exist: " + source.toAbsolutePath()); } }
