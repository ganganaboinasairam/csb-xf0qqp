public String UpdateSLATEMP(DocumentImpl paraDOM) { try { APLog.reportLog("ParaDOM for SLA insert and delete" + XMLManager.convertToString(paraDOM)); Node nodeDomData = XMLManager.findChildNode(paraDOM.getDocumentElement(), "domData"); // Fetch common nodes String userIdValue = XMLManager.getChildNodeValue(nodeDomData, "C_USER_ID", "value", true); String cUnitCodeValue = XMLManager.getChildNodeValue(nodeDomData, "C_UNIT_CODE", "value", true); String slaUserTypeValue = XMLManager.getChildNodeValue(nodeDomData, "C_SLA_USER_TYPE", "value", true); // Delete existing SLA temp processDeleteSLATemp(userIdValue, cUnitCodeValue); // Process each SLA_GRP_ID value Node slaGrpID = XMLManager.findChildNode(nodeDomData, "SLA_GRP_ID"); NodeList values = slaGrpID.getChildNodes(); for (int i = 0; i < values.getLength(); i++) { Node valueNode = values.item(i); if (valueNode.getNodeType() == Node.ELEMENT_NODE && "value".equals(valueNode.getNodeName())) { String slaGrpIDValue = valueNode.getTextContent().trim(); if (!slaGrpIDValue.isEmpty()) { processInsertSLATemp(slaGrpIDValue, userIdValue, cUnitCodeValue, slaUserTypeValue); } } } } catch (Exception e) { APLog.reportLog("An error has occurred while updating the flag value : " + e.getMessage()); APLog.reportErrLog(e.getMessage()); } return null; }
