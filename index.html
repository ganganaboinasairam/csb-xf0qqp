function reArrangeAmtFields() { $.each(amountFields, function(_, sec) { var ccy = $('#' + sec.ccy); var amt = $('#' + sec.amt); if (ccy.length === 0 || amt.length === 0) { console.warn(`Missing elements for: ${sec.ccy} or ${sec.amt}`); return; // Skip this pair if elements are missing } var ccyDiv = ccy.parent(); // Apply flex styling ccyDiv.css({ display: 'flex', alignItems: 'center', gap: '10px' }); // Move amount field inside currency div if not already inside if (!amt.parent().is(ccyDiv)) { ccyDiv.append(amt); } }); }
