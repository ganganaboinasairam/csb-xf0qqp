package com.tcs.git.ui; import javax.swing.*; import java.awt.*; import java.util.*; import java.util.List; public class FileSelectionDialog extends JDialog { private static final long serialVersionUID = 1L; private final Map fileMap = new LinkedHashMap<>(); private boolean confirmed = false; public FileSelectionDialog(Frame parent, List files) { super(parent, "Select Files to Commit", true); setSize(500, 400); setLocationRelativeTo(parent); initUI(files); } private void initUI(List files) { JPanel main = new JPanel(new BorderLayout(5, 5)); /* ================= FILE LIST ================= */ JPanel listPanel = new JPanel(); listPanel.setLayout(new BoxLayout(listPanel, BoxLayout.Y_AXIS)); for (String file : files) { JCheckBox cb = new JCheckBox(file, true); fileMap.put(cb, file); listPanel.add(cb); } JScrollPane scroll = new JScrollPane(listPanel); /* ================= BUTTON BAR ================= */ JButton selectAll = new JButton("Select All"); JButton deselectAll = new JButton("Deselect All"); JButton ok = new JButton("Commit"); JButton cancel = new JButton("Cancel"); selectAll.addActionListener(e -> fileMap.keySet().forEach(cb -> cb.setSelected(true)) ); deselectAll.addActionListener(e -> fileMap.keySet().forEach(cb -> cb.setSelected(false)) ); ok.addActionListener(e -> { confirmed = true; dispose(); }); cancel.addActionListener(e -> dispose()); JPanel btnBar = new JPanel(new FlowLayout(FlowLayout.RIGHT)); btnBar.add(selectAll); btnBar.add(deselectAll); btnBar.add(ok); btnBar.add(cancel); main.add(scroll, BorderLayout.CENTER); main.add(btnBar, BorderLayout.SOUTH); add(main); } public boolean isConfirmed() { return confirmed; } public List getSelectedFiles() { List selected = new ArrayList<>(); for (Map.Entry e : fileMap.entrySet()) { if (e.getKey().isSelected()) { selected.add(e.getValue()); } } return selected; } } public static List getChangedFiles(String region) throws Exception { File repo = new File(Config.get("git.repo.path." + region)); String status = GitCli.run(repo, "status", "--porcelain"); List files = new ArrayList<>(); for (String line : status.split("\\R")) { if (line.length() > 3) { files.add(line.substring(3).trim()); } } return files; } public static List getChangedFiles(String region) throws Exception { File repo = new File(Config.get("git.repo.path." + region)); String status = GitCli.run(repo, "status", "--porcelain"); List files = new ArrayList<>(); for (String line : status.split("\\R")) { if (line.length() > 3) { files.add(line.substring(3).trim()); } } return files; } private void runTask() { String region = regionBox.getSelectedItem().toString().toLowerCase(); String crNo = crField.getText().trim(); if (!crNo.matches("\\d+")) { JOptionPane.showMessageDialog(this, "Invalid CR Number"); return; } try { List files = GitAutomationService.getChangedFiles(region); if (files.isEmpty()) { JOptionPane.showMessageDialog(this, "No changes to commit"); return; } FileSelectionDialog dlg = new FileSelectionDialog(this, files); dlg.setVisible(true); if (!dlg.isConfirmed()) { outputArea.append("Commit cancelled by user\n"); return; } List selected = dlg.getSelectedFiles(); if (selected.isEmpty()) { JOptionPane.showMessageDialog(this, "No files selected"); return; } executeCommit(region, crNo, selected); } catch (Exception e) { outputArea.append("ERROR: " + e.getMessage() + "\n"); } } private void executeCommit( String region, String crNo, List files) { SwingWorker worker = new SwingWorker() { protected Void doInBackground() throws Exception { GitAutomationService.commitSelected( region, crNo, files, msg -> publish(msg) ); return null; } protected void process(List msgs) { msgs.forEach(m -> outputArea.append(m + "\n")); } }; worker.execute(); } public static void commitSelected( String region, String crNo, List files, Consumer log) throws Exception { File repo = new File(Config.get("git.repo.path." + region)); String branch = Config.get("git.branch.name." + region); log.accept("Staging selected files..."); for (String f : files) { GitCli.run(repo, "add", f); log.accept("Added: " + f); } String msg = CommitMessageGenerator.generate(repo, crNo); log.accept("Committing: " + msg); GitCli.run(repo, "commit", "-m", msg); log.accept("Rebasing..."); GitCli.run(repo, "pull", "--rebase", "origin", branch); log.accept("Pushing..."); GitCli.run(repo, "push", "origin", branch); log.accept("Commit pushed successfully"); }
