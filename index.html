SELECT inlc.C_MAIN_REF AS LC_Reference, inad.C_MAIN_REF AS Advising_Ref, inlc.LC_AMT AS LC_Amount, SUM(CASE WHEN emamd.C_FUNC_SHORT_NAME IN ('RailLCAmedDEO', 'eTradeSBIAmendLC') AND emamd.C_TRX_STATUS = 'M' THEN emamd.INC_AMT ELSE 0 END) AS Increment_LC_Amount, SUM(CASE WHEN emamd.C_FUNC_SHORT_NAME IN ('RailLCAmedDEO', 'eTradeSBIAmendLC') AND emamd.C_TRX_STATUS = 'M' THEN emamd.DEC_AMT ELSE 0 END) AS Decrement_LC_Amount, inlc.AMD_NO AS Amendment_Number, inlc.LC_BAL_PRSNT AS Presentation_Balance_INLC, SUM(iine.TTL_NEGO_AMT) AS Total_Negotiated_Amount, inad.PRES_BAL_LC AS Presentation_Balance_INAD FROM INAD_MASTER inad FULL JOIN INLC_MASTER inlc ON inad.LC_NO = inlc.C_MAIN_REF FULL JOIN IINE_MASTER iine ON inad.LC_NO = iine.LC_NO FULL JOIN INLC_EM_AMENDMENT emamd ON inlc.C_MAIN_REF = emamd.C_MAIN_REF WHERE inad.LC_NO IN ( '0069121LC0000035', '0069120LC1000289', '0069121LC0000012', '0069120LC1000464', '0069123LC0000102', '0069123LC0000148', '0069123LC0000147', '0069123LC0000123' ) GROUP BY inlc.C_MAIN_REF, inad.C_MAIN_REF, inad.PRES_BAL_LC, inlc.LC_AMT, inlc.LC_BAL_PRSNT, inlc.INC_AMT, inlc.DEC_AMT, inlc.AMD_NO;
