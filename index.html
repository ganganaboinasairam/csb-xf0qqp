import java.nio.charset.*; import java.nio.file.*; import java.io.*; import java.util.*; public final class FileUtil { private FileUtil() {} public static List safeReadAllLines(Path file) throws IOException { try { return Files.readAllLines(file, StandardCharsets.UTF_8); } catch (MalformedInputException e) { // fallback for legacy encodings return Files.readAllLines(file, Charset.defaultCharset()); } } }
