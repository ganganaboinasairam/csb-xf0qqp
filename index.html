package com.tcs.git.servlet; import java.io.BufferedReader; import java.io.IOException; import java.io.PrintWriter; import java.util.ArrayList; import java.util.Collections; import java.util.List; import java.util.function.Consumer; import javax.servlet.annotation.WebServlet; import javax.servlet.http.HttpServlet; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletResponse; import com.tcs.git.GitAutomationService; import com.tcs.git.util.AppLogger; /** * Streams commit-and-push progress via Server-Sent Events (SSE). * * Step 1 â€” uses AppLogger.combined() so every log line goes to both * the browser terminal and the daily log file simultaneously. * * GET /api/git/commit/stream?region=ee&cr=CR-1234 */ @WebServlet("/api/git/commit/stream") public class CommitServlet extends HttpServlet { private static final long serialVersionUID = 1L; @Override protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws IOException { String region = sanitize(req.getParameter("region")); String cr = req.getParameter("cr"); int commitAll = Integer.parseInt(req.getParameter("isAllFilesChecked")); List files = readLines(req); if (region == null || cr == null || cr.isEmpty()) { resp.sendError(400, "region and cr are required"); return; } resp.setContentType("text/event-stream"); resp.setCharacterEncoding("UTF-8"); resp.setHeader("Cache-Control", "no-cache"); resp.setHeader("X-Accel-Buffering", "no"); PrintWriter out = resp.getWriter(); // Step 1: combined logger = file AND browser SSE stream AppLogger.logSeparator(); AppLogger.log("[COMMIT] region=" + region + " cr=" + cr); Consumer logger = AppLogger.combined( line -> sendEvent(out, "log", line) ); try { boolean committed = GitAutomationService.commitSelected( region, cr, files, logger, stage -> { AppLogger.log("[STAGE] " + stage.name()); sendEvent(out, "stage", stage.name()); }, pct -> sendEvent(out, "progress", String.valueOf(pct)), commitAll ); if (committed) { sendEvent(out, "done", ""); AppLogger.log("[COMMIT] completed successfully cr=" + cr); } else { sendEvent(out, "noop", "No files added to commit. Click Add first, select files, then commit and push."); AppLogger.log("[COMMIT] skipped - no staged files cr=" + cr); } } catch (Exception e) { // Step 3 â€” full stacktrace to file; short message to browser AppLogger.logException("CommitServlet: commit failed cr=" + cr, e, logger); sendEvent(out, "error", e.getMessage()); } finally { AppLogger.logSeparator(); out.flush(); } } private void sendEvent(PrintWriter out, String event, String data) { String safe = data == null ? "" : data.replace("\n", "\\n"); out.write("event: " + event + "\n"); out.write("data: " + safe + "\n\n"); out.flush(); } private String sanitize(String s) { if (s == null) return null; s = s.trim().toLowerCase(); return (s.equals("ee") || s.equals("ce")) ? s : null; } private List readLines(HttpServletRequest req) throws IOException { List files = new ArrayList<>(); try (BufferedReader reader = req.getReader()) { String line; while ((line = reader.readLine()) != null) { line = line.trim(); if (!line.isEmpty()) { files.add(line); } } } return files; } }
