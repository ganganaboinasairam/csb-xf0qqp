private void waitForPrompt() throws IOException { long startTime = System.currentTimeMillis(); long timeout = 10000; // 10 seconds timeout StringBuilder output = new StringBuilder(); String line; while (System.currentTimeMillis() - startTime < timeout) { if (reader.ready() && (line = reader.readLine()) != null) { output.append(line).append("\n"); System.out.println("SFTP >> " + line); if (line.trim().endsWith("sftp>")) { return; } } } System.out.println("Timed out waiting for sftp> prompt."); }
