import java.io.*; import java.text.SimpleDateFormat; import java.util.*; public class FolderFileListTool { public static void main(String[] args) { // Base folder (root) String basePath = "C:\\your\\target\\folder"; // Change this String outputPath = "C:\\your\\output\\file_list.txt"; // Change this File baseDir = new File(basePath); if (!baseDir.exists() || !baseDir.isDirectory()) { System.out.println("Invalid folder path."); return; } // TreeMap to sort and group by relative path + filename TreeMap> fileMap = new TreeMap<>(); collectFiles(baseDir, baseDir, fileMap); try (BufferedWriter writer = new BufferedWriter(new FileWriter(outputPath))) { writer.write(String.format("%-40s %-70s %-20s", "File Name", "File Path", "Last Modified")); writer.newLine(); writer.write("-".repeat(135)); writer.newLine(); SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd HH:mm:ss"); for (Map.Entry> entry : fileMap.entrySet()) { for (File file : entry.getValue()) { String fileName = file.getName(); String filePath = file.getAbsolutePath(); String lastModified = sdf.format(new Date(file.lastModified())); writer.write(String.format("%-40s %-70s %-20s", fileName, filePath, lastModified)); writer.newLine(); } } System.out.println("File list saved successfully to: " + outputPath); } catch (IOException e) { e.printStackTrace(); } } private static void collectFiles(File baseDir, File currentDir, TreeMap> map) { File[] files = currentDir.listFiles(); if (files == null) return; for (File file : files) { if (file.isFile()) { String relativePath = baseDir.toPath().relativize(file.toPath()).toString(); String key = relativePath.replace(File.separatorChar, '/'); // normalize separator map.computeIfAbsent(key, k -> new ArrayList<>()).add(file); } else if (file.isDirectory()) { collectFiles(baseDir, file, map); } } } }
