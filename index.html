package com.chat; import javax.servlet.ServletException; import javax.servlet.annotation.WebServlet; import javax.servlet.http.HttpServlet; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletResponse; import com.tcs.chat.util.ChatQADao; import com.google.gson.Gson; import com.google.gson.JsonSyntaxException; import java.io.BufferedReader; import java.io.IOException; import java.io.PrintWriter; @WebServlet("/getBotReply") public class GetBotReply extends HttpServlet { private static final long serialVersionUID = 1L; private Gson gson = new Gson(); ChatQADao dao = new ChatQADao(); protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException { try { // Set response type response.setContentType("application/json"); PrintWriter out = response.getWriter(); // Read the request body StringBuilder sb = new StringBuilder(); BufferedReader reader = request.getReader(); String line; while ((line = reader.readLine()) != null) { sb.append(line); } // Parse the incoming JSON String jsonInput = sb.toString(); UserQuestion userQuestion = gson.fromJson(jsonInput, UserQuestion.class); // Generate a response based on the question String botResponse = null; try { botResponse = dao.findBestAnswer(userQuestion.getQuestion()); } catch (Exception e) { // TODO Auto-generated catch block e.printStackTrace(); } System.out.println("botResponse is :: "+botResponse); // Send the response back to the client out.print(gson.toJson(botResponse)); out.flush(); } catch (JsonSyntaxException e) { // TODO Auto-generated catch block e.printStackTrace(); } } private class UserQuestion { private String question; public String getQuestion() { return question; } } }
