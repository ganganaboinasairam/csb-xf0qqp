import javax.swing.*; import java.awt.*; import java.awt.event.*; import java.util.Timer; import java.util.TimerTask; public class WaterReminder { private static final int REMINDER_MINUTES = 30; // Change this to 20, 40 etc private static Timer timer = new Timer(); public static void main(String[] args) { System.out.println("Water Reminder Started..."); timer.schedule(new TimerTask() { @Override public void run() { showReminder(); } }, 0, REMINDER_MINUTES * 60 * 1000); } private static void showReminder() { SwingUtilities.invokeLater(() -> { JFrame frame = new JFrame("Drink Water"); frame.setUndecorated(true); frame.setExtendedState(JFrame.MAXIMIZED_BOTH); frame.setAlwaysOnTop(true); frame.setDefaultCloseOperation(JFrame.DO_NOTHING_ON_CLOSE); JPanel panel = new JPanel(); panel.setBackground(new Color(30, 144, 255)); panel.setLayout(new BorderLayout()); JLabel label = new JLabel("TIME TO DRINK WATER!", SwingConstants.CENTER); label.setFont(new Font("Arial", Font.BOLD, 60)); label.setForeground(Color.WHITE); JButton btn = new JButton("I Drank Water"); btn.setFont(new Font("Arial", Font.BOLD, 40)); btn.setBackground(Color.GREEN); btn.setForeground(Color.BLACK); btn.addActionListener(e -> frame.dispose()); panel.add(label, BorderLayout.CENTER); panel.add(btn, BorderLayout.SOUTH); frame.add(panel); frame.setVisible(true); }); } }
