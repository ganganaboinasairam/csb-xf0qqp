function adjustFieldsOnTextChange() { $('.amount-in-words').each(function () { var amountInWords = $(this); var parentDiv = amountInWords.closest('.field-wrapper'); // Get the parent wrapper var selectWrapper = parentDiv.find('.select-wrapper'); // Get the select-wrapper div var ccyField = selectWrapper.find('select'); // Currency field var amtField = selectWrapper.find('input'); // Amount field // Get the height of the text inside the span var textHeight = amountInWords.outerHeight(); // Calculate new height for the fields based on text size var baseHeight = 40; // Default field height var newHeight = baseHeight - (textHeight / 5); // Reduce height dynamically newHeight = Math.max(25, newHeight); // Ensure min height of 25px // Apply styles with higher priority ccyField.attr('style', `height: ${newHeight}px !important;`); amtField.attr('style', `height: ${newHeight}px !important;`); // Ensure text wraps properly amountInWords.css({ display: 'block', width: '100%', whiteSpace: 'normal', wordWrap: 'break-word', overflowWrap: 'break-word' }); // Allow select-wrapper to wrap elements when needed selectWrapper.css('flex-wrap', 'wrap'); }); } // Run when the page loads $(document).ready(function () { adjustFieldsOnTextChange(); }); // Detect changes dynamically var observer = new MutationObserver(function () { adjustFieldsOnTextChange(); }); $('.amount-in-words').each(function () { observer.observe(this, { childList: true, subtree: true, characterData: true }); });
