private String escape(String s) { if (s == null) { return ""; } StringBuilder out = new StringBuilder(); for (char c : s.toCharArray()) { switch (c) { case '\\': out.append("\\\\"); break; case '"': out.append("\\\""); break; case '\n': out.append("\\n"); break; case '\r': out.append("\\r"); break; case '\t': out.append("\\t"); break; case '\b': out.append("\\b"); break; case '\f': out.append("\\f"); break; default: // escape all control chars if (c < 32) { out.append(String.format("\\u%04x", (int) c)); } else { out.append(c); } } } return out.toString(); }
