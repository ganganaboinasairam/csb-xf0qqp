class Solution { public static int[] twoSum(int[] nums, int target) { try { int[] output = new int[2]; int sum = target +1 ; for (int i = 0; i <= nums.length ; i++) { for (int j = i; j < nums.length ; j++) { if ( i != j) { sum = nums[i] + nums[j]; } if (sum == target) { output[0] = i; output[1] = j; sum = 0; return output; } } } return output; } catch (Exception e) { e.printStackTrace(); return null; } } }
