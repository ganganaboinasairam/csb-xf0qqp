function adjustFieldsOnTextChange() { $('.amount-in-words').each(function () { var amountInWords = $(this); var parentDiv = amountInWords.closest('.field-wrapper'); // Get the parent wrapper var selectWrapper = parentDiv.find('.select-wrapper'); // Get the select-wrapper div var ccyField = selectWrapper.find('select'); // Currency field var amtField = selectWrapper.find('input'); // Amount field // Get the height of the text inside the span var textHeight = amountInWords[0].scrollHeight; // Calculate new height based on text size var newHeight = Math.max(30, 60 - (textHeight - 20)); // Ensure it doesn't go below 30px // Apply the new height dynamically ccyField.css('height', newHeight + 'px'); amtField.css('height', newHeight + 'px'); // Ensure text wraps correctly within its parent amountInWords.css({ display: 'block', width: '100%', whiteSpace: 'normal', wordWrap: 'break-word' }); // Ensure select-wrapper handles text wrapping selectWrapper.css('flex-wrap', 'wrap'); }); } // Call function when the page loads $(document).ready(function () { adjustFieldsOnTextChange(); }); // If text inside the span changes dynamically (e.g., AJAX update), use MutationObserver var observer = new MutationObserver(function () { adjustFieldsOnTextChange(); }); $('.amount-in-words').each(function () { observer.observe(this, { childList: true, subtree: true, characterData: true }); });
