function determineMode(data) { const { cardRate, dealerRate, forwardRate, onlineRate, readyAmount = 0, totalTransaction = 0, eefcDebitTxnCCY = 0, eefcAvailable = false, customerRateObtained = false, nostroCreditReceived = false, forwardUtilizationAmount = 0, eefcSameCurrencyAmount = 0, eefcConvertedAmount = 0 } = data; // ----- IDEA CONDITIONS ----- // Only Card Rate if (cardRate && !dealerRate && !forwardRate && !onlineRate) { return "IDEA"; } // Card Rate + Online Rate if (cardRate && onlineRate && !dealerRate && !forwardRate && totalTransaction === readyAmount) { return "IDEA"; } // Only Online Rate if (onlineRate && !cardRate && !dealerRate && !forwardRate && totalTransaction === readyAmount) { return "IDEA"; } // Card Rate + Online Rate + EEFC Debit if (cardRate && onlineRate && readyAmount + eefcDebitTxnCCY === totalTransaction) { return "IDEA"; } // Only Online Rate + EEFC Debit if (onlineRate && !cardRate && !dealerRate && !forwardRate && (readyAmount + eefcDebitTxnCCY === totalTransaction)) { return "IDEA"; } // Entire transaction from EEFC if (eefcAvailable && eefcDebitTxnCCY === totalTransaction) { return "IDEA"; } // ----- MERCURY CONDITIONS ----- // Customer/Branch rate already obtained if (customerRateObtained) { return "MERCURY"; } // Nostro credit received if (nostroCreditReceived) { return "MERCURY"; } // Forward utilization if (forwardUtilizationAmount > 0) { return "MERCURY"; } // EEFC + converted currency if (eefcAvailable && readyAmount === (eefcSameCurrencyAmount + eefcConvertedAmount)) { return "MERCURY"; } // Entire ready amount from EEFC conversion if (eefcAvailable && readyAmount === eefcConvertedAmount) { return "MERCURY"; } // Any Dealer Rate if (dealerRate) { return "MERCURY"; } // Any Forward Rate if (forwardRate) { return "MERCURY"; } // Dealer + Online if (dealerRate && onlineRate) { return "MERCURY"; } // Forward + Online if (forwardRate && onlineRate) { return "MERCURY"; } // Dealer + Forward if (dealerRate && forwardRate) { return "MERCURY"; } // Dealer + Forward + Online if (dealerRate && forwardRate && onlineRate) { return "MERCURY"; } return "UNKNOWN"; }
