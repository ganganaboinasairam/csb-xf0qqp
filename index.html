function changeFieldClass(inputId, param) { // Get the input field by ID const inputField = document.getElementById(inputId); if (!inputField) { console.error(`No input field found with ID: ${inputId}`); return; } // Get the corresponding element const tdElements = document.querySelectorAll("td"); let relatedTd = null; tdElements.forEach(td => { if (td.innerText.trim() === inputField.title.trim()) { relatedTd = td; } }); if (!relatedTd) { console.error(`No matching element found for input field with title: ${inputField.title}`); return; } // Update the input field and related based on the param if (param === "M") { inputField.className = "CHAR_M"; relatedTd.className = "FldLabel star"; inputField.readOnly = false; // Ensure it is not readonly } else if (param === "O") { inputField.className = "CHAR_O"; relatedTd.className = "FldLabel"; inputField.readOnly = false; // Ensure it is not readonly } else if (param === "P") { inputField.className = "CHAR_P"; relatedTd.className = "FldLabel"; inputField.readOnly = true; // Set readonly } else { console.error(`Invalid parameter value: ${param}`); } }
