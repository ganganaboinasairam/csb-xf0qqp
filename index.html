function convertToJson(text) { const blocks = text .replace(/\r/g, "") .split("===================================") .map(b => b.trim()) .filter(b => b.length); return blocks.map(parseBlock); } function parseBlock(block) { const obj = { attributes: [], versionLabels: {} }; const lines = block.split("\n"); let section = ""; for (let line of lines) { line = line.trim(); if (!line || line === "Change History") continue; if (line.startsWith("Archive:")) { obj.archive = line.replace("Archive:", "").trim(); } else if (line.startsWith("Workfile:")) { obj.workfile = line.replace("Workfile:", "").trim(); } else if (line.startsWith("Archive created:")) { obj.archiveCreated = line.replace("Archive created:", "").trim(); } else if (line.startsWith("Split mode:")) { obj.splitMode = line.replace("Split mode:", "").trim(); } else if (line.startsWith("Owner:")) { obj.owner = line.replace("Owner:", "").trim(); } else if (line.startsWith("Last trunk rev:")) { obj.lastTrunkRevision = line.replace("Last trunk rev:", "").trim(); } else if (line.startsWith("Locks:")) { obj.locks = line.replace("Locks:", "").trim(); } else if (line.startsWith("Groups:")) { obj.groups = line.replace("Groups:", "").trim(); } else if (line.startsWith("Rev count:")) { obj.revisionCount = Number(line.replace("Rev count:", "").trim()); } else if (line === "Attributes:") { section = "attributes"; } else if (line === "Version labels:") { section = "labels"; } else if (line === "Description:") { section = "description"; obj.description = ""; } else if (section === "attributes") { if (line.includes("=")) { const [key, value] = line.split("="); obj[key.trim()] = value.trim().replace(/^"|"$/g, ""); } else { obj.attributes.push(line); } } else if (section === "labels") { const match = line.match(/"(.+?)"\s*=\s*(.+)/); if (match) { obj.versionLabels[match[1]] = match[2]; } } else if (section === "description") { obj.description += (obj.description ? "\n" : "") + line; } } return obj; }
