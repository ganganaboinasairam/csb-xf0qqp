async function openIframe(custId, branchCode, custName) { try { // 1) request short-lived chat token from main app var csrfObj = csrfResolution('../servlets/create-chat-token'); $.ajax({ url: csrfObj.url, method: 'POST', dataType: 'json', xhrFields: { withCredentials: true }, // same as fetch credentials: 'include' data: { CSRFTOKEN: csrfObj.csrf, REFRESH_TOKEN: csrfObj.refreshToken }, success: function (resp) { const chatToken = resp.chatToken; // 2) build a hidden form to POST into the chat app launch endpoint (token not in URL) const form = document.createElement('form'); form.method = 'POST'; form.target = 'iframeBox'; form.action = 'http://10.177.141.86:7001/ChatboxEECE/chat-init'; // helper for hidden fields function addField(name, value) { const i = document.createElement('input'); i.type = 'hidden'; i.name = name; i.value = value || ''; form.appendChild(i); } addField('chatToken', chatToken); addField('custId', custId); addField('branchCode', branchCode); addField('custName', custName); document.body.appendChild(form); // show iframe and submit $('#iframeBox').show(); form.submit(); // cleanup setTimeout(() => document.body.removeChild(form), 1000); }, error: function (xhr, status, err) { alert('Cannot open chat: ' + err); } }); } catch (e) { alert('Cannot open chat: ' + e.message); } }
