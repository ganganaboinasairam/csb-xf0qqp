@PostMapping("/updateMsgReadStatus") public ResponseEntity updateMsgReadStatus(@RequestBody UpdateMsgReadStatusRequest request) { Map response = new HashMap<>(); try { System.out.println("/////////////////////////////////////// enter updateMsgReadStatus //////////////////////////////"); ChatMessageManager.updateMsgReadStatus( request.getReqFrom(), request.getCustId(), request.getBranchCode() ); response.put("info", "Successfully updated msg status for " + request.getCustId()); System.out.println("Successfully updated msg status for " + request.getCustId()); return ResponseEntity.ok(response); } catch (Exception e) { System.out.println("/////////////////////////////////////// enter updateMsgReadStatus catch //////////////////////////////"); response.put("error", "Server error while loading messages"); return ResponseEntity.status(500).body(response); } }
