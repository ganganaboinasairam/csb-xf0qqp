public class GitCli { public static String run(File repo, String... args) throws Exception { System.out.println("in GitCli --> run with args " + Arrays.asList(args)); List command = new ArrayList<>(); command.add(Config.get("git.executable")); command.addAll(Arrays.asList(args)); ProcessBuilder pb = new ProcessBuilder(command); pb.directory(repo); pb.redirectErrorStream(true); Process p = pb.start(); StringBuilder output = new StringBuilder(); try (BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()))) { String line; while ((line = br.readLine()) != null) { output.append(line).append("\n"); } } int exit = p.waitFor(); if (exit != 0) { throw new RuntimeException( "Git command failed:\n" + String.join(" ", command) + "\n\n" + output.toString() ); } return output.toString(); } }
