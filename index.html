private String normalizeFormBody(HttpServletRequest req, byte[] bodyBytes) throws IOException { if (bodyBytes == null || bodyBytes.length == 0) { return ""; } String charsetName = req.getCharacterEncoding(); Charset charset = charsetName != null ? Charset.forName(charsetName) : StandardCharsets.UTF_8; // IMPORTANT: do NOT re-encode or reorder return new String(bodyBytes, charset).trim(); }
