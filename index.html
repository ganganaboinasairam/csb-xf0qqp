public void commitRebasePush(String crNo) throws Exception { long start = System.currentTimeMillis(); System.out.println("Opening repository..."); File repoDir = new File(Config.get("git.repo.path")); try (Git git = Git.open(repoDir)) { git.getRepository().getConfig() .setBoolean("core", null, "filemode", false); System.out.println("Checking status..."); Status status = git.status().call(); if (status.isClean()) { System.out.println("No changes to commit."); return; } System.out.println("Staging changed files..."); for (String f : status.getModified()) { git.add().addFilepattern(f).call(); } for (String f : status.getUntracked()) { git.add().addFilepattern(f).call(); } System.out.println("Generating commit message..."); String msg = CommitMessageGenerator.generate(git, crNo); System.out.println("Committing..."); git.commit() .setMessage(msg) .setAuthor( Config.get("git.author.name"), Config.get("git.author.email")) .call(); System.out.println("Committed: " + msg); System.out.println("Local commit time: " + (System.currentTimeMillis() - start) + " ms"); rebaseIfRequired(git); push(git); } }
