function out(text, cls) { const div = document.getElementById('output'); // Get existing logs const logs = JSON.parse(localStorage.getItem('consoleLogs') || '[]'); text.split('\n').forEach(line => { if (!line.trim() && text.split('\n').length > 1) return; const cssClass = cls ? 'log-' + cls : detectClass(line); const el = document.createElement('div'); el.className = cssClass; el.textContent = line; div.appendChild(el); // Save log entry logs.push({ text: line, className: cssClass }); }); localStorage.setItem('consoleLogs', JSON.stringify(logs)); div.scrollTop = div.scrollHeight; } // Load logs on page load function loadLogs() { const div = document.getElementById('output'); const logs = JSON.parse(localStorage.getItem('consoleLogs') || '[]'); logs.forEach(log => { const el = document.createElement('div'); el.className = log.className; el.textContent = log.text; div.appendChild(el); }); div.scrollTop = div.scrollHeight; } window.addEventListener('DOMContentLoaded', loadLogs); logs.push({ text: line, className: cssClass }); if (logs.length > 1000) { logs.shift(); // keep only latest 1000 lines }
