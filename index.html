private String buildFormBodyFromParameters(HttpServletRequest request) { return request.getParameterMap().entrySet().stream() .sorted(java.util.Map.Entry.comparingByKey()) // canonical order .map(e -> { String key = e.getKey(); String[] values = e.getValue(); return key + "=" + (values.length > 0 ? values[0] : ""); }) .reduce((a, b) -> a + "&" + b) .orElse(""); }
