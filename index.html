public static String runCustom(String region, String command) throws Exception { if (command == null || command.trim().isEmpty()) { throw new IllegalArgumentException("Command is empty"); } command = command.trim(); if (!command.startsWith("git ")) { throw new IllegalArgumentException( "Only git commands are allowed (must start with 'git ')" ); } String lower = command.toLowerCase(); // ============================================================ // BLOCK DANGEROUS / DESTRUCTIVE COMMANDS // ============================================================ String[] blocked = { // destructive reset "git reset --hard", "git clean -fd", "git clean -f -d", "git clean -xdf", // branch deletion "git branch -d", "git branch -D", // force push "git push --force", "git push -f", // history rewrite "git filter-branch", "git reflog expire", // checkout overwrite "git checkout .", "git restore .", // invalid / dangerous pull "git pull --hard" }; for (String bad : blocked) { if (lower.startsWith(bad)) { throw new IllegalArgumentException( "Blocked fragile command: " + bad ); } } // ============================================================ // AUTO-LIMIT GIT LOG // ============================================================ // user typed only: git log if (lower.equals("git log")) { command = "git log --oneline -20"; } // user typed: git log --oneline else if (lower.equals("git log --oneline")) { command = "git log --oneline -20"; } // user typed git log but forgot count else if (lower.startsWith("git log")) { boolean hasCount = lower.matches(".*\\s-\\d+.*") || lower.contains("--max-count"); if (!hasCount) { command += " --oneline -20"; } } String gitExe = Config.get("git.executable"); File repo = new File(Config.get("git.repo.path." + region)); GitHelper.validateGitIdentity(repo); List cmd = new ArrayList<>(); cmd.add(gitExe); for (String part : command.substring(4).split("\\s+")) { if (!part.isEmpty()) { cmd.add(part.trim()); } } ProcessBuilder pb = new ProcessBuilder(cmd); pb.directory(repo); pb.redirectErrorStream(true); Process p = pb.start(); StringBuilder out = new StringBuilder(); try (BufferedReader br = new BufferedReader( new InputStreamReader(p.getInputStream()))) { String line; while ((line = br.readLine()) != null) { out.append(line).append("\n"); } } int exit = p.waitFor(); out.append("\n[INFO] Exit code: ").append(exit); return out.toString(); }
