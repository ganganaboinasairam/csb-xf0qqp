private void push(Git git) throws Exception { String remote = Config.get("git.remote.name"); // origin String branch = Config.get("git.branch.name"); // main / develop System.out.println("Pushing to " + remote + "/" + branch); Iterable results = git.push() .setRemote(remote) .setRefSpecs( new RefSpec("refs/heads/" + branch + ":refs/heads/" + branch)) .setCredentialsProvider(credentials()) .call(); boolean pushed = false; for (PushResult result : results) { for (RemoteRefUpdate update : result.getRemoteUpdates()) { RemoteRefUpdate.Status status = update.getStatus(); System.out.println("Push status: " + status); if (status == RemoteRefUpdate.Status.OK || status == RemoteRefUpdate.Status.UP_TO_DATE) { pushed = true; } else { throw new RuntimeException( "Push failed: " + status + " - " + update.getMessage()); } } } if (!pushed) { throw new RuntimeException( "Push reported success but no refs were updated"); } System.out.println("Push completed successfully."); }
