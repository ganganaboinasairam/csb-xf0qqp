public class Main { public static void main(String[] args) { String testContent = "1000.50USD"; String[] result = getForeignAmountAndCurrency(testContent); System.out.println("Foreign Amount: " + result[0]); // Output: 1000.50 System.out.println("Foreign Currency: " + result[1]); // Output: USD } private static String[] getForeignAmountAndCurrency(String t_Content) { String foreignAmount = ""; String foreignCurrency = ""; if (t_Content == null || t_Content.isEmpty()) { return new String[]{foreignAmount, foreignCurrency}; } try { // Extract FORN_AMT value String[] amtArray = t_Content.split(""); if (amtArray.length > 1) { String[] amtSplitted = amtArray[1].split(""); if (amtSplitted.length > 0) { foreignAmount = amtSplitted[0]; } } // Extract FRGN_CCY value String[] ccyArray = t_Content.split(""); if (ccyArray.length > 1) { String[] ccySplitted = ccyArray[1].split(""); if (ccySplitted.length > 0) { foreignCurrency = ccySplitted[0]; } } } catch (Exception e) { e.printStackTrace(); } return new String[]{foreignAmount, foreignCurrency}; } }
