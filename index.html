package com.chat; import java.io.IOException; import java.io.PrintWriter; import javax.servlet.ServletException; import javax.servlet.annotation.WebServlet; import javax.servlet.http.HttpServlet; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletResponse; import com.google.gson.JsonObject; @WebServlet("/getJwtToken") public class JwtTokenServlet extends HttpServlet { @Override protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException { // For simplicity, assume authentication is done elsewhere, and we generate token based on customerId // In production, validate credentials here String customerId = request.getParameter("customerId"); String branchCode = request.getParameter("branchCode"); if (customerId == null || branchCode == null) { response.setStatus(HttpServletResponse.SC_BAD_REQUEST); response.getWriter().write("{\"error\":\"Missing customerId or branchCode\"}"); return; } // Generate JWT token with subject as customerId-branchCode String subject = customerId + "-" + branchCode; String token = JwtUtil.generateToken(subject); // Return token as JSON response.setContentType("application/json"); PrintWriter out = response.getWriter(); JsonObject json = new JsonObject(); json.addProperty("token", token); out.print(json.toString()); out.flush(); } } package com.chat; import io.jsonwebtoken.Claims; import io.jsonwebtoken.Jwts; import io.jsonwebtoken.SignatureAlgorithm; import javax.crypto.spec.SecretKeySpec; import javax.crypto.SecretKey; import java.nio.charset.StandardCharsets; import java.util.Date; public class JwtUtil { // IMPORTANT: Replace this with a secure key and load it from env/config in production. // HS256 requires a 256-bit (32 byte) key. Below is an example 32+ byte string. private static final String SECRET_STRING = "ReplaceThisWithASecureRandom32ByteKey!!!!"; private static final byte[] SECRET_BYTES = SECRET_STRING.getBytes(StandardCharsets.UTF_8); // Create SecretKeySpec for HS256 private static final SecretKey SECRET_KEY = new SecretKeySpec(SECRET_BYTES, SignatureAlgorithm.HS256.getJcaName()); // Token validity in milliseconds (e.g., 1 hour) private static final long TOKEN_VALIDITY = 3600_000L; // 1 hour // Generate JWT token for a given subject (e.g., customerId) public static String generateToken(String subject) { long nowMillis = System.currentTimeMillis(); Date now = new Date(nowMillis); Date expiryDate = new Date(nowMillis + TOKEN_VALIDITY); return Jwts.builder() .setSubject(subject) .setIssuedAt(now) .setExpiration(expiryDate) // In jjwt 0.9.1, use signWith(SignatureAlgorithm, key) .signWith(SignatureAlgorithm.HS256, SECRET_KEY) .compact(); } // Validate JWT token and return claims if valid, else null public static Claims validateToken(String token) { try { // In jjwt 0.9.1 use Jwts.parser() return Jwts.parser() .setSigningKey(SECRET_KEY) .parseClaimsJws(token) .getBody(); } catch (Exception e) { // logging only; in production use a proper logger System.err.println("Invalid JWT token: " + e.getMessage()); return null; } } }
