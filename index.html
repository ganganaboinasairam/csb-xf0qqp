private static final Pattern TOOL_COMMIT_PATTERN = Pattern.compile("^(CR|IR)_\\d+_\\d{6}_(\\d+)$"); public static String generate(File repo, String crNo) throws Exception { String date = new SimpleDateFormat("ddMMyy").format(new Date()); int maxSeq = 0; String log = GitCli.run( repo, "log", "--pretty=format:%s", "--grep=" + crNo + "_" + date ); for (String line : log.split("\\R")) { Matcher m = TOOL_COMMIT_PATTERN.matcher(line.trim()); if (!m.matches()) { // 🔒 Ignore user-edited or custom messages continue; } int seq = Integer.parseInt(m.group(2)); maxSeq = Math.max(maxSeq, seq); } return crNo + "_" + date + "_" + (maxSeq + 1); }
