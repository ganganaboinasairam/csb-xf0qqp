package com.company.analyzer.search; import com.company.analyzer.model.LogBlock; import com.company.analyzer.source.LogSource; import java.util.ArrayList; import java.util.List; import java.util.regex.Matcher; import java.util.regex.Pattern; public class LogSearchService { private static final Pattern HEADER_PATTERN = Pattern.compile( "\\[start\\]\\s+" + "\\[([^\\]]+)\\]\\s+" + "\\[([^\\]]+)\\]\\s+" + "\\[([^\\]]+)\\]" ); public List findErrorByCMainRef( LogSource source, String cMainRef) throws Exception { List results = new ArrayList(); source.scan((sourceName, content) -> { Matcher matcher = HEADER_PATTERN.matcher(content); if (!matcher.find()) { return; } String level = matcher.group(1); // Only ERROR logs are relevant if (!"ERROR".equalsIgnoreCase(level)) { return; } // Then check C_MAIN_REF if (!content.contains(cMainRef)) { return; } String timestamp = matcher.group(2); String correlationId = matcher.group(3); LogBlock block = new LogBlock( sourceName, level, timestamp, correlationId, content ); results.add(block); }); return results; } }
