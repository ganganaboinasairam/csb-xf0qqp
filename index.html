private static void generateExcel(File[] files, String outputPath) throws Exception { Workbook workbook = new XSSFWorkbook(); Sheet sheet = workbook.createSheet("Evidence"); Drawing drawing = sheet.createDrawingPatriarch(); CreationHelper helper = workbook.getCreationHelper(); // layout config int imageWidthPx = 500; // adjust based on your screenshots int imageHeightPx = 300; int gapX = 40; // horizontal spacing int gapY = 40; // vertical spacing int maxPerRow = 2; // desktop screenshots → 2 per row int currentX = 0; int currentY = 0; int countInRow = 0; for (File file : files) { byte[] bytes = Files.readAllBytes(file.toPath()); int pictureIdx = workbook.addPicture(bytes, Workbook.PICTURE_TYPE_PNG); ClientAnchor anchor = helper.createClientAnchor(); // Always anchor to (0,0) but move using offsets anchor.setCol1(0); anchor.setRow1(0); // Convert pixels → EMU (1 px = 9525 EMU) int dx1 = currentX * 9525; int dy1 = currentY * 9525; int dx2 = (currentX + imageWidthPx) * 9525; int dy2 = (currentY + imageHeightPx) * 9525; anchor.setDx1(dx1); anchor.setDy1(dy1); anchor.setDx2(dx2); anchor.setDy2(dy2); drawing.createPicture(anchor, pictureIdx); // move horizontally currentX += imageWidthPx + gapX; countInRow++; // wrap to next row if (countInRow == maxPerRow) { currentX = 0; currentY += imageHeightPx + gapY; countInRow = 0; } } FileOutputStream fos = new FileOutputStream(outputPath); workbook.write(fos); fos.close(); }
