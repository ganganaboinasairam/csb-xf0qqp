useEffect(() => { loadPreviousMessages(); }, []); const loadPreviousMessages = async () => { try { const branchCode = getUrlParameter('branch_code') || '00000'; const response = await fetch(`./LoadPreviousMesgs?cust_id=00000&branch_code=${branchCode}`); if (!response.ok) throw new Error('Failed to fetch previous messages'); let data = await response.json(); // Group messages by custID1 const customerMap = {}; data.forEach(msg => { const custId = msg.custID1; if (!customerMap[custId]) { customerMap[custId] = { id: custId, name: `Customer ${custId}`, // Change if name is available lastMessage: '', lastMessageTime: '', status: "online", unreadCount: 0, messages: [] }; } customerMap[custId].messages.push({ content: msg.message, isCustomer: msg.sender.toLowerCase() === 'customer', timestamp: msg.date }); }); // Sort messages inside each customer Object.values(customerMap).forEach(customer => { customer.messages.sort((a, b) => new Date(a.timestamp) - new Date(b.timestamp)); // Set last message details const lastMsg = customer.messages[customer.messages.length - 1]; customer.lastMessage = lastMsg.content; customer.lastMessageTime = new Date(lastMsg.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }); }); // Convert to array and sort by last message time (desc) const customerList = Object.values(customerMap).sort((a, b) => { const timeA = new Date(a.messages[a.messages.length - 1].timestamp); const timeB = new Date(b.messages[b.messages.length - 1].timestamp); return timeB - timeA; }); // Update state setCustomers(customerList); // Optionally select the most recent customer automatically if (customerList.length > 0) { setSelectedCustomer(customerList[0]); } } catch (error) { console.error('Error loading previous messages:', error); } };
