function emailidcheck(email) { let pattern = /^[a-zA-Z0-9_.+-]+@[a-zA-Z0-9-]+\.[a-zA-Z]{2,}$/; return pattern.test(email); } // Example usage: console.log(emailidcheck("test@example.com")); // true console.log(emailidcheck("user.name@domain.co.in")); // true console.log(emailidcheck("sddsf@gmail.gmail.com.com")); // false console.log(emailidcheck("invalid-email.com")); // false console.log(emailidcheck("test@domain")); // false
