package com.chat.helper; import java.io.ByteArrayInputStream; import javax.servlet.ReadListener; import javax.servlet.ServletInputStream; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletRequestWrapper; public class CachedBodyHttpServletRequest extends HttpServletRequestWrapper { private final byte[] cachedBody; public CachedBodyHttpServletRequest(HttpServletRequest request, byte[] body) { super(request); this.cachedBody = body == null ? new byte[0] : body; } @Override public ServletInputStream getInputStream() { final ByteArrayInputStream bais = new ByteArrayInputStream(this.cachedBody); return new ServletInputStream() { @Override public int read() { return bais.read(); } @Override public boolean isFinished() { return bais.available() == 0; } @Override public boolean isReady() { return true; } @Override public void setReadListener(ReadListener readListener) { // no-op for blocking IO } }; } @Override public int getContentLength() { return this.cachedBody.length; } @Override public long getContentLengthLong() { return this.cachedBody.length; } // If someone calls getReader(), the container will wrap the InputStream -> Reader, // so nothing else needed here (default wrapper behavior is fine). }
