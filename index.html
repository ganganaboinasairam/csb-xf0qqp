private static void generateExcel(File[] files, String outputPath) throws Exception { Workbook workbook = new XSSFWorkbook(); Sheet sheet = workbook.createSheet("Evidence"); CreationHelper helper = workbook.getCreationHelper(); Drawing drawing = sheet.createDrawingPatriarch(); int imagesPerRow = 2; // Set column width (wide for desktop screenshots) for (int i = 0; i < imagesPerRow; i++) { sheet.setColumnWidth(i, 80 * 256); } int rowIndex = 0; int colIndex = 0; Row row = sheet.createRow(rowIndex); row.setHeightInPoints(300); // moderate height for (int i = 0; i < files.length; i++) { File file = files[i]; // KEEP ORIGINAL QUALITY byte[] imageBytes = Files.readAllBytes(file.toPath()); int pictureIdx = workbook.addPicture(imageBytes, Workbook.PICTURE_TYPE_PNG); ClientAnchor anchor = helper.createClientAnchor(); anchor.setCol1(colIndex); anchor.setRow1(rowIndex); Picture pict = drawing.createPicture(anchor, pictureIdx); // Controlled resize (not aggressive) pict.resize(0.5); // tweak: 0.4 to 0.7 depending on size colIndex++; if (colIndex == imagesPerRow) { colIndex = 0; rowIndex++; row = sheet.createRow(rowIndex); row.setHeightInPoints(300); } } FileOutputStream fos = new FileOutputStream(outputPath); workbook.write(fos); fos.close(); }
