Commit " + commitMsg + " pushed successfully."); } public static void chkStatus( String region, java.util.function.Consumer log) throws Exception { File repo = new File( Config.get("git.repo.path." + region)); log.accept("Repository: " + repo.getAbsolutePath()); String status = GitCli.run(repo, "status", "--porcelain"); if (status.trim().isEmpty()) { log.accept("no changes found, working tree clean"); return;
