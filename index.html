public static void retrieveFiles( String region, String crOrIr, java.util.function.Consumer log, java.util.function.IntConsumer progress) throws Exception { File repo = new File(Config.get("git.repo.path." + region)); log.accept("Repository : " + repo.getAbsolutePath()); log.accept("Searching commits for : " + crOrIr); log.accept(""); final int BATCH_SIZE = 50; int offset = 0; int batchNo = 0; Set files = new TreeSet<>(); while (true) { batchNo++; log.accept("Scanning batch " + batchNo + " (commits " + offset + " - " + (offset + BATCH_SIZE) + ")"); String out = GitCli.run( repo, "log", "--skip=" + offset, "--max-count=" + BATCH_SIZE, "--name-only", "--pretty=format:", "--grep=" + crOrIr ); // If nothing returned, check if history ended if (out.trim().isEmpty()) { String hasMore = GitCli.run( repo, "log", "--skip=" + offset, "--max-count=1", "--pretty=oneline" ); if (hasMore.trim().isEmpty()) { break; // end of history } offset += BATCH_SIZE; progress.accept(Math.min(90, batchNo * 10)); continue; } // Parse files for (String line : out.split("\\R")) { if (!line.trim().isEmpty()) { files.add(line.trim()); } } break; // found matches, stop scanning } if (files.isEmpty()) { log.accept("No files found for " + crOrIr); progress.accept(100); return; } progress.accept(100); log.accept(""); log.accept("Files committed under " + crOrIr + ":"); log.accept("----------------------------------------------"); for (String f : files) { log.accept(f); } } SwingWorker worker = new SwingWorker() { @Override protected Void doInBackground() throws Exception { GitAutomationService.retrieveFiles( region, crOrIr, msg -> publish(msg), percent -> setProgress(percent) ); return null; } @Override protected void process(java.util.List chunks) { for (String s : chunks) { outputArea.append(s + "\n"); } } @Override protected void done() { progressBar.setValue(100); statusLabel.setText("Completed"); outputArea.append("\n========== DONE ==========\n"); } }; worker.addPropertyChangeListener(evt -> { if ("progress".equals(evt.getPropertyName())) { progressBar.setValue((Integer) evt.getNewValue()); } }); worker.execute();
