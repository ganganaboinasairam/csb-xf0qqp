private static void validateGitIdentity(File repo) { try { String name = GitCli.run(repo, "config", "user.name"); String email = GitCli.run(repo, "config", "user.email"); if (name.trim().isEmpty() || email.trim().isEmpty()) { throw new RuntimeException( "Git user identity not configured.\n" + "Please configure:\n" + "git config --global user.name \"Your Name\"\n" + "git config --global user.email \"you@company.com\"" ); } } catch (Exception e) { throw new RuntimeException( "Git user identity validation failed", e); } }
