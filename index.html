@PostMapping( value = "sendmessage", consumes = MediaType.APPLICATION_FORM_URLENCODED_VALUE ) public ResponseEntity addMessageToDb(@ModelAttribute AddMessageToDb request) { Map response = new HashMap<>(); try { JSONObject obj = new JSONObject(); obj.put("customerId", request.getCustomerId()); obj.put("branchCode", request.getBranchCode()); obj.put("sender", request.getSender()); obj.put("customerName", request.getCustomerName()); obj.put("type", request.getType()); obj.put("role", request.getRole()); obj.put("msgSrc", request.getMsgSrc()); obj.put("timestamp", request.getTimestamp()); obj.put("content", request.getContent()); System.out.println("obj is ::::::::::::::::::::::::::::: " + obj); ChatMessageManager.saveMessagesToDB(obj); response.put("info", "Successfully added msg to db"); return ResponseEntity.ok(response); } catch (Exception e) { response.put("error", "Server error while adding messages: " + e.getMessage()); return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(response); } }
