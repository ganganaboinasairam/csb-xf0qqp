import org.w3c.dom.*; import javax.xml.parsers.*; import java.io.File; import java.util.*; public class ProductModuleExtractor { public static void main(String[] args) { try { // Sample product IDs to search String[] productIds = {"P05101000006", "P08080400000", "P05122400001"}; // Parse XML file File xmlFile = new File("products.xml"); // <-- put your XML file path DocumentBuilderFactory dbFactory = DocumentBuilderFactory.newInstance(); DocumentBuilder dBuilder = dbFactory.newDocumentBuilder(); Document doc = dBuilder.parse(xmlFile); doc.getDocumentElement().normalize(); // Get all elements NodeList productList = doc.getElementsByTagName("product"); // Convert to Set for faster lookup Set searchIds = new HashSet<>(Arrays.asList(productIds)); List moduleList = new ArrayList<>(); for (int i = 0; i < productList.getLength(); i++) { Node productNode = productList.item(i); if (productNode.getNodeType() == Node.ELEMENT_NODE) { Element productElement = (Element) productNode; // Extract value String id = productElement.getElementsByTagName("id").item(0).getTextContent(); // If this product's ID matches input array, get if (searchIds.contains(id)) { String module = productElement.getElementsByTagName("module").item(0).getTextContent(); moduleList.add(module); } } } // Convert to array String[] modulesArray = moduleList.toArray(new String[0]); // Print result System.out.println("Modules found: " + Arrays.toString(modulesArray)); } catch (Exception e) { e.printStackTrace(); } } }
