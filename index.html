function parseIRDetails(html) { const records = []; const textareaRegex = /]*>([\s\S]*?)<\/textarea>/g; const irNumberRegex = /IR Number:\s*(\d+)/g; const statusRegex = /STATUS:\s*([A-Z\s]+?)<\/p>/g; const textareas = [...html.matchAll(textareaRegex)].map(m => m[1]); const irNumbers = [...html.matchAll(irNumberRegex)].map(m => m[1]); const statuses = [...html.matchAll(statusRegex)].map(m => m[1].trim()); textareas.forEach((raw, i) => { const refMatch = raw.match(/(TF[A-Z]{2}-\d{8}-\d+)/); const descMatch = raw.match(/Desc:([\s\S]*?)\/UAT-HASH/); records.push({ irNumber: irNumbers[i] || null, reference: refMatch ? refMatch[1] : null, status: statuses[i] || null, description: descMatch ? descMatch[1].trim().replace(/\s+/g, ' ') : raw.trim().replace(/\s+/g, ' ') }); }); return records; } async function fetchAndRenderIRTable(url, targetElementId) { try { const res = await fetch(url); const html = await res.text(); const records = parseIRDetails(html); renderTable(records, targetElementId); return records; } catch (err) { console.error('Fetch failed:', err); document.getElementById(targetElementId).innerHTML = `

Error fetching data: ${err.message}

`; } } function renderTable(records, targetElementId) { const container = document.getElementById(targetElementId); let html = ``; records.forEach(r => { html += ``; }); html += `

${r.irNumber ?? ''} ${r.reference ?? ''} ${r.status ?? ''} ${r.description ?? ''}
`; container.innerHTML = html; }
