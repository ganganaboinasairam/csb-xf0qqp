private String lastFindKeyword = ""; private int lastFindIndex = -1; private void setupCtrlF() { KeyStroke ctrlF = KeyStroke.getKeyStroke( "control F" ); textPane.getInputMap( JComponent.WHEN_IN_FOCUSED_WINDOW ).put( ctrlF, "find" ); textPane.getActionMap().put( "find", new AbstractAction() { @Override public void actionPerformed( java.awt.event.ActionEvent e ) { String keyword = JOptionPane.showInputDialog( frame, "Find Text" ); if (keyword == null || keyword.trim().isEmpty()) { return; } lastFindKeyword = keyword; lastFindIndex = -1; findNextOccurrence(); } } ); KeyStroke enter = KeyStroke.getKeyStroke( "ENTER" ); textPane.getInputMap( JComponent.WHEN_IN_FOCUSED_WINDOW ).put( enter, "findNext" ); textPane.getActionMap().put( "findNext", new AbstractAction() { @Override public void actionPerformed( java.awt.event.ActionEvent e ) { if (lastFindKeyword == null || lastFindKeyword.isEmpty()) { return; } findNextOccurrence(); } } ); } private void findNextOccurrence() { try { Highlighter highlighter = textPane.getHighlighter(); highlighter.removeAllHighlights(); String text = textPane.getDocument().getText( 0, textPane.getDocument().getLength() ); int startIndex = lastFindIndex + 1; int found = text.indexOf( lastFindKeyword, startIndex ); if (found < 0) { found = text.indexOf( lastFindKeyword ); if (found < 0) { JOptionPane.showMessageDialog( frame, "Text not found" ); return; } } int end = found + lastFindKeyword.length(); highlighter.addHighlight( found, end, new DefaultHighlighter.DefaultHighlightPainter( Color.CYAN ) ); textPane.setCaretPosition(end); Rectangle view = textPane.modelToView(found); if (view != null) { textPane.scrollRectToVisible(view); } lastFindIndex = found; } catch (Exception ex) { ex.printStackTrace(); } }
