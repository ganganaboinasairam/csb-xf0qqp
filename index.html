package com.tcs.git; import java.io.File; import java.util.regex.Matcher; import java.util.regex.Pattern; import com.tcs.git.config.Config; public class CommitMessageGenerator { private static final Pattern TOOL_COMMIT_PATTERN = Pattern.compile("^(CR|IR)_\\d+_[vV]\\d+_(\\d+)$"); public static String generate(File repo, String crNo ,String region) throws Exception { int maxSeq = 0; String branch = Config.get("git.branch.name."+region); String user = Config.get("jenkins.user").trim(); String log = GitCli.run( repo, "log", "origin/"+branch, "--pretty=format:%s", "--grep=" + crNo + "_" + user ); for (String line : log.split("\\R")) { Matcher m = TOOL_COMMIT_PATTERN.matcher(line.trim()); if (!m.matches()) { // Ignore user-edited or custom messages continue; } int seq = Integer.parseInt(m.group(2)); maxSeq = Math.max(maxSeq, seq); } return "CR_" + crNo + "_" + user + "_" + (maxSeq + 1); } } the commit message can be like this also "CR_EFOREX_v1013325" adjust the pattern
