const fetchButtons = [ "FETCH_RATE_BTN_1", "FETCH_RATE_BTN_2", "FETCH_RATE_BTN_3", "FETCH_RATE_BTN_4" ]; function createFetchLoaders() { fetchButtons.forEach(function(btnId) { const btn = document.getElementById(btnId); if (!btn) return; // Prevent duplicate loader creation if (document.getElementById(btnId + "_loader")) return; const loader = document.createElement("img"); loader.id = btnId + "_loader"; loader.src = "../Images/sbk.gif"; loader.style.width = "16px"; loader.style.height = "16px"; loader.style.marginLeft = "5px"; loader.style.verticalAlign = "middle"; loader.style.display = "none"; // Initially hidden btn.insertAdjacentElement("afterend", loader); }); }
