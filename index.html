package com.tcs.git; import java.io.FileInputStream; import java.util.Properties; public class Config { private static final Properties props = new Properties(); static { try (FileInputStream fis = new FileInputStream("application.properties")) { props.load(fis); } catch (Exception e) { throw new RuntimeException( "Unable to load application.properties", e); } } public static String get(String key) { return props.getProperty(key); } public static int getInt(String key) { return Integer.parseInt(props.getProperty(key)); } } package com.tcs.git; import java.io.*; import java.util.*; public class GitCli { public static String run(File repo, String... args) throws Exception { List command = new ArrayList<>(); command.add(Config.get("git.executable")); command.addAll(Arrays.asList(args)); ProcessBuilder pb = new ProcessBuilder(command); pb.directory(repo); pb.redirectErrorStream(true); Process p = pb.start(); StringBuilder output = new StringBuilder(); try (BufferedReader br = new BufferedReader( new InputStreamReader(p.getInputStream()))) { String line; while ((line = br.readLine()) != null) { output.append(line).append("\n"); } } int exit = p.waitFor(); if (exit != 0) { throw new RuntimeException( "Git failed: " + String.join(" ", args)); } return output.toString(); } } package com.tcs.git; import java.io.File; import java.time.LocalDate; import java.time.format.DateTimeFormatter; public class CommitMessageGenerator { private static final DateTimeFormatter DF = DateTimeFormatter.ofPattern("ddMMyy"); public static String generate(File repo, String crNo) throws Exception { String today = LocalDate.now().format(DF); int maxSeq = 0; String log = GitCli.run( repo, "log", "--pretty=%s", "-n", String.valueOf(Config.getInt("git.log.scan.limit")) ); for (String line : log.split("\n")) { if (line.startsWith("CR_" + crNo + "_" + today + "_")) { String[] parts = line.split("_"); int seq = Integer.parseInt(parts[3]); maxSeq = Math.max(maxSeq, seq); } } return "CR_" + crNo + "_" + today + "_" + (maxSeq + 1); } } package com.tcs.git; import java.io.File; public class GitAutomationService { public static void execute( String region, String crNo, java.util.function.Consumer log) throws Exception { File repo = new File( Config.get("git.repo.path." + region)); String branch = Config.get("git.branch.name." + region); log.accept("Repository: " + repo.getAbsolutePath()); // Pull first log.accept("Pulling with rebase..."); GitCli.run(repo, "pull", "--rebase", "origin", branch); // Check changes String status = GitCli.run(repo, "status", "--porcelain"); if (status.trim().isEmpty()) { log.accept("No changes to commit."); return; } // Generate commit message String commitMsg = CommitMessageGenerator.generate(repo, crNo); log.accept("Staging changes..."); GitCli.run(repo, "add", "-A"); log.accept("Committing: " + commitMsg); GitCli.run(repo, "commit", "-m", commitMsg); log.accept("Pushing to origin/" + branch); GitCli.run(repo, "push", "origin", branch); log.accept("DONE. Commit " + commitMsg + " pushed successfully."); } } package com.tcs.git; import javax.swing.*; import java.awt.*; public class MainUI extends JFrame { private JTextField crField; private JComboBox regionBox; private JTextArea outputArea; private JButton pushButton; public MainUI() { setTitle("Git Commit & Push Tool"); setSize(750, 500); setDefaultCloseOperation(EXIT_ON_CLOSE); setLocationRelativeTo(null); initUI(); } private void initUI() { JPanel top = new JPanel(new GridBagLayout()); GridBagConstraints c = new GridBagConstraints(); c.insets = new Insets(5,5,5,5); c.gridx = 0; c.gridy = 0; top.add(new JLabel("Region:"), c); c.gridx = 1; regionBox = new JComboBox<>(new String[]{"EE", "CE"}); top.add(regionBox, c); c.gridx = 0; c.gridy = 1; top.add(new JLabel("CR Number:"), c); c.gridx = 1; crField = new JTextField(15); top.add(crField, c); c.gridx = 1; c.gridy = 2; pushButton = new JButton("Commit & Push"); top.add(pushButton, c); outputArea = new JTextArea(); outputArea.setEditable(false); outputArea.setFont(new Font("Consolas", Font.PLAIN, 12)); JScrollPane scroll = new JScrollPane(outputArea, JScrollPane.VERTICAL_SCROLLBAR_ALWAYS, JScrollPane.HORIZONTAL_SCROLLBAR_AS_NEEDED); add(top, BorderLayout.NORTH); add(scroll, BorderLayout.CENTER); pushButton.addActionListener(e -> runTask()); } private void runTask() { String crNo = crField.getText().trim(); String region = regionBox.getSelectedItem().toString().toLowerCase(); if (!crNo.matches("\\d+")) { JOptionPane.showMessageDialog( this, "Invalid CR Number", "Error", JOptionPane.ERROR_MESSAGE); return; } outputArea.setText(""); pushButton.setEnabled(false); SwingWorker worker = new SwingWorker<>() { @Override protected Void doInBackground() throws Exception { GitAutomationService.execute( region, crNo, msg -> publish(msg)); return null; } @Override protected void process(java.util.List chunks) { for (String s : chunks) { outputArea.append(s + "\n"); } } @Override protected void done() { pushButton.setEnabled(true); try { get(); } catch (Exception ex) { outputArea.append( "\nERROR: " + ex.getMessage()); } } }; worker.execute(); } public static void main(String[] args) { SwingUtilities.invokeLater( () -> new MainUI().setVisible(true)); } }
