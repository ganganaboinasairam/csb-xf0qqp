function changeFieldClassNew(inputId, param) { var inputField = document.getElementById(inputId); if (!inputField) { console.error(`No input field found with ID: ${inputId}`); return; } // Remove all classes that start with Char_ or date_ const toRemove = Array.from(inputField.classList).filter(cls => cls.startsWith('Char_') || cls.startsWith('date_') ); toRemove.forEach(cls => inputField.classList.remove(cls)); var fieldType = inputField.className.split('_')[0]; var parentTd = inputField.parentElement; if (!parentTd) { console.error(`No parent found for input field with ID: ${inputId}`); return; } if (['select-wrapper', 'date-wrapper', 'textarea-wrapper'].includes(parentTd.className)) { parentTd = inputField.parentElement.parentElement.firstChild; } else { parentTd = inputField.parentElement.firstChild; } var previousTd = parentTd; if (!previousTd) { console.error(`No previous found for input field with ID: ${inputId}`); return; } if (param === "M") { const newClass = fieldType + "_M"; inputField.classList.add(newClass); if (fieldType === 'DATE') { inputField.classList.add("hasDatepicker", "air-datepicker-initialized"); addDatePicker(inputField); } previousTd.className = "FldLabel star"; inputField.readOnly = false; inputField.removeAttribute('disabled'); } else if (param === "O") { inputField.classList.add(fieldType + "_O"); previousTd.className = "FldLabel"; inputField.readOnly = false; inputField.removeAttribute('disabled'); } else if (param === "P") { inputField.classList.add(fieldType + "_P"); previousTd.className = "FldLabel"; inputField.readOnly = true; inputField.setAttribute('disabled', 'true'); } else { console.error(`Invalid parameter value: ${param}`); } }
