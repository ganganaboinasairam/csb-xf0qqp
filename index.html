''; customer.lastMessageTime = lastMsg ? formatDate(lastMsg.timestamp) : ''; }); const customerList = Object.values(customerMap).sort((a, b) => { const timeA = new Date(a.messages[a.messages.length - 1]?.timestamp || 0); const timeB = new Date(b.messages[b.messages.length - 1]?.timestamp || 0); return timeB - timeA; }); setCustomers(customerList); const allTimestamps = customerList.flatMap(c => c.messages.map(m => m.timestamp)); const latestTimestamp = allTimestamps.length > 0 ? allTimestamps.sort().pop() : null; setLastTimestamp(latestTimestamp); if (customerList.length > 0) setSelectedCustomer(customerList[0]); // NEW: fetch unread counts for all (authoritative) refreshUnreadCountsForAll(customerList); } catch (error) { console.error('Error loading previous messages:', error); } };
