SELECT inlc.C_MAIN_REF AS LC_Reference, inad.C_MAIN_REF AS Advising_Ref, inlc.LC_AMT AS LC_Amount_INLC_MASTER, emiss.LC_AMT AS LC_Amount_EVENT, SUM(DISTINCT emamd.INC_AMT) AS Increment_LC_Amount, SUM(DISTINCT emamd.DEC_AMT) AS Decrement_LC_Amount, inlc.AMD_NO AS Amendment_Number, inlc.LC_BAL_PRSNT AS Presentation_Balance_INLC, SUM(DISTINCT iine.TTL_NEGO_AMT) AS Total_Negotiated_Amount, inad.PRES_BAL_LC AS Presentation_Balance_INAD FROM INAD_MASTER inad FULL JOIN INLC_MASTER inlc ON inad.LC_NO = inlc.C_MAIN_REF FULL JOIN IINE_MASTER iine ON inad.LC_NO = iine.LC_NO FULL JOIN INLC_EM_AMENDMENT emamd ON inlc.C_MAIN_REF = emamd.C_MAIN_REF AND emamd.C_FUNC_SHORT_NAME IN ('RailLCAmedDEO', 'eTradeSBIAmendLC') AND emamd.C_TRX_STATUS = 'M' FULL JOIN INLC_EM_ISSUANCE emiss ON inlc.C_MAIN_REF = emiss.C_MAIN_REF AND emiss.I_EVENT_TIMES = '1' WHERE inad.LC_NO IN ('0069121LC0000035') GROUP BY inlc.C_MAIN_REF, inad.C_MAIN_REF, inad.PRES_BAL_LC, inlc.LC_AMT, emiss.LC_AMT, inlc.LC_BAL_PRSNT, inlc.AMD_NO;
