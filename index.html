private void setupCtrlF() { KeyStroke keyStroke = KeyStroke.getKeyStroke( "control F" ); textPane.getInputMap( JComponent.WHEN_IN_FOCUSED_WINDOW ).put( keyStroke, "find" ); textPane.getActionMap().put( "find", new AbstractAction() { @Override public void actionPerformed( java.awt.event.ActionEvent e ) { String keyword = JOptionPane.showInputDialog( frame, "Find Text" ); if (keyword == null || keyword.trim().isEmpty()) { return; } highlightFind(keyword); } } ); } private void highlightFind(String keyword) { String text = textPane.getText(); int index = text.indexOf(keyword); if (index < 0) { JOptionPane.showMessageDialog( frame, "Text not found" ); return; } textPane.requestFocus(); textPane.select( index, index + keyword.length() ); }
