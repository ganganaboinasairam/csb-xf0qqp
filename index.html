function processLegalCompliance() { if (!SYS_MULTI_DATA || SYS_MULTI_DATA.length === 0) return false; // Helper to extract column function getColumn(colName) { for (var i = 0; i < SYS_MULTI_DATA.length; i++) { if (SYS_MULTI_DATA[i][0] === colName) { return SYS_MULTI_DATA[i][1]; } } return []; } var amounts = getColumn("N_AMT"); var custIds = getColumn("C_CUST_ID"); var glList = getColumn("C_GL_SL"); var titles = getColumn("C_AC_TITLE"); var accOutput = []; var titleOutput = []; for (var i = 0; i < amounts.length; i++) { var amt = parseFloat(amounts[i] || 0); var custId = (custIds[i] || "").trim(); if (amt > 10000000 && custId !== "") { var gl = glList[i] || ""; var title = titles[i] || ""; accOutput.push(gl + " : " + amt); titleOutput.push(gl + " : " + title); } } // No qualifying records if (accOutput.length === 0) { return false; } // Append to screen fields document.MAINFORM.OPERA_ACC_NO_LEG_COMP.value = accOutput.join(", "); document.MAINFORM.NATURE_LEG_COMP.value = titleOutput.join(", "); return true; }
