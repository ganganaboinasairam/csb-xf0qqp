private static void verifyCommitBeforePush( File repo, String branch, String expectedCommitMsg, List selectedFiles, Consumer log) throws Exception { info(log, "Verifying commit before push..."); // 1️⃣ Verify HEAD commit message String headMsg = GitCli.run( repo, "log", "-1", "--pretty=format:%s" ).trim(); if (!headMsg.equals(expectedCommitMsg)) { throw new RuntimeException( "Commit verification failed: HEAD message mismatch\n" + "Expected: " + expectedCommitMsg + "\n" + "Actual : " + headMsg ); } info(log, "✔ Commit message verified"); // 2️⃣ Verify branch is ahead of remote String ahead = GitCli.run( repo, "rev-list", "--count", "origin/" + branch + "..HEAD" ).trim(); if ("0".equals(ahead)) { throw new RuntimeException( "Commit verification failed: No new commits to push" ); } info(log, "✔ Local branch ahead by " + ahead + " commit(s)"); // 3️⃣ Verify committed files String committedFiles = GitCli.run( repo, "show", "--name-only", "--pretty=format:", "HEAD" ); for (String f : selectedFiles) { if (!committedFiles.contains(f)) { throw new RuntimeException( "Commit verification failed: File missing in commit -> " + f ); } } info(log, "✔ Selected files verified in commit"); // 4️⃣ Verify clean working tree String status = GitCli.run(repo, "status", "--porcelain").trim(); if (!status.isEmpty()) { throw new RuntimeException( "Working tree not clean after commit:\n" + status ); } info(log, "✔ Working tree clean"); }
