package com.service; import java.io.IOException; import java.nio.file.DirectoryStream; import java.nio.file.Files; import java.nio.file.Path; import java.nio.file.Paths; import java.nio.file.StandardCopyOption; import java.text.SimpleDateFormat; import java.util.ArrayList; import java.util.Arrays; import java.util.Date; import java.util.HashMap; import java.util.List; import java.util.Map; import java.util.stream.Stream; public class FolderReconstructor { public static void main(String[] args) throws IOException { if (args.length < 2) { System.err.println( "\"Use Case ::::: java -jar FolderReconstructor.jar \""); System.exit(1); } Date date = new Date(); SimpleDateFormat sdf = new SimpleDateFormat("_ddMMyyyy_HHmm"); String strDate = sdf.format(date); Path baseDomain = Paths.get(args[1]); Path prodFolder = Paths.get(args[0]); Path outputFolder = Paths.get("E:/Reconstructed/base_domain" + strDate); List allowedFolders = loadAllowedFolders(); System.out.println("Allowed Folders: " + allowedFolders); Map> fileMap = buildLocalFileMap(baseDomain, allowedFolders); System.out.println("File map size: " + fileMap.size()); Files.createDirectories(outputFolder); try (DirectoryStream stream = Files.newDirectoryStream(prodFolder)) { for (Path prodFile : stream) { if (!Files.isRegularFile(prodFile)) continue; String originalName = prodFile.getFileName().toString(); String normalizedName = normalizeFileName(originalName); Path targetPath; // JSP SPECIAL HANDLING if (normalizedName.toLowerCase().endsWith(".jsp")) { boolean isRelVal = isReleaseValidationJsp(prodFile); if (isRelVal) { targetPath = outputFolder .resolve("EE.ear/EximBillWeb.war/SCRN/releaseValidation/" + normalizedName); System.out.println("JSP (RelVal): " + normalizedName); } else { targetPath = outputFolder.resolve("EE.ear/EximBillWeb.war/SCRN/" + normalizedName); System.out.println("JSP (SCRN): " + normalizedName); } } else { List matches = fileMap.get(normalizedName.toLowerCase()); if (matches != null && matches.size() == 1) { targetPath = outputFolder.resolve(matches.get(0)); System.out.println("Mapped: " + normalizedName); } else if (matches != null && matches.size() > 1) { targetPath = outputFolder.resolve("FOUND_IN_MULTIPLE_FOLDERS/" + normalizedName); System.out.println("Duplicate match: " + normalizedName); } else { targetPath = outputFolder.resolve("UNMAPPED/" + normalizedName); System.out.println("Unmapped: " + normalizedName); } } Files.createDirectories(targetPath.getParent()); Files.move(prodFile, targetPath, StandardCopyOption.REPLACE_EXISTING); } } System.out.println("\n Reconstruction completed. Folder path is '" + outputFolder + "'"); } private static List loadAllowedFolders() throws IOException { return Arrays.asList("caljs", "ee.ear", "default", "int", "ee_sys", "properties", "lib"); } // Build file map ONLY for allowed folders private static Map> buildLocalFileMap(Path baseDomain, List allowedFolders) throws IOException { Map> map = new HashMap<>(); Files.walk(baseDomain, 10).filter(Files::isRegularFile).forEach(path -> { Path relativePath = baseDomain.relativize(path); String relativePathStr = relativePath.toString().replace("\\", "/").toLowerCase(); boolean isAllowed = allowedFolders.stream().anyMatch(relativePathStr::startsWith); if (isAllowed) { String fileName = path.getFileName().toString().toLowerCase(); map.computeIfAbsent(fileName, k -> new ArrayList<>()).add(relativePath); } }); return map; } // Normalize file name (.js.txt .js) private static String normalizeFileName(String fileName) { String lower = fileName.toLowerCase(); if (lower.endsWith(".js.txt") || lower.endsWith(".css.txt") || lower.endsWith(".jsp.txt") || lower.endsWith(".html.txt") || lower.endsWith(".json.txt")) { return fileName.substring(0, fileName.length() - 4); } if (lower.endsWith(".txt")) { return fileName.substring(0, fileName.length() - 4); } return fileName; } // JSP Rule (Java 8 compatible) private static boolean isReleaseValidationJsp(Path file) { try { byte[] content = Files.readAllBytes(file); String text = new String(content, java.nio.charset.StandardCharsets.ISO_8859_1); return text.contains("<%@include") && text.contains("RelVal.jsp"); } catch (IOException e) { e.printStackTrace(); } return false; } }
