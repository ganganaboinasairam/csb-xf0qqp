function enableAutoRound(input) { if (!input) return; const valueDescriptor = Object.getOwnPropertyDescriptor( HTMLInputElement.prototype, "value" ); Object.defineProperty(input, "value", { get: function () { return valueDescriptor.get.call(this); }, set: function (val) { if (val !== "" && !isNaN(val)) { val = Number(val).toFixed(2); } valueDescriptor.set.call(this, val); } }); }
