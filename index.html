import java.io.*; import java.util.Properties; import javax.xml.parsers.*; import org.w3c.dom.*; import org.xml.sax.InputSource; import storexml.StoreXML; public class SFTPFileProcessor { public void fetchAndProcess() { Properties pr = new Properties(); try { pr.load(new FileInputStream("/TFEEAPP/base_domain/Railways/properties/XmlgenPath.properties")); String remotePath = pr.getProperty("sftpDir"); // example: /remote/path/ String localTemp = pr.getProperty("localTemp"); // example: /tmp/xml/ // Step 1: List remote files ProcessBuilder pbList = new ProcessBuilder("sftp", "-oPort=2201", "user@12.02.54.256"); Process listProcess = pbList.start(); BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(listProcess.getOutputStream())); writer.write("ls " + remotePath + "\n"); writer.write("exit\n"); writer.flush(); BufferedReader reader = new BufferedReader(new InputStreamReader(listProcess.getInputStream())); String line; while ((line = reader.readLine()) != null) { if (!line.trim().isEmpty() && !line.contains("sftp>")) { String fileName = line.trim(); System.out.println("Found file: " + fileName); // Step 2: Download file downloadFile(remotePath, localTemp, fileName); // Step 3: Parse & Process File downloadedFile = new File(localTemp + fileName); String content = new String(java.nio.file.Files.readAllBytes(downloadedFile.toPath()), "UTF-8"); processXML(content); } } writer.close(); reader.close(); listProcess.waitFor(); } catch (Exception e) { e.printStackTrace(); } } private void downloadFile(String remotePath, String localTemp, String fileName) { try { ProcessBuilder pbGet = new ProcessBuilder("sftp", "-oPort=2201", "user@12.02.54.256"); Process getProcess = pbGet.start(); BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(getProcess.getOutputStream())); writer.write("get " + remotePath + fileName + " " + localTemp + fileName + "\n"); writer.write("exit\n"); writer.flush(); writer.close(); getProcess.waitFor(); System.out.println("Downloaded: " + fileName); } catch (Exception e) { e.printStackTrace(); } } private void processXML(String xmlContent) { try { DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance(); factory.setNamespaceAware(true); DocumentBuilder builder = factory.newDocumentBuilder(); Document document = builder.parse(new InputSource(new StringReader(xmlContent))); NodeList contractNodes = document.getElementsByTagName("CONTRACTDTL"); NodeList vendorNodes = document.getElementsByTagName("VENDORDTL"); if (contractNodes.getLength() > 0) { for (int i = 0; i < contractNodes.getLength(); i++) { Element contractElement = (Element) contractNodes.item(i); String contractId = getTagValue(contractElement, "contractid"); String contractValue = contractElement.getTextContent().trim(); new StoreXML(contractElement, contractId, contractValue, "C"); } } else if (vendorNodes.getLength() > 0) { for (int i = 0; i < vendorNodes.getLength(); i++) { Element vendorElement = (Element) vendorNodes.item(i); String vendorId = getTagValue(vendorElement, "vendorid"); String vendorValue = vendorElement.getTextContent().trim(); new StoreXML(vendorElement, vendorId, vendorValue, "V"); } } else { System.out.println("No CONTRACTDTL or VENDORDTL found in the file."); } } catch (Exception e) { e.printStackTrace(); } } private String getTagValue(Element parent, String tagName) { NodeList nodeList = parent.getElementsByTagName(tagName); if (nodeList.getLength() > 0) { return nodeList.item(0).getTextContent().trim(); } return null; } }
