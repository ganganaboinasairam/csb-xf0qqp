import java.io.*; import java.nio.file.*; import java.util.*; public class SFTPManager { private Process process; private BufferedWriter writer; private BufferedReader reader; public SFTPManager(String user, String host, int port) throws IOException { ProcessBuilder pb = new ProcessBuilder("sftp", "-oPort=" + port, user + "@" + host); pb.redirectErrorStream(true); process = pb.start(); writer = new BufferedWriter(new OutputStreamWriter(process.getOutputStream())); reader = new BufferedReader(new InputStreamReader(process.getInputStream())); waitForPrompt(); } private void waitForPrompt() throws IOException { String line; long start = System.currentTimeMillis(); while ((line = reader.readLine()) != null) { System.out.println("SFTP >> " + line); if (line.trim().endsWith("sftp>")) break; if (System.currentTimeMillis() - start > 10000) { throw new IOException("Timed out waiting for sftp prompt"); } } } private void sendCommand(String command) throws IOException { writer.write(command); writer.newLine(); writer.flush(); waitForPrompt(); } public void getFile(String remotePath, String localPath) throws IOException { Path localDir = Paths.get(localPath).getParent(); if (localDir != null && !Files.exists(localDir)) { Files.createDirectories(localDir); } sendCommand("get " + remotePath + " " + localPath); } public void putFile(String localPath, String remotePath) throws IOException { sendCommand("put " + localPath + " " + remotePath); } public void close() throws IOException { sendCommand("bye"); process.destroy(); } public static List loadFilePairs(String filePath) throws IOException { List filePairs = new ArrayList<>(); try (BufferedReader br = new BufferedReader(new FileReader(filePath))) { String line; while ((line = br.readLine()) != null) { if (!line.trim().isEmpty()) { String[] parts = line.trim().split(";", 2); // local;remote if (parts.length == 2) { filePairs.add(parts); } else { throw new IOException("Invalid line in file list: " + line); } } } } return filePairs; } public static void main(String[] args) { try { Properties props = new Properties(); props.load(new FileInputStream("config.properties")); String host = props.getProperty("sftp.host"); int port = Integer.parseInt(props.getProperty("sftp.port")); String user = props.getProperty("sftp.user"); String fileListPath = props.getProperty("file.list.path"); String operation = props.getProperty("operation", "download"); // "download" or "upload" SFTPManager sftp = new SFTPManager(user, host, port); List filePairs = loadFilePairs(fileListPath); for (String[] pair : filePairs) { if ("download".equalsIgnoreCase(operation)) { String remote = pair[0]; String local = pair[1]; System.out.println("Downloading: " + remote); sftp.getFile(remote, local); } else if ("upload".equalsIgnoreCase(operation)) { String local = pair[0]; String remote = pair[1]; System.out.println("Uploading: " + local); sftp.putFile(local, remote); } } sftp.close(); System.out.println("Operation completed."); } catch (Exception e) { e.printStackTrace(); } } }
