async function fetchIRDetails(irNumber) { const url = `https://irtra.stnktimes.in/CGI/Utilities/GetSubjectline.cgi?ir=${irNumber}`; const res = await fetch(url); const html = await res.text(); const parser = new DOMParser(); const doc = parser.parseFromString(html, 'text/html'); // IR Number const irNumEl = doc.querySelector('a.nav-item'); const irNum = irNumEl?.textContent.replace('IR Number:', '').trim(); // IR Description — from textarea, extract after "Desc:" const textarea = doc.querySelector('textarea'); const descMatch = textarea?.value.match(/Desc:([^/]+)/); const irDesc = descMatch?.[1]?.trim(); // IR Status const allText = doc.body.innerText; const statusMatch = allText.match(/STATUS:\s*(.+)/i); const irStatus = statusMatch?.[1]?.trim(); return { irNum, irDesc, irStatus }; }
