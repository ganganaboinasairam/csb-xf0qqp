function convertToJson(text) { const lines = text.replace(/\r/g, "").split("\n"); const result = { attributes: [], versionLabels: {}, revisions: [] }; let section = ""; let currentRevision = null; for (let i = 0; i < lines.length; i++) { const line = lines[i].trim(); if (!line) continue; if (line.startsWith("Archive:")) { result.archive = line.substring(8).trim(); } else if (line.startsWith("Workfile:")) { result.workfile = line.substring(9).trim(); } else if (line.startsWith("Archive created:")) { result.archiveCreated = line.substring(16).trim(); } else if (line.startsWith("Split mode:")) { result.splitMode = line.substring(11).trim(); } else if (line.startsWith("Owner:")) { result.owner = line.substring(6).trim(); } else if (line.startsWith("Last trunk rev:")) { result.lastTrunkRevision = line.substring(15).trim(); } else if (line.startsWith("Locks:")) { result.locks = line.substring(6).trim(); } else if (line.startsWith("Groups:")) { result.groups = line.substring(7).trim(); } else if (line.startsWith("Rev count:")) { result.revisionCount = Number(line.substring(10).trim()); } else if (line === "Attributes:") { section = "attributes"; } else if (line === "Version labels:") { section = "labels"; } else if (line === "Description:") { section = "description"; } else if (line.startsWith("Rev ")) { section = "revision"; if (currentRevision) result.revisions.push(currentRevision); currentRevision = { revision: line.replace("Rev", "").trim() }; } else if (section === "attributes") { if (line.includes("=")) { const parts = line.split("="); result[parts[0].trim()] = parts[1].trim().replace(/^"|"$/g, ""); } else { result.attributes.push(line); } } else if (section === "labels") { const match = line.match(/"(.+)"\s*=\s*(.+)/); if (match) { result.versionLabels[match[1]] = match[2].trim(); } } else if (section === "description") { if (!result.description) result.description = ""; result.description += line + "\n"; } else if (section === "revision") { if (line.startsWith("Checked in:")) { currentRevision.checkedIn = line.substring(11).trim(); } else if (line.startsWith("Last modified:")) { currentRevision.lastModified = line.substring(14).trim(); } else if (line.startsWith("Author id:")) { const m = line.match( /Author id:\s*(\S+).*?(\d+)\/(\d+)\/(\d+)/ ); if (m) { currentRevision.authorId = m[1]; currentRevision.linesDeleted = Number(m[2]); currentRevision.linesAdded = Number(m[3]); currentRevision.linesMoved = Number(m[4]); } } else { currentRevision.remarks = (currentRevision.remarks || "") + " " + line; } } } if (currentRevision) result.revisions.push(currentRevision); if (result.description) result.description = result.description.trim(); return result; }
