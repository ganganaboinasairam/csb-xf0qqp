function getMainFormValues() { var form = document.getElementById("MAINFORM"); var result = []; var fields = form.querySelectorAll("input, select, textarea"); fields.forEach(function(field) { // Ignore buttons if (field.tagName.toLowerCase() === "button" || field.type === "button" || field.type === "submit" || field.type === "reset" || field.type === "image") { return; } var value = ""; // Radio / Checkbox if (field.type === "radio" || field.type === "checkbox") { if (!field.checked) { return; } value = field.value || ""; } else { value = field.value || ""; } // Ignore fields with no value if (value.trim() === "") { return; } // -------------------------------- // Field Title // -------------------------------- var title = field.getAttribute("title") || field.getAttribute("aria-label") || ""; if (!title && field.id) { var label = form.querySelector( 'label[for="' + field.id + '"]' ); if (label) { title = label.innerText.trim(); } } if (!title) { title = field.name || field.id || ""; } // -------------------------------- // Field Length // -------------------------------- var maxLength = field.getAttribute("maxlength"); var fieldLength; if (maxLength !== null && maxLength !== "") { fieldLength = parseInt(maxLength, 10); } else { fieldLength = value.length; } // -------------------------------- // Field Status // -------------------------------- var fieldStatus = ""; if (field.classList.contains("CHAR_O")) { fieldStatus = "Optional"; } else if (field.classList.contains("CHAR_M")) { fieldStatus = "Mandatory"; } else if (field.classList.contains("CHAR_P")) { fieldStatus = "Protected"; } // -------------------------------- // Add Result // -------------------------------- result.push({ "Field Title": title, "Field ID": field.id || "", "Field Value": value, "Field Length": fieldLength, "Field Status": fieldStatus }); }); console.table(result); return result; } getMainFormValues();
