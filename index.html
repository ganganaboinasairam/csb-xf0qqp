import com.jcraft.jsch.*; public class SFTPWithKeyAuth { public static void main(String[] args) { String host = "your.sftp.server.com"; int port = 22; // Default SFTP port String user = "yourUsername"; String privateKeyPath = "C:/Users/yourUser/.ssh/id_rsa"; // Path to your private key file Session session = null; ChannelSftp channelSftp = null; try { // Create a JSch instance JSch jsch = new JSch(); // Add private key for authentication jsch.addIdentity(privateKeyPath); // Create a session session = jsch.getSession(user, host, port); // Disable host key checking (for testing; remove in production) session.setConfig("StrictHostKeyChecking", "no"); // Connect to the server session.connect(); System.out.println("Connected to SFTP server using private key"); // Open SFTP channel Channel channel = session.openChannel("sftp"); channel.connect(); channelSftp = (ChannelSftp) channel; System.out.println("SFTP Channel opened"); // List files in the root directory System.out.println("Listing files in home directory:"); for (Object entry : channelSftp.ls(".")) { System.out.println(entry.toString()); } // Close connections channelSftp.exit(); session.disconnect(); System.out.println("Disconnected from SFTP server"); } catch (JSchException | SftpException e) { e.printStackTrace(); } } }
