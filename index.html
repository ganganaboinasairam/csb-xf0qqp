package com.tcs.git; import java.io.BufferedReader; import java.io.File; import java.io.InputStreamReader; import java.util.ArrayList; import java.util.List; import java.util.Set; import java.util.TreeSet; import com.tcs.git.config.Config; public class GitAutomationService { public static void execute(String region, String crNo, String selection, java.util.function.Consumer log) throws Exception { try { File repo = new File(Config.get("git.repo.path." + region)); String branch = Config.get("git.branch.name." + region); log.accept("Repository: " + repo.getAbsolutePath()); // Check changes String status = GitCli.run(repo, "status", "--porcelain"); if (status.trim().isEmpty()) { log.accept("No changes to commit."); return; } // Generate commit message String commitMsg = CommitMessageGenerator.generate(repo, crNo); log.accept("Staging changes..."); GitCli.run(repo, "add", "-A"); log.accept("Committing: " + commitMsg); GitCli.run(repo, "commit", "-m", commitMsg); // Pull first log.accept("Pulling with rebase..."); GitCli.run(repo, "pull", "--rebase", "origin", branch); log.accept("Pushing to origin/" + branch); GitCli.run(repo, "push", "origin", branch); log.accept("DONE. Commit " + commitMsg + " pushed successfully."); MainUI.commitMsgField.setText(commitMsg); } catch (Exception e) { // TODO Auto-generated catch block e.printStackTrace(); } } public static void jnksTrigger(String region, String commitMsg, String selection, java.util.function.Consumer log) throws Exception { log.accept("Triggering Jenkins build..."); String queueUrl = JenkinsClient.trigger(region, commitMsg, selection); log.accept("Jenkins queued: " + queueUrl); int buildNo = JenkinsClient.waitForBuildNumber(queueUrl, 120); log.accept("Jenkins build started: #" + buildNo); String result = JenkinsClient.waitForBuildResult(region, buildNo, log); log.accept("Jenkins build result: " + result); } public static String runCustom(String region, String command) throws Exception { if (!command.startsWith("git ")) { throw new IllegalArgumentException("Only git commands allowed"); } String gitExe = Config.get("git.executable"); File repo = new File(Config.get("git.repo.path." + region)); List parts = new ArrayList<>(); parts.add(gitExe); for (String p : command.substring(4).split(" ")) { if (!p.trim().isEmpty()) { parts.add(p); } } ProcessBuilder pb = new ProcessBuilder(parts); pb.directory(repo); pb.redirectErrorStream(true); Process p = pb.start(); StringBuilder out = new StringBuilder(); try (BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()))) { String line; while ((line = br.readLine()) != null) { out.append(line).append("\n"); } } int exit = p.waitFor(); out.append("\nExit code: ").append(exit); return out.toString(); } public static void chkStatus(String region, java.util.function.Consumer log) throws Exception { File repo = new File(Config.get("git.repo.path." + region)); log.accept("Repository: " + repo.getAbsolutePath()); log.accept("Checking status...\n"); String status = GitCli.run(repo, "status", "--porcelain"); if (status.trim().isEmpty()) { log.accept("No changes found. Working tree clean."); return; } log.accept("Files to be committed:\n"); String[] lines = status.split("\\r?\\n"); for (String line : lines) { if (line.length() < 3) continue; String flag = line.substring(0, 2).trim(); String file = line.substring(3); if ("??".equals(flag)) { log.accept("[NEW] " + file); } else if (flag.contains("M")) { log.accept("[MODIFIED] " + file); } else if (flag.contains("A")) { log.accept("[ADDED] " + file); } else if (flag.contains("D")) { log.accept("[DELETED] " + file); } else { log.accept("[CHANGED] " + file); } } } public static void rebase(String region, java.util.function.Consumer log) throws Exception { File repo = new File(Config.get("git.repo.path." + region)); String branch = Config.get("git.branch.name." + region); log.accept("Repository: " + repo.getAbsolutePath()); log.accept("rebasing...\n"); String status = GitCli.run(repo, "pull", "--rebase", "origin", branch); if (status.trim().isEmpty()) { log.accept("Working tree clean."); return; } log.accept("Rebase completed \n"); String[] lines = status.split("\\r?\\n"); for (String line : lines) { log.accept(line); } } public static void retrieveFiles( String region, String crOrIr, java.util.function.Consumer log, java.util.function.IntConsumer progress) throws Exception { File repo = new File(Config.get("git.repo.path." + region)); log.accept("Repository : " + repo.getAbsolutePath()); log.accept("Searching commits for : " + crOrIr); log.accept(""); final int BATCH_SIZE = 50; int offset = 0; int batchNo = 0; Set files = new TreeSet<>(); while (true) { batchNo++; log.accept("Scanning batch " + batchNo + " (commits " + offset + " - " + (offset + BATCH_SIZE) + ")"); String out = GitCli.run( repo, "log", "--skip=" + offset, "--max-count=" + BATCH_SIZE, "--name-only", "--pretty=format:", "--grep=" + crOrIr ); // If nothing returned, check if history ended if (out.trim().isEmpty()) { String hasMore = GitCli.run( repo, "log", "--skip=" + offset, "--max-count=1", "--pretty=oneline" ); if (hasMore.trim().isEmpty()) { break; // end of history } offset += BATCH_SIZE; progress.accept(Math.min(90, batchNo * 10)); continue; } // Parse files for (String line : out.split("\\R")) { if (!line.trim().isEmpty()) { files.add(line.trim()); } } break; // found matches, stop scanning } if (files.isEmpty()) { log.accept("No files found for " + crOrIr); progress.accept(100); return; } progress.accept(100); log.accept(""); log.accept("Files committed under " + crOrIr + ":"); log.accept("----------------------------------------------"); for (String f : files) { log.accept(f); } } }
