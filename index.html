public static void chkStatus( String region, java.util.function.Consumer log) throws Exception { File repo = new File(Config.get("git.repo.path." + region)); log.accept("[INFO] Checking status in: " + repo.getAbsolutePath()); log.accept(""); String status = GitCli.run(repo, "status", "--porcelain"); if (status.trim().isEmpty()) { log.accept("[CLEAN] Working tree clean"); return; } log.accept("[INFO] Changes detected:"); log.accept("----------------------------------------------"); for (String line : status.split("\\R")) { if (line.length() < 3) continue; String indexStatus = line.substring(0, 1); String workTreeStatus = line.substring(1, 2); String file = line.substring(3); // Untracked if ("?".equals(indexStatus) && "?".equals(workTreeStatus)) { log.accept("[UNTRACKED] " + file); continue; } // Added if ("A".equals(indexStatus)) { log.accept("[ADDED] " + file); continue; } // Modified if ("M".equals(indexStatus) || "M".equals(workTreeStatus)) { log.accept("[MODIFIED] " + file); continue; } // Deleted if ("D".equals(indexStatus) || "D".equals(workTreeStatus)) { log.accept("[DELETED] " + file); continue; } // Renamed / Copied if ("R".equals(indexStatus) || "C".equals(indexStatus)) { log.accept("[RENAMED] " + file); continue; } // Fallback log.accept("[CHANGED] " + file); } }
