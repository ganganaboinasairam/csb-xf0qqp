function checkAllKeysHaveValues(jsonObject) { // Define custom alert messages for specific keys const alertMessages = { name: "Name is required.", age: "Age is missing or invalid.", city: "City cannot be empty." }; for (let key in jsonObject) { if (jsonObject.hasOwnProperty(key)) { // Check if the value is null, undefined, or an empty string if (jsonObject[key] === null || jsonObject[key] === undefined || jsonObject[key] === "") { // Use custom alert message if defined, otherwise use a default message const message = alertMessages[key] || `The key "${key}" is missing a value.`; alert(message); return false; // Stop further checks if a missing value is found } } } return true; // All keys have valid values } // Example usage const exampleObject = { name: "", age: null, city: "New York" }; if (checkAllKeysHaveValues(exampleObject)) { console.log("All keys have valid values."); } else { console.log("Some keys are missing values."); }
