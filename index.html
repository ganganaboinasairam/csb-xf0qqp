package com.reminder; import java.util.List; import java.util.concurrent.*; import java.util.stream.Collectors; public class BubbleReminderScheduler { private static final int INTERVAL_MINUTES = 30; private static final int DELAY_BETWEEN_BUBBLES_SEC = 6; public static void start() { ScheduledExecutorService scheduler = Executors.newSingleThreadScheduledExecutor(); scheduler.scheduleAtFixedRate(() -> { List tasks = TaskJsonStore.loadTasks(); List pendingTasks = tasks.stream() .filter(t -> "PENDING".equalsIgnoreCase(t.getStatus())) .collect(Collectors.toList()); if (pendingTasks.isEmpty()) { return; } for (Task task : pendingTasks) { SystemTrayManager.showBubble( "Pending Task [" + task.getPriority() + "]", task.getDescription() ); try { Thread.sleep(DELAY_BETWEEN_BUBBLES_SEC * 1000L); } catch (InterruptedException ignored) { } } }, 0, INTERVAL_MINUTES, TimeUnit.MINUTES); } }
