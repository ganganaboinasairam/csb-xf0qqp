package com.tcs.sbi.chat; import java.io.IOException; import java.io.PrintWriter; import java.util.List; import javax.servlet.ServletException; import javax.servlet.http.HttpServlet; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletResponse; import org.json.JSONArray; import com.tcs.sbi.chat.util.ChatQADao; public class GetQuestionSuggestions extends HttpServlet { private static final long serialVersionUID = 1L; private ChatQADao dao = new ChatQADao(); @Override protected void doGet(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException { String query = request.getParameter("query"); response.setContentType("application/json"); PrintWriter out = response.getWriter(); if (query == null || query.trim().isEmpty()) { // Return empty JSON array out.print(new JSONArray().toString()); out.flush(); return; } try { List suggestions = this.dao.findQuestionsByPartial(query.trim()); // Convert List to JSONArray JSONArray jsonArray = new JSONArray(suggestions); out.print(jsonArray.toString()); } catch (Exception e) { e.printStackTrace(); // Return empty array in case of error out.print(new JSONArray().toString()); } out.flush(); } }
