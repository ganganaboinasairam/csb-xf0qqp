public static String runCustom(String region, String command) throws Exception { if (!command.startsWith("git ")) { throw new IllegalArgumentException("Only git commands allowed"); } String gitExe = Config.get("git.executable"); File repo = new File(Config.get("git.repo.path." + region)); List parts = new ArrayList<>(); parts.add(gitExe); for (String p : command.substring(4).split(" ")) { if (!p.isBlank()) { parts.add(p); } } ProcessBuilder pb = new ProcessBuilder(parts); pb.directory(repo); pb.redirectErrorStream(true); Process p = pb.start(); StringBuilder out = new StringBuilder(); try (BufferedReader br = new BufferedReader( new InputStreamReader(p.getInputStream()))) { String line; while ((line = br.readLine()) != null) { out.append(line).append("\n"); } } int exit = p.waitFor(); out.append("\nExit code: ").append(exit); return out.toString(); }
