const jsonObject = '{"actionType": "CREATE_QUOTE","BRANCHCODE": "09995","C_MAIN_REF": ""}'; function validateJsonObject(jsonStr) { // Parse the JSON string into a JavaScript object const jsonObject = JSON.parse(jsonStr); // Custom alert messages for specific keys const alertMessages = { actionType: "Action type is required.", BRANCHCODE: "Branch code is missing.", C_MAIN_REF: "Main reference is required." }; for (let key in jsonObject) { if (jsonObject.hasOwnProperty(key)) { // Check if the value is null, undefined, or an empty string if (jsonObject[key] === null || jsonObject[key] === undefined || jsonObject[key].toString().trim() === "") { // Use custom alert message if defined, otherwise use a default message const message = alertMessages[key] || `The key "${key}" is missing a value.`; alert(message); return false; // Stop further checks if a missing value is found } } } return true; // All keys have valid values } // Example usage if (validateJsonObject(jsonObject)) { console.log("All keys have valid values."); } else { console.log("Some keys are missing values."); }
