package com.cs.ceweb.servlets; import java.io.File; import java.io.FileOutputStream; import java.io.IOException; import java.io.PrintWriter; import java.sql.Connection; import java.sql.PreparedStatement; import java.sql.ResultSet; import javax.servlet.ServletException; import javax.servlet.http.HttpServlet; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletResponse; import com.cs.base.utility.ASPathConst; import java.util.logging.Logger; public class ImageDownloadServlet extends HttpServlet { private static final long serialVersionUID = 1L; private static final Logger LOGGER = Logger.getLogger(ImageDownloadServlet.class.getName()); protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException { String indexList = request.getParameter("indexList"); String c_main_ref = request.getParameter("cMainRef"); String reqType = request.getParameter("reqType"); String imgIndex = request.getParameter("imgIndex"); try (Connection conn = DSManager.getConnection("CET")) { if ("download".equals(reqType)) { handleDownloadRequest(indexList, c_main_ref, conn, response); } else if ("viewDoc".equals(reqType)) { handleViewDocRequest(imgIndex, c_main_ref, conn, response); } } catch (Exception e) { LOGGER.log(Level.SEVERE, "Error occurred while processing the request", e); response.sendError(HttpServletResponse.SC_INTERNAL_SERVER_ERROR, "An error occurred while processing the request."); } } private void handleDownloadRequest(String indexList, String cMainRef, Connection conn, HttpServletResponse response) throws Exception { String[] array = indexList.split(";"); for (String value : array) { try (PreparedStatement pstmt = conn.prepareStatement( "SELECT B_IMG_CONTENT, C_IMG_FILE_TYPE, C_IMG_DOC_TYPE FROM TRX_IMAGES_DTAL WHERE C_IMG_INDX = ? AND C_MAIN_REF = ?")) { pstmt.setString(1, value); pstmt.setString(2, cMainRef); try (ResultSet rs = pstmt.executeQuery()) { if (rs.next()) { processRecord(rs, value, response); } else { handleNoDataFound(response); } } } } response.setContentType("application/json"); response.getWriter().write("{\"message\":\"Files Saved Successfully\"}"); } private void handleViewDocRequest(String imgIndex, String cMainRef, Connection conn, HttpServletResponse response) throws Exception { try (PreparedStatement pstmt = conn.prepareStatement( "SELECT B_IMG_CONTENT, C_IMG_FILE_TYPE, C_IMG_DOC_TYPE FROM TRX_IMAGES_DTAL WHERE C_IMG_INDX = ? AND C_MAIN_REF = ?")) { pstmt.setString(1, imgIndex); pstmt.setString(2, cMainRef); try (ResultSet rs = pstmt.executeQuery()) { if (rs.next()) { processRecord(rs, imgIndex, response); } else { handleNoDataFound(response); } } } } private void processRecord(ResultSet rs, String index, HttpServletResponse response) throws Exception { String C_IMG_DOC_TYPE = rs.getString("C_IMG_DOC_TYPE"); String base64 = rs.getString("B_IMG_CONTENT"); String fileType = rs.getString("C_IMG_FILE_TYPE"); if (base64 != null && !base64.isEmpty() && !"null".equalsIgnoreCase(base64)) { if ("image".equals(C_IMG_DOC_TYPE)) { Object obj = WSZipUtil.decompressByBase64(base64); printContentInTheFile((String) obj, index); } else if ("form".equals(C_IMG_DOC_TYPE)) { processFileResponse(base64, fileType, response); } } else { handleNoDataFound(response); } } private void printContentInTheFile(String obj, String c_img_indx) throws Exception { LOGGER.log(Level.INFO, "Inside printContentInTheFile method"); String configFilePath = ASPathConst.getUserDirPath() + File.separator + "PARAM" + File.separator + "Default" + File.separator + "SYST" + File.separator + "sysparam.xml"; Node sImagePathNode = XMLManager.findNode( XMLManager.xmlFileToDom(configFilePath).getDocumentElement(), "WEB_IMAGE_ROOT_PATH"); String sImagePath = XMLManager.getNodeAttribute(sImagePathNode, "VALUE"); File file = new File(sImagePath, c_img_indx + ".pdf"); try (FileOutputStream out = CESecuFileHelper.newFileOutputStream(file)) { byte[] b = obj.getBytes("ISO-8859-1"); out.write(b); } } private void processFileResponse(String base64, String fileType, HttpServletResponse response) throws IOException { try { byte[] b = null; Object obj = WSZipUtil.decompressByBase64(base64); response.setContentType(getContentType(fileType)); response.setHeader("Pragma", "No-cache"); response.setHeader("Cache-Control", "no-cache"); response.setDateHeader("Expires", 2002L); try (ObjectOutputStream out = new ObjectOutputStream(response.getOutputStream())) { out.writeObject(obj); } } catch (IllegalArgumentException e) { LOGGER.log(Level.WARNING, "Invalid base64 content", e); response.sendError(HttpServletResponse.SC_BAD_REQUEST, "Invalid base64 content."); } catch (Exception e) { LOGGER.log(Level.SEVERE, "Error while processing the file", e); response.sendError(HttpServletResponse.SC_INTERNAL_SERVER_ERROR, "An error occurred while processing the file."); } } private String getContentType(String fileType) { switch (fileType.toUpperCase()) { case "PDF": return "application/pdf"; case "TXT": return "text/plain"; case "XLS": case "XLSX": return "application/vnd.ms-excel"; case "JPG": case "JPEG": return "image/jpeg"; default: return "application/octet-stream"; } } private void handleNoDataFound(HttpServletResponse response) throws IOException { LOGGER.log(Level.WARNING, "No Report data found in DB."); response.setContentType("text/html"); try (PrintWriter out = response.getWriter()) { out.println(""); out.println( "
Report file not found...
"); out.println(""); } } }
