public static RebaseResult rebase( String region, java.util.function.Consumer log) { File repo = new File(Config.get("git.repo.path." + region)); String branch = Config.get("git.branch.name." + region); log.accept("[INFO] Repository : " + repo.getAbsolutePath()); log.accept("[INFO] Rebasing from origin/" + branch); log.accept(""); try { // 1️⃣ Capture HEAD before rebase String beforeHead = GitCli.run(repo, "rev-parse", "HEAD").trim(); log.accept("[DEBUG] HEAD before rebase: " + beforeHead); // 2️⃣ Perform rebase GitCli.run(repo, "pull", "--rebase", "--autostash", "origin", branch); // 3️⃣ Check conflicts String conflicts = GitCli.run(repo, "diff", "--name-only", "--diff-filter=U").trim(); if (!conflicts.isEmpty()) { log.accept("[ERROR] Rebase stopped due to conflicts:"); for (String f : conflicts.split("\\R")) { log.accept(" " + f); } return RebaseResult.conflict(); } // 4️⃣ Capture HEAD after rebase String afterHead = GitCli.run(repo, "rev-parse", "HEAD").trim(); log.accept("[DEBUG] HEAD after rebase : " + afterHead); // 5️⃣ If HEAD unchanged → nothing rebased if (beforeHead.equals(afterHead)) { log.accept("[INFO] Rebase completed – branch already up to date."); return RebaseResult.noChange(); } // 6️⃣ Detect changed files between commits String diff = GitCli.run( repo, "diff-tree", "--no-commit-id", "--name-status", "-r", beforeHead, afterHead ).trim(); if (diff.isEmpty()) { log.accept("[INFO] Rebase completed – commits changed, files identical."); return RebaseResult.noFileChange(); } List files = new ArrayList<>(); log.accept("[INFO] Files changed due to rebase:"); log.accept("----------------------------------------------"); for (String line : diff.split("\\R")) { log.accept(" " + line); files.add(line); } log.accept(""); log.accept("[INFO] Total files changed: " + files.size()); return RebaseResult.changed(files); } catch (Exception e) { log.accept("[ERROR] Rebase failed"); log.accept("[ERROR] " + e.getMessage()); return RebaseResult.failed(e.getMessage()); } } public class RebaseResult { public final boolean success; public final boolean conflict; public final boolean fileChanged; public final List files; public final String error; private RebaseResult( boolean success, boolean conflict, boolean fileChanged, List files, String error) { this.success = success; this.conflict = conflict; this.fileChanged = fileChanged; this.files = files; this.error = error; } public static RebaseResult conflict() { return new RebaseResult(false, true, false, null, null); } public static RebaseResult noChange() { return new RebaseResult(true, false, false, null, null); } public static RebaseResult noFileChange() { return new RebaseResult(true, false, false, null, null); } public static RebaseResult changed(List files) { return new RebaseResult(true, false, true, files, null); } public static RebaseResult failed(String err) { return new RebaseResult(false, false, false, null, err); } }
