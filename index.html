async function getChatMsgCount(){ try{ var custId = document.getElementById('CUST_ID').value; const response = await fetch(`${window.location.origin}/ChatServer/getUnreadMsgCount?custID=${custId}&reqFrom=customer`); if (!response.ok) throw new Error('Failed to fetch previous messages'); const data = await response.json(); if(data.msgCount > 0){ document.getElementById('chatMsgCount').innerHTML=data.msgCount; $('#chatMsgCount').show(); }else{ $('#chatMsgCount').hide(); } }catch (error) { console.error('Error loading previous messages:', error); } }
