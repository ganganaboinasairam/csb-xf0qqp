function disableTable(tableId) { const table = document.getElementById(tableId); if (!table) return; // Disable all TDs table.querySelectorAll("td").forEach(td => { td.classList.add("table-td-disabled"); // Disable interactive elements inside TD td.querySelectorAll("input, select, textarea, button").forEach(el => { el.disabled = true; }); }); } function enableTable(tableId) { const table = document.getElementById(tableId); if (!table) return; // Enable all TDs table.querySelectorAll("td").forEach(td => { td.classList.remove("table-td-disabled"); // Enable interactive elements inside TD td.querySelectorAll("input, select, textarea, button").forEach(el => { el.disabled = false; }); }); }.table-td-disabled { pointer-events: none !important; user-select: none !important; opacity: 0.6; }
