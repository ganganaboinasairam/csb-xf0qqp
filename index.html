public final class CommentUtil { private CommentUtil() {} public static boolean isAlreadyCommented(String line, FileType type) { String t = line.trim(); switch (type) { case JS: case JSP: return t.startsWith("//") || t.startsWith("/*") || t.endsWith("*/"); case XML: return t.startsWith(""); default: return false; } } public static String startMarker(String tag, String indent, FileType type) { switch (type) { case XML: return indent + ""; default: return indent + "//START OF " + tag; } } public static String endMarker(String tag, String indent, FileType type) { switch (type) { case XML: return indent + ""; default: return indent + "//END OF " + tag; } } public static String commentLine(String line, FileType type) { switch (type) { case XML: return ""; default: return "//" + line; } } public static String leadingWhitespace(String line) { int i = 0; while (i < line.length() && Character.isWhitespace(line.charAt(i))) i++; return line.substring(0, i); } }
