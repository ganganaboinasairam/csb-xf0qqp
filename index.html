// Jenkins UI components private JTextField commitMsgField; private JCheckBox chkCheckout; private JCheckBox chkFileTransfer; private JCheckBox chkDeployment; private JCheckBox chkRestart; private JButton buildButton; private void initUI() { JPanel top = new JPanel(new GridLayout(1, 2)); /* ================= LEFT PANEL (GIT) ================= */ JPanel gitPanel = new JPanel(new GridBagLayout()); gitPanel.setBorder(BorderFactory.createTitledBorder("Git Operations")); GridBagConstraints c = new GridBagConstraints(); c.insets = new Insets(5,5,5,5); c.anchor = GridBagConstraints.WEST; c.gridx = 0; c.gridy = 0; gitPanel.add(new JLabel("Region:"), c); c.gridx = 1; regionBox = new JComboBox<>(new String[]{"EE", "CE"}); gitPanel.add(regionBox, c); c.gridx = 0; c.gridy = 1; gitPanel.add(new JLabel("CR Number:"), c); c.gridx = 1; crField = new JTextField(10); gitPanel.add(crField, c); c.gridx = 0; c.gridy = 2; pushButton = new JButton("Commit & Push"); gitPanel.add(pushButton, c); c.gridx = 1; statusButton = new JButton("Status"); gitPanel.add(statusButton, c); c.gridx = 2; rebaseButton = new JButton("Rebase"); gitPanel.add(rebaseButton, c); c.gridx = 0; c.gridy = 3; stashButton = new JButton("Stash"); gitPanel.add(stashButton, c); c.gridx = 1; unStashButton = new JButton("Unstash"); unStashButton.setVisible(false); gitPanel.add(unStashButton, c); /* ================= RIGHT PANEL (JENKINS) ================= */ JPanel jenkinsPanel = new JPanel(new GridBagLayout()); jenkinsPanel.setBorder(BorderFactory.createTitledBorder("Jenkins Build")); GridBagConstraints j = new GridBagConstraints(); j.insets = new Insets(5,5,5,5); j.anchor = GridBagConstraints.WEST; j.gridx = 0; j.gridy = 0; jenkinsPanel.add(new JLabel("Commit Message:"), j); j.gridx = 1; commitMsgField = new JTextField(18); jenkinsPanel.add(commitMsgField, j); j.gridx = 0; j.gridy = 1; chkCheckout = new JCheckBox("Checkout", true); jenkinsPanel.add(chkCheckout, j); j.gridx = 1; chkFileTransfer = new JCheckBox("File Transfer", true); jenkinsPanel.add(chkFileTransfer, j); j.gridx = 0; j.gridy = 2; chkDeployment = new JCheckBox("Update Deployment"); jenkinsPanel.add(chkDeployment, j); j.gridx = 1; chkRestart = new JCheckBox("Server Restart"); jenkinsPanel.add(chkRestart, j); j.gridx = 0; j.gridy = 3; j.gridwidth = 2; buildButton = new JButton("Build"); jenkinsPanel.add(buildButton, j); top.add(gitPanel); top.add(jenkinsPanel); outputArea = new JTextArea(); outputArea.setEditable(false); outputArea.setFont(new Font("Consolas", Font.PLAIN, 12)); JScrollPane scroll = new JScrollPane(outputArea); add(top, BorderLayout.NORTH); add(scroll, BorderLayout.CENTER); /* ACTIONS */ pushButton.addActionListener(e -> runTask()); statusButton.addActionListener(e -> checkStatus()); rebaseButton.addActionListener(e -> gitRebase()); stashButton.addActionListener(e -> gitStash()); unStashButton.addActionListener(e -> gitUnStash()); buildButton.addActionListener(e -> triggerJenkins()); } private void triggerJenkins() { String region = regionBox.getSelectedItem().toString().toLowerCase(); String commitMsg = commitMsgField.getText().trim(); if (commitMsg.isEmpty()) { JOptionPane.showMessageDialog( this, "Commit message required"); return; } StringBuilder sel = new StringBuilder(); if (chkCheckout.isSelected()) sel.append("Checkout,"); if (chkFileTransfer.isSelected()) sel.append("File Transfer,"); if (chkDeployment.isSelected()) sel.append("Deployment,"); if (chkRestart.isSelected()) sel.append("Restart,"); if (sel.length() == 0) { JOptionPane.showMessageDialog( this, "Select at least one Jenkins option"); return; } String selection = sel.substring(0, sel.length() - 1); outputArea.append("\n================ JENKINS BUILD =================\n"); SwingWorker worker = new SwingWorker() { protected Void doInBackground() throws Exception { String queueUrl = JenkinsClient.trigger(region, commitMsg, selection); publish("Build queued: " + queueUrl); return null; } protected void process(java.util.List msgs) { msgs.forEach(m -> outputArea.append(m + "\n")); } }; worker.execute(); }
