import java.io.*; import java.nio.file.*; import java.util.Properties; import java.util.logging.Logger; import javax.xml.parsers.DocumentBuilder; import javax.xml.parsers.DocumentBuilderFactory; import org.w3c.dom.*; import org.xml.sax.InputSource; import storexml.StoreXML; public class SFTPFileProcessor { private static final Logger logg = Logger.getLogger(SFTPFileProcessor.class.getName()); private static final int SFTP_PORT = 2202; private Properties properties; public SFTPFileProcessor() { loadProperties(); } private void loadProperties() { properties = new Properties(); try (FileInputStream fis = new FileInputStream("properties/XmlreadPath.properties")) { properties.load(fis); } catch (IOException e) { logg.severe("Failed to load properties: " + e.getMessage()); } } public void fetchAndProcess() { String remotePath = properties.getProperty("path"); String localTemp = properties.getProperty("localTemp"); String user = properties.getProperty("user"); String host = properties.getProperty("host"); try { for (String fileName : listRemoteFiles(remotePath, user, host)) { if (!fileName.toLowerCase().endsWith(".xml")) { logg.info("Skipping non-XML file: " + fileName); continue; } File localFile = new File(localTemp + fileName); if (localFile.exists()) { logg.info("Skipping (already downloaded): " + fileName); continue; } logg.info("Found new XML file: " + fileName); if (downloadFile(remotePath, localTemp, fileName, user, host)) { processDownloadedFile(localFile, fileName, user, host); } } } catch (Exception e) { logg.severe("Error in fetchAndProcess: " + e.getMessage()); e.printStackTrace(); } } private String[] listRemoteFiles(String remotePath, String user, String host) throws Exception { ProcessBuilder pb = new ProcessBuilder("sftp", "-oPort=" + SFTP_PORT, user + "@" + host); Process process = pb.start(); try (BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(process.getOutputStream())); BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream()))) { writer.write("ls " + remotePath + "\nexit\n"); writer.flush(); return reader.lines() .filter(line -> !line.trim().isEmpty() && !line.contains("sftp>")) .toArray(String[]::new); } finally { process.waitFor(); } } private boolean downloadFile(String remotePath, String localTemp, String fileName, String user, String host) { try { ProcessBuilder pb = new ProcessBuilder("sftp", "-oPort=" + SFTP_PORT, user + "@" + host); Process process = pb.start(); try (BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(process.getOutputStream()))) { writer.write("get " + remotePath + fileName + " " + localTemp + "\nexit\n"); writer.flush(); } int exitCode = process.waitFor(); logg.info("Downloaded " + fileName + " with exit code: " + exitCode); return new File(localTemp + fileName).exists(); } catch (Exception e) { logg.severe("Download failed for " + fileName + ": " + e.getMessage()); return false; } } private void processDownloadedFile(File localFile, String fileName, String user, String host) { try { String content = Files.readString(localFile.toPath()); processXML(content, fileName, user, host); } catch (IOException e) { logg.severe("Error reading downloaded file: " + e.getMessage()); } } private void processXML(String xmlContent, String fileName, String user, String host) { try { DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance(); factory.setNamespaceAware(true); Document document = factory.newDocumentBuilder().parse(new InputSource(new StringReader(xmlContent))); NodeList contracts = document.getElementsByTagName("CONTRACTDTL"); NodeList vendors = document.getElementsByTagName("VENDORDTL"); if (contracts.getLength() > 0) { processNodes(contracts, "C", fileName, user, host); } else if (vendors.getLength() > 0) { processNodes(vendors, "V", fileName, user, host); } else { logg.warning("No CONTRACTDTL or VENDORDTL found in file: " + fileName); } } catch (Exception e) { logg.severe("Error parsing XML: " + e.getMessage()); } } private void processNodes(NodeList nodes, String type, String fileName, String user, String host) { for (int i = 0; i < nodes.getLength(); i++) { Element element = (Element) nodes.item(i); String id = type.equals("C") ? getTagValue(element, "CONTRACTID") : getTagValue(element, "VENDOR_ID"); String contractValue = type.equals("C") ? getTagValue(element, "CONTRACT_VALUE") : ""; try { new StoreXML(element, id, contractValue, type); logg.info("Record inserted successfully: " + id); success(fileName, id, type, user, host); } catch (Exception e) { logg.severe("Failed to process ID: " + id + ", Error: " + e.getMessage()); fail(e, fileName, id, type, user, host); } } } private String getTagValue(Element parent, String tagName) { NodeList nodeList = parent.getElementsByTagName(tagName); return (nodeList.getLength() > 0) ? nodeList.item(0).getTextContent().trim() : null; } private void success(String filename, String id, String type, String user, String host) { String baseName = filename.substring(0, filename.lastIndexOf('.')); String statusFile = baseName + "_" + id + "_success.txt"; String ackPath = properties.getProperty("ackPath"); String localOutbox = properties.getProperty("localOutbox"); String message = (type.equals("C") ? "Contract ID :" : "Vendor ID :") + id + " from " + baseName + ".xml successfully received."; try { Files.writeString(Paths.get(statusFile), message); uploadFileToSFTP(statusFile, ackPath, user, host); Files.copy(Paths.get(statusFile), Paths.get(localOutbox, new File(statusFile).getName()), StandardCopyOption.REPLACE_EXISTING); Files.delete(Paths.get(statusFile)); logg.info("Success acknowledgment sent for: " + baseName + ".xml"); } catch (Exception e) { logg.severe("Error in success acknowledgment for " + id + ": " + e.getMessage()); } } private void fail(Exception e, String filename, String id, String type, String user, String host) { String baseName = filename.substring(0, filename.lastIndexOf('.')); String suffix = e.getMessage().contains("ORA-00001") ? "_duplicate" : "_fail"; String statusFile = baseName + "_" + id + suffix + ".txt"; String errorMsg = " from " + baseName + ".xml failed due to reason " + e; String message = switch (type) { case "C" -> "Contract ID :" + id + errorMsg; case "V" -> "Vendor ID :" + id + errorMsg; default -> "Data processing " + errorMsg; }; try { Files.writeString(Paths.get(statusFile), message); uploadFileToSFTP(statusFile, properties.getProperty("ackPath"), user, host); Files.copy(Paths.get(statusFile), Paths.get(properties.getProperty("localOutbox"), new File(statusFile).getName()), StandardCopyOption.REPLACE_EXISTING); Files.delete(Paths.get(statusFile)); logg.info("Failure acknowledgment sent for: " + baseName + ".xml"); } catch (Exception ex) { logg.severe("Error writing fail file for " + id + ": " + ex.getMessage()); } } private void uploadFileToSFTP(String localFilePath, String remotePath, String user, String host) throws Exception { ProcessBuilder pb = new ProcessBuilder("sftp", "-oPort=" + SFTP_PORT, user + "@" + host); Process process = pb.start(); try (BufferedWriter writer = new BufferedWriter(new OutputStreamWriter(process.getOutputStream())); BufferedReader outputReader = new BufferedReader(new InputStreamReader(process.getInputStream())); BufferedReader errorReader = new BufferedReader(new InputStreamReader(process.getErrorStream()))) { writer.write("put " + localFilePath + " " + remotePath + "\nexit\n"); writer.flush(); outputReader.lines().forEach(logg::info); errorReader.lines().forEach(line -> logg.severe("SFTP error: " + line)); int exitCode = process.waitFor(); logg.info("SFTP upload finished for: " + localFilePath + ", exit code: " + exitCode); } } }
