public static boolean isConfigComplete() { String[] requiredKeys = { // Git "git.repo.path.ee", "git.repo.path.ce", "git.branch.name.ee", "git.branch.name.ce", "git.executable", // Jenkins "jenkins.url", "jenkins.user", "jenkins.token", "jenkins.job.ee", "jenkins.job.ce", // Environment selection "ee.sel", "ce.sel" }; for (String key : requiredKeys) { String val = Config.get(key); if (val == null || val.trim().isEmpty()) { return false; } } // At least one environment must be enabled boolean ee = "true".equalsIgnoreCase(Config.get("ee.sel")); boolean ce = "true".equalsIgnoreCase(Config.get("ce.sel")); return ee || ce; } public static void main(String[] args) { SSLUtil.disableSSLVerification(); AppLogger.logSeparator(); AppLogger.log("Application started"); try { File repo = new File(Config.get("git.repo.path.ee")); AppLogger.log("User: " + GitCli.run(repo, "config", "user.name")); } catch (Exception e) { AppLogger.log("Git user not configured yet"); } AppLogger.logSeparator(); SwingUtilities.invokeLater(() -> { // ✅ Open config dialog ONLY if config is incomplete if (!ConfigManager.isConfigComplete()) { ConfigDialog dialog = new ConfigDialog(null); dialog.setVisible(true); // ❌ User closed dialog without saving if (!dialog.isSaved()) { AppLogger.log("Application closed: configuration not completed"); System.exit(0); } } // ✅ Config is valid → launch main UI new MainUI().setVisible(true); }); }
