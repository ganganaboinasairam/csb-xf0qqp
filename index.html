function convertStrc() { const fileList = document.getElementById("manualInput").value.trim().split("\n"); const groups = {}; fileList.forEach(path => { path = path.trim(); if (!path) return; const parts = path.split("/"); const folder = parts[parts.length - 2]; // FUNCLEVEL, SCRN, FUNCATTR const fileName = parts[parts.length - 1]; // Actual file name if (!groups[folder]) { groups[folder] = []; } groups[folder].push(`*${fileName}*`); }); let output = ""; for (const folder in groups) { output += `${folder} : ${groups[folder].join(";")}\n`; } document.getElementById("regexOutput").value = output.trim(); }
