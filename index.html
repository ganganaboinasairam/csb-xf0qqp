import java.io.*; import java.nio.file.*; import java.util.*; public class FolderReconstructor { public static void main(String[] args) throws IOException { Path localRepo = Paths.get("C:/local-repo"); Path prodFolder = Paths.get("C:/prod-dump"); Path outputFolder = Paths.get("C:/reconstructed"); Map fileMap = buildLocalFileMap(localRepo); Files.createDirectories(outputFolder); try (DirectoryStream stream = Files.newDirectoryStream(prodFolder)) { for (Path prodFile : stream) { String originalName = prodFile.getFileName().toString(); String normalizedName = normalizeFileName(originalName); Path targetRelativePath = fileMap.get(normalizedName.toLowerCase()); if (targetRelativePath != null) { Path targetPath = outputFolder.resolve(targetRelativePath); Files.createDirectories(targetPath.getParent()); Files.move(prodFile, targetPath, StandardCopyOption.REPLACE_EXISTING); } else { Path unmappedDir = outputFolder.resolve("UNMAPPED"); Files.createDirectories(unmappedDir); Files.move(prodFile, unmappedDir.resolve(normalizedName), StandardCopyOption.REPLACE_EXISTING); } } } System.out.println("Reconstruction completed."); } private static Map buildLocalFileMap(Path root) throws IOException { Map map = new HashMap<>(); Files.walk(root) .filter(Files::isRegularFile) .forEach(path -> { String fileName = path.getFileName().toString().toLowerCase(); Path relativePath = root.relativize(path); map.put(fileName, relativePath); }); return map; } private static String normalizeFileName(String fileName) { // Handle .js.txt → .js if (fileName.endsWith(".txt")) { fileName = fileName.substring(0, fileName.length() - 4); } return fileName; } } private static boolean isReleaseValidationJsp(Path file) { try { String content = Files.readString(file); return content.contains("<%@include file=\"RelVal.jsp\" %>"); } catch (IOException e) { e.printStackTrace(); } return false; } String originalName = prodFile.getFileName().toString(); String normalizedName = normalizeFileName(originalName); Path targetPath; if (normalizedName.toLowerCase().endsWith(".jsp")) { boolean isRelVal = isReleaseValidationJsp(prodFile); if (isRelVal) { targetPath = outputFolder.resolve( "base_domain/EE.ear/EximBillWeb.war/SCRN/releaseValidation/" + normalizedName ); } else { targetPath = outputFolder.resolve( "base_domain/EE.ear/EximBillWeb.war/SCRN/" + normalizedName ); } } else { Path targetRelativePath = fileMap.get(normalizedName.toLowerCase()); if (targetRelativePath != null) { targetPath = outputFolder.resolve(targetRelativePath); } else { targetPath = outputFolder.resolve("UNMAPPED/" + normalizedName); } } Files.createDirectories(targetPath.getParent()); Files.move(prodFile, targetPath, StandardCopyOption.REPLACE_EXISTING);
