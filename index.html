package com.reminder; import java.awt.*; import java.util.List; public class SystemTrayManager { private static TrayIcon trayIcon; public static void init() { if (!SystemTray.isSupported()) return; try { Image image = Toolkit.getDefaultToolkit().getImage( SystemTrayManager.class .getResource("/com/reminder/reminder.png") ); PopupMenu menu = new PopupMenu(); MenuItem open = new MenuItem("Open Tasks"); open.addActionListener(e -> { List tasks = TaskJsonStore.loadTasks(); ReminderUI.show(tasks); }); MenuItem exit = new MenuItem("Exit"); exit.addActionListener(e -> System.exit(0)); menu.add(open); menu.add(exit); trayIcon = new TrayIcon(image, "Task Reminder", menu); trayIcon.setImageAutoSize(true); SystemTray.getSystemTray().add(trayIcon); } catch (Exception e) { e.printStackTrace(); } } public static void showBubble(String title, String message) { if (trayIcon != null) { trayIcon.displayMessage( title, message, TrayIcon.MessageType.INFO ); } } } package com.reminder; import java.util.*; import java.util.concurrent.*; import java.util.stream.Collectors; public class BubbleReminderScheduler { private static int currentIndex = 0; public static void start() { ScheduledExecutorService scheduler = Executors.newSingleThreadScheduledExecutor(); scheduler.scheduleAtFixedRate(() -> { List tasks = TaskJsonStore.loadTasks(); List pendingTasks = tasks.stream() .filter(t -> "PENDING".equalsIgnoreCase(t.getStatus())) .collect(Collectors.toList()); if (pendingTasks.isEmpty()) { return; } if (currentIndex >= pendingTasks.size()) { currentIndex = 0; } Task task = pendingTasks.get(currentIndex); currentIndex++; SystemTrayManager.showBubble( "Pending Task (" + task.getPriority() + ")", task.getDescription() ); }, 0, 30, TimeUnit.MINUTES); } }
