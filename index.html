package com.company.git; import javax.net.ssl.*; import java.security.SecureRandom; import java.security.cert.X509Certificate; public final class SSLUtil { private SSLUtil() {} public static void disableSSLVerification() { try { TrustManager[] trustAllCerts = new TrustManager[]{ new X509TrustManager() { public X509Certificate[] getAcceptedIssuers() { return new X509Certificate[0]; } public void checkClientTrusted( X509Certificate[] certs, String authType) {} public void checkServerTrusted( X509Certificate[] certs, String authType) {} } }; SSLContext sc = SSLContext.getInstance("TLS"); sc.init(null, trustAllCerts, new SecureRandom()); HttpsURLConnection.setDefaultSSLSocketFactory( sc.getSocketFactory()); HttpsURLConnection.setDefaultHostnameVerifier( (hostname, session) -> true); System.out.println("SSL verification DISABLED (DEV mode)"); } catch (Exception e) { throw new RuntimeException("Failed to disable SSL", e); } } }
