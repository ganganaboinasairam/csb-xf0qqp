package com.jarupdater; import java.io.*; import java.nio.file.*; import javax.servlet.annotation.WebServlet; import javax.servlet.http.*; import com.google.gson.*; @WebServlet("/api/updateJar") public class UpdateJarServlet extends HttpServlet { protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws IOException { try { String projectId = req.getParameter("projectId"); String classFile = req.getParameter("classFile"); JsonArray arr = ConfigLoader.getConfig().getAsJsonArray("projects"); String jarPath = null, workspace = null; for (int i = 0; i < arr.size(); i++) { JsonObject o = arr.get(i).getAsJsonObject(); if (projectId.equals(o.get("id").getAsString())) { jarPath = o.get("jarPath").getAsString(); workspace = o.get("workspace").getAsString(); } } String backup = ConfigLoader.getConfig().get("backupFolder").getAsString(); System.out.println("backup folder : "+backup); Files.createDirectories(Paths.get(backup)); Path jar = Paths.get(jarPath); System.out.println("jar folder : "+jar); Files.copy(jar, Paths.get(backup, jar.getFileName().toString()), StandardCopyOption.REPLACE_EXISTING); ProcessBuilder pb = new ProcessBuilder("jar", "uf", jarPath, "-C", workspace, classFile); int rc = pb.start().waitFor(); resp.getWriter().write(rc == 0 ? "SUCCESS" : "FAILED"); } catch (Exception e) { resp.getWriter().write(e.getMessage()); } } }
