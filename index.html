const selected = Array.from( document.querySelectorAll('.add-file-check:checked') ).map(el => el.value); if (!selected.length) { showToast('Select at least one file to add', 'err'); return; } out(`[INFO] Starting commit and push for ${cmtMsg} on ${currentRegion.toUpperCase()}...`); const isAllFilesChecked = document.getElementById("isAllFilesChecked").value; try { const response = await fetch( `${API}/commit/stream?region=${currentRegion}&cr=${encodeURIComponent(cmtMsg)}&isAllFilesChecked=${isAllFilesChecked}`, { method: 'POST', headers: { 'Content-Type': 'text/plain' }, body: selected.join('\n') } ); if (!response.ok) { throw new Error(`HTTP ${response.status}`); } const reader = response.body.getReader(); const decoder = new TextDecoder(); let buffer = ''; while (true) { const { done, value } = await reader.read(); if (done) { break; } buffer += decoder.decode(value, { stream: true }); const events = buffer.split('\n\n'); buffer = events.pop(); for (const evt of events) { const lines = evt.split('\n'); let eventName = ''; let data = ''; for (const line of lines) { if (line.startsWith('event:')) { eventName = line.substring(6).trim(); } if (line.startsWith('data:')) { data = line.substring(5).trim(); } } switch (eventName) { case 'log': out(data); break; case 'progress': setProgress(parseInt(data)); break; case 'stage': setStage(data); out(`[STAGE] ${data}`, 'stage'); break; case 'done': out('[INFO] Commit pushed successfully.', 'success'); setProgress(100); setStage('DONE'); document.getElementById('commitMsg').value = cmtMsg; break; case 'noop': out('[CLEAN] ' + data, 'success'); hideProgress(); break; case 'error': out('[ERROR] ' + data, 'error'); hideProgress(); break; } } } } catch (err) { out('[ERROR] ' + err.message, 'error'); hideProgress(); }
