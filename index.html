import java.io.*; import java.nio.file.*; public class LargeFileSplitter { public static void main(String[] args) { String inputFilePath = "C:\\path\\to\\your\\largefile.txt"; // Input file path String outputDirPath = "C:\\path\\to\\output\\directory\\"; // Directory to save split files int linesPerFile = 1000; // Number of lines per split file (you can change this) try { // Create output directory if it doesn't exist Files.createDirectories(Paths.get(outputDirPath)); BufferedReader reader = new BufferedReader(new FileReader(inputFilePath)); String line; int fileCount = 1; int lineCount = 0; BufferedWriter writer = new BufferedWriter(new FileWriter(outputDirPath + "part" + fileCount + ".txt")); // Read lines from the input file and write to smaller files while ((line = reader.readLine()) != null) { if (lineCount == linesPerFile) { writer.close(); fileCount++; lineCount = 0; writer = new BufferedWriter(new FileWriter(outputDirPath + "part" + fileCount + ".txt")); } writer.write(line); writer.newLine(); lineCount++; } // Close the last writer writer.close(); reader.close(); System.out.println("File split successfully into " + fileCount + " parts."); } catch (IOException e) { System.out.println("Error: " + e.getMessage()); } } }
