import java.util.regex.Matcher; import java.util.regex.Pattern; public final class CRTagDetector { private static final Pattern START_PATTERN = Pattern.compile("(?i)\\bstart\\b\\s*[-_:]*\\s*\\bof\\b\\s*[-_:]*\\s*\\b(ir|cr)\\b\\s*[-_:]*\\s*(\\d+)"); private static final Pattern END_PATTERN = Pattern.compile("(?i)\\bend\\b\\s*[-_:]*\\s*\\bof\\b\\s*[-_:]*\\s*\\b(ir|cr)\\b\\s*[-_:]*\\s*(\\d+)"); private CRTagDetector() {} public static String extractStartId(String line) { if (line == null) return null;
