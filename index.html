package com.logviewer; import javax.swing.*; import javax.swing.text.*; import java.awt.*; import java.io.File; import java.util.List; public class Main { private JFrame frame; private JTextField searchField; private JButton openButton; private JButton nextPageButton; private JButton prevPageButton; private JButton searchButton; private JButton nextMatchButton; private JButton folderSearchButton; private JTextPane textPane; private JLabel statusLabel; private LargeFileReader reader; private static final int PAGE_SIZE = 5000; private long currentPage = 0; private boolean loading = false; public static void main(String[] args) { SwingUtilities.invokeLater(() -> { try { UIManager.setLookAndFeel( UIManager.getSystemLookAndFeelClassName() ); } catch (Exception ignored) { } new Main().start(); }); } private void start() { frame = new JFrame("Large Log Viewer"); frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE); frame.setSize(1400, 900); frame.setLocationRelativeTo(null); JPanel topPanel = new JPanel( new FlowLayout(FlowLayout.LEFT) ); openButton = new JButton("Open"); prevPageButton = new JButton("Prev Page"); nextPageButton = new JButton("Next Page"); searchField = new JTextField(30); searchButton = new JButton("Search"); nextMatchButton = new JButton("Next Match"); folderSearchButton = new JButton("Search Folder"); topPanel.add(openButton); topPanel.add(prevPageButton); topPanel.add(nextPageButton); topPanel.add(searchField); topPanel.add(searchButton); topPanel.add(nextMatchButton); topPanel.add(folderSearchButton); textPane = new JTextPane(); textPane.setEditable(false); textPane.setFont( new Font(Font.MONOSPACED, Font.PLAIN, 13) ); JScrollPane scrollPane = new JScrollPane(textPane); statusLabel = new JLabel("No file opened"); frame.add(topPanel, BorderLayout.NORTH); frame.add(scrollPane, BorderLayout.CENTER); frame.add(statusLabel, BorderLayout.SOUTH); openButton.addActionListener(e -> openFile()); nextPageButton.addActionListener( e -> nextPage() ); prevPageButton.addActionListener( e -> prevPage() ); searchButton.addActionListener( e -> startSearch() ); nextMatchButton.addActionListener( e -> nextMatch() ); folderSearchButton.addActionListener( e -> searchFolder() ); frame.setVisible(true); } private void openFile() { JFileChooser chooser = new JFileChooser(); int result = chooser.showOpenDialog(frame); if (result != JFileChooser.APPROVE_OPTION) { return; } File file = chooser.getSelectedFile(); reader = new LargeFileReader(file); currentPage = 0; loadPage(currentPage); } private void loadPage(long page) { if (loading || reader == null) { return; } loading = true; textPane.setText(""); long startLine = page * PAGE_SIZE; statusLabel.setText( "Loading page " + (page + 1) ); SwingWorker, Void> worker = new SwingWorker, Void>() { @Override protected List doInBackground() throws Exception { return reader.readLines( startLine, PAGE_SIZE ); } @Override protected void done() { try { List lines = get(); StringBuilder builder = new StringBuilder(); for (String line : lines) { builder.append(line) .append("\n"); } setHighlightedText( builder.toString(), searchField.getText() ); textPane.setCaretPosition(0); statusLabel.setText( "Page " + (currentPage + 1) + " | Lines " + lines.size() ); } catch (Exception ex) { showError(ex); } finally { loading = false; } } }; worker.execute(); } private void nextPage() { currentPage++; loadPage(currentPage); } private void prevPage() { if (currentPage == 0) { return; } currentPage--; loadPage(currentPage); } private void startSearch() { if (reader == null || loading) { return; } String keyword = searchField.getText(); if (keyword == null || keyword.trim().isEmpty()) { return; } loading = true; textPane.setText(""); statusLabel.setText("Searching..."); SwingWorker worker = new SwingWorker() { @Override protected SearchResult doInBackground() throws Exception { reader.resetSearch(); return reader.findNextMatch( keyword, 100 ); } @Override protected void done() { try { SearchResult result = get(); showSearchResult(result); } catch (Exception ex) { showError(ex); } finally { loading = false; } } }; worker.execute(); } private void nextMatch() { if (reader == null || loading) { return; } String keyword = searchField.getText(); if (keyword == null || keyword.trim().isEmpty()) { return; } loading = true; statusLabel.setText("Finding next match..."); SwingWorker worker = new SwingWorker() { @Override protected SearchResult doInBackground() throws Exception { return reader.findNextMatch( keyword, 100 ); } @Override protected void done() { try { SearchResult result = get(); showSearchResult(result); } catch (Exception ex) { showError(ex); } finally { loading = false; } } }; worker.execute(); } private void searchFolder() { JFileChooser chooser = new JFileChooser(); chooser.setFileSelectionMode( JFileChooser.DIRECTORIES_ONLY ); int result = chooser.showOpenDialog(frame); if (result != JFileChooser.APPROVE_OPTION) { return; } File folder = chooser.getSelectedFile(); String keyword = searchField.getText(); if (keyword == null || keyword.trim().isEmpty()) { return; } loading = true; textPane.setText(""); statusLabel.setText( "Searching folder..." ); SwingWorker, Void> worker = new SwingWorker, Void>() { @Override protected List doInBackground() throws Exception { FolderSearchManager manager = new FolderSearchManager(); return manager.searchFolder( folder, keyword, 100 ); } @Override protected void done() { try { List results = get(); StringBuilder builder = new StringBuilder(); for (SearchResult r : results) { builder.append( "================================================\n" ); builder.append( "FILE : " + r.getFileName() + "\n" ); builder.append( "LINE : " + r.getLineNumber() + "\n" ); builder.append( "================================================\n" ); for (String line : r.getContextLines()) { builder.append(line) .append("\n"); } builder.append("\n\n"); } setHighlightedText( builder.toString(), searchField.getText() ); textPane.setCaretPosition(0); statusLabel.setText( "Matches : " + results.size() ); } catch (Exception ex) { showError(ex); } finally { loading = false; } } }; worker.execute(); } private void showSearchResult( SearchResult result ) { if (result == null) { textPane.setText(""); statusLabel.setText( "No more matches" ); return; } StringBuilder builder = new StringBuilder(); builder.append( "================================================\n" ); builder.append( "MATCH AT LINE : " + result.getLineNumber() + "\n" ); builder.append( "================================================\n\n" ); for (String line : result.getContextLines()) { builder.append(line) .append("\n"); } setHighlightedText( builder.toString(), searchField.getText() ); textPane.setCaretPosition(0); statusLabel.setText( "Match at line " + result.getLineNumber() ); } private void setHighlightedText( String text, String keyword ) { StyledDocument doc = textPane.getStyledDocument(); Style defaultStyle = textPane.addStyle("Default", null); StyleConstants.setForeground( defaultStyle, Color.BLACK ); Style highlightStyle = textPane.addStyle("Highlight", null); StyleConstants.setBackground( highlightStyle, Color.YELLOW ); StyleConstants.setBold( highlightStyle, true ); try { doc.remove( 0, doc.getLength() ); if (keyword == null || keyword.trim().isEmpty()) { doc.insertString( 0, text, defaultStyle ); return; } int index = 0; while (true) { int match = text.indexOf(keyword, index); if (match < 0) { doc.insertString( doc.getLength(), text.substring(index), defaultStyle ); break; } doc.insertString( doc.getLength(), text.substring(index, match), defaultStyle ); doc.insertString( doc.getLength(), keyword, highlightStyle ); index = match + keyword.length(); } } catch (Exception ex) { ex.printStackTrace(); } } private void showError(Exception ex) { ex.printStackTrace(); JOptionPane.showMessageDialog( frame, ex.getMessage(), "Error", JOptionPane.ERROR_MESSAGE ); } }
