package com.reminder; import javax.swing.*; import javax.swing.table.*; import java.awt.*; import java.util.List; public class ReminderUI { public static void show(List tasks) { JDialog dialog = new JDialog((Frame) null, "Task Reminder", true); dialog.setSize(800, 450); dialog.setLayout(new BorderLayout()); String[] columns = {"ID", "Task", "Status"}; DefaultTableModel model = new DefaultTableModel(columns, 0) { @Override public boolean isCellEditable(int row, int col) { return col == 2; // only status editable } }; for (Task t : tasks) { model.addRow(new Object[]{ t.getId(), t.getDescription(), t.getStatus() }); } JTable table = new JTable(model); table.setRowHeight(40); table.setFont(new Font("Segoe UI", Font.PLAIN, 14)); table.getTableHeader().setFont( new Font("Segoe UI", Font.BOLD, 14) ); table.setSelectionMode(ListSelectionModel.SINGLE_SELECTION); /* ================= Column Sizes ================= */ TableColumnModel colModel = table.getColumnModel(); colModel.getColumn(0).setPreferredWidth(50); // ID small colModel.getColumn(1).setPreferredWidth(500); // Task large colModel.getColumn(2).setPreferredWidth(120); // Status /* ================= Task Text Wrap ================= */ colModel.getColumn(1).setCellRenderer(new TextWrapRenderer()); /* ================= Status Dropdown ================= */ JComboBox statusBox = new JComboBox<>(new String[]{"PENDING", "DONE"}); colModel.getColumn(2).setCellEditor(new DefaultCellEditor(statusBox)); colModel.getColumn(2).setCellRenderer(new StatusColorRenderer()); JScrollPane scrollPane = new JScrollPane(table); /* ================= Buttons ================= */ JButton addBtn = new JButton("Add Task"); JButton editBtn = new JButton("Edit Task"); JButton deleteBtn = new JButton("Delete Task"); JButton saveBtn = new JButton("Save & Close"); addBtn.addActionListener(e -> { String desc = JOptionPane.showInputDialog( dialog, "Enter task description:" ); if (desc != null && !desc.trim().isEmpty()) { model.addRow(new Object[]{ getNextId(model), desc.trim(), "PENDING" }); } }); editBtn.addActionListener(e -> { int row = table.getSelectedRow(); if (row == -1) { showMsg(dialog, "Please select a task to edit"); return; } String current = model.getValueAt(row, 1).toString(); String updated = JOptionPane.showInputDialog( dialog, "Edit task description:", current ); if (updated != null && !updated.trim().isEmpty()) { model.setValueAt(updated.trim(), row, 1); } }); deleteBtn.addActionListener(e -> { int row = table.getSelectedRow(); if (row == -1) { showMsg(dialog, "Please select a task to delete"); return; } if (JOptionPane.showConfirmDialog( dialog, "Delete selected task?", "Confirm", JOptionPane.YES_NO_OPTION ) == JOptionPane.YES_OPTION) { model.removeRow(row); } }); saveBtn.addActionListener(e -> { tasks.clear(); for (int i = 0; i < model.getRowCount(); i++) { tasks.add(new Task( Integer.parseInt(model.getValueAt(i, 0).toString()), model.getValueAt(i, 1).toString(), model.getValueAt(i, 2).toString() )); } TaskJsonStore.saveTasks(tasks); dialog.dispose(); }); JPanel btnPanel = new JPanel(new FlowLayout(FlowLayout.RIGHT)); btnPanel.add(addBtn); btnPanel.add(editBtn); btnPanel.add(deleteBtn); btnPanel.add(saveBtn); dialog.add(scrollPane, BorderLayout.CENTER); dialog.add(btnPanel, BorderLayout.SOUTH); dialog.setLocationRelativeTo(null); dialog.setVisible(true); } /* ================= Helper Methods ================= */ private static int getNextId(DefaultTableModel model) { int max = 0; for (int i = 0; i < model.getRowCount(); i++) { max = Math.max(max, Integer.parseInt(model.getValueAt(i, 0).toString())); } return max + 1; } private static void showMsg(Component parent, String msg) { JOptionPane.showMessageDialog(parent, msg); } /* ================= Custom Renderers ================= */ /** Wraps long task text */ static class TextWrapRenderer extends JTextArea implements TableCellRenderer { public TextWrapRenderer() { setLineWrap(true); setWrapStyleWord(true); setOpaque(true); setFont(new Font("Segoe UI", Font.PLAIN, 14)); } @Override public Component getTableCellRendererComponent( JTable table, Object value, boolean isSelected, boolean hasFocus, int row, int column) { setText(value == null ? "" : value.toString()); if (isSelected) { setBackground(table.getSelectionBackground()); setForeground(table.getSelectionForeground()); } else { setBackground(Color.WHITE); setForeground(Color.BLACK); } setSize(table.getColumnModel().getColumn(column).getWidth(), getPreferredSize().height); if (table.getRowHeight(row) != getPreferredSize().height) { table.setRowHeight(row, getPreferredSize().height); } return this; } } /** Colors status cells */ static class StatusColorRenderer extends DefaultTableCellRenderer { @Override public Component getTableCellRendererComponent( JTable table, Object value, boolean isSelected, boolean hasFocus, int row, int column) { super.getTableCellRendererComponent( table, value, isSelected, hasFocus, row, column); setFont(new Font("Segoe UI", Font.BOLD, 13)); setHorizontalAlignment(CENTER); if ("DONE".equalsIgnoreCase(String.valueOf(value))) { setBackground(new Color(144, 238, 144)); // green } else { setBackground(new Color(255, 255, 153)); // yellow } if (isSelected) { setBackground(table.getSelectionBackground()); } return this; } } }
