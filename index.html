function changeFieldClass(inputId, param) { // Get the input field by ID const inputField = document.getElementById(inputId); if (!inputField) { console.error(`No input field found with ID: ${inputId}`); return; } // Get the parent of the input field const parentTd = inputField.closest("td"); if (!parentTd) { console.error(`No parent found for input field with ID: ${inputId}`); return; } // Get the previous sibling const previousTd = parentTd.previousElementSibling; if (!previousTd) { console.error(`No previous found for input field with ID: ${inputId}`); return; } // Update the input field and related based on the param if (param === "M") { inputField.className = "CHAR_M"; previousTd.className = "FldLabel star"; inputField.readOnly = false; // Ensure it is not readonly } else if (param === "O") { inputField.className = "CHAR_O"; previousTd.className = "FldLabel"; inputField.readOnly = false; // Ensure it is not readonly } else if (param === "P") { inputField.className = "CHAR_P"; previousTd.className = "FldLabel"; inputField.readOnly = true; // Set readonly } else { console.error(`Invalid parameter value: ${param}`); } }
