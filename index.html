package com.reminder; public class Task { private int id; private String description; private String status; private boolean dailyReset; public Task(int id, String description, String status) { this(id, description, status, false); } public Task(int id, String description, String status, boolean dailyReset) { this.id = id; this.description = description; this.status = status; this.dailyReset = dailyReset; } public int getId() { return id; } public String getDescription() { return description; } public String getStatus() { return status; } public boolean isDailyReset() { return dailyReset; } public void setDescription(String description) { this.description = description; } public void setStatus(String status) { this.status = status; } public void setDailyReset(boolean dailyReset) { this.dailyReset = dailyReset; } } package com.reminder; import com.google.gson.*; import java.io.*; import java.time.LocalDate; import java.util.*; public class TaskJsonStore { private static final String FILE = "tasks.json"; private static final Gson gson = new GsonBuilder().setPrettyPrinting().create(); public static List loadTasks() { try { File file = new File(FILE); if (!file.exists()) { saveAll(new ArrayList<>(), LocalDate.now()); return new ArrayList<>(); } JsonObject root = JsonParser .parseReader(new FileReader(file)) .getAsJsonObject(); LocalDate lastReset = LocalDate.parse(root.get("lastResetDate").getAsString()); LocalDate today = LocalDate.now(); Task[] tasks = gson.fromJson( root.getAsJsonArray("tasks"), Task[].class ); List taskList = new ArrayList<>(Arrays.asList(tasks)); // === Daily Reset (ONLY ONCE PER DAY) === if (!today.equals(lastReset)) { for (Task t : taskList) { if (t.isDailyReset() && "DONE".equalsIgnoreCase(t.getStatus())) { t.setStatus("PENDING"); } } saveAll(taskList, today); } return taskList; } catch (Exception e) { e.printStackTrace(); return new ArrayList<>(); } } public static void saveTasks(List tasks) { saveAll(tasks, LocalDate.now()); } private static void saveAll(List tasks, LocalDate resetDate) { try (Writer writer = new FileWriter(FILE)) { JsonObject root = new JsonObject(); root.addProperty("lastResetDate", resetDate.toString()); root.add("tasks", gson.toJsonTree(tasks)); gson.toJson(root, writer); } catch (Exception e) { e.printStackTrace(); } } }
