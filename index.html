// Command history private java.util.List cmdHistory = new java.util.ArrayList<>(); private int historyIndex = -1; private void setupCommandHistoryKeys() { InputMap im = customCmdField.getInputMap(JComponent.WHEN_FOCUSED); ActionMap am = customCmdField.getActionMap(); // UP arrow → previous command im.put(KeyStroke.getKeyStroke("UP"), "historyUp"); am.put("historyUp", new AbstractAction() { @Override public void actionPerformed(java.awt.event.ActionEvent e) { if (cmdHistory.isEmpty()) return; if (historyIndex <= 0) { historyIndex = 0; } else { historyIndex--; } customCmdField.setText(cmdHistory.get(historyIndex)); } }); // DOWN arrow → next command im.put(KeyStroke.getKeyStroke("DOWN"), "historyDown"); am.put("historyDown", new AbstractAction() { @Override public void actionPerformed(java.awt.event.ActionEvent e) { if (cmdHistory.isEmpty()) return; if (historyIndex >= cmdHistory.size() - 1) { historyIndex = cmdHistory.size(); customCmdField.setText(""); } else { historyIndex++; customCmdField.setText(cmdHistory.get(historyIndex)); } } }); } private void customCmd() { String cmd = customCmdField.getText().trim(); if (cmd.isEmpty()) return; // Store in history (avoid duplicates back-to-back) if (cmdHistory.isEmpty() || !cmd.equals(cmdHistory.get(cmdHistory.size() - 1))) { cmdHistory.add(cmd); } historyIndex = cmdHistory.size(); // reset pointer
