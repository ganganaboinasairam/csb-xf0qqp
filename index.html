function formatAmountFields(fields) { fields.split(";").forEach(function(field) { field = field.trim(); if (!field) return; var input = document.getElementById(field) || document.MAINFORM[field]; if (!input) return; var value = String(input.value || "").replace(/,/g, "").trim(); if (value !== "" && !isNaN(value)) { input.value = Number(value).toFixed(2); } }); }
