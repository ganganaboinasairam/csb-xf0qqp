function reArrangeTextArea(firstFld, secFld, edtbtn) { var $div1 = $('#' + firstFld).parent(); $div1.wrap('
'); var $parent = $('#' + firstFld + '_parent'); $parent.css({ display: 'flex', gap: '30px' }); $('#' + secFld).parent().next().remove(); $parent.append($('#' + secFld).parent()); var $titleTD = $parent.prev(); $titleTD.wrap('
'); var $titleWrapper = $('#' + firstFld + '_title'); $titleWrapper.css('margin-top', '20px'); $titleWrapper.append($('#' + edtbtn)); $('#' + secFld).parent().hide(); $('#' + edtbtn).on('click', function () { $('#' + secFld).parent().show(); }); }
