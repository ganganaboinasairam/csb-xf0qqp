private JButton retrieveFilesBtn; retrieveFilesBtn = new JButton("Retrieve Files"); gitPanel.add(retrieveFilesBtn, c); retrieveFilesBtn.addActionListener(e -> retrieveFiles()); private void retrieveFiles() { String ref = crField.getText().trim(); String region = regionBox.getSelectedItem().toString().toLowerCase(); if (!ref.matches("(CR|IR)_\\d+.*")) { JOptionPane.showMessageDialog( this, "Enter valid CR or IR (example: CR_12345)", "Invalid Input", JOptionPane.ERROR_MESSAGE ); return; } outputArea.append("\n=================================================\n"); outputArea.append("FILES FOR " + ref + " | " + region.toUpperCase()); outputArea.append("\n=================================================\n"); SwingWorker worker = new SwingWorker() { @Override protected Void doInBackground() throws Exception { GitAutomationService.retrieveFiles(region, ref, msg -> publish(msg)); return null; } @Override protected void process(java.util.List chunks) { for (String s : chunks) { outputArea.append(s + "\n"); } } @Override protected void done() { outputArea.append("\n==================== DONE ====================\n"); } }; worker.execute(); } public static void retrieveFiles( String region, String crOrIr, java.util.function.Consumer log) throws Exception { File repo = new File(Config.get("git.repo.path." + region)); String git = Config.get("git.executable"); log.accept("Repository : " + repo.getAbsolutePath()); log.accept("Searching commits for : " + crOrIr); log.accept(""); // 1. Find commit hashes String commits = GitCli.run( repo, git, "log", "--grep=" + crOrIr, "--pretty=format:%H" ); if (commits.trim().isEmpty()) { log.accept("No commits found for " + crOrIr); return; } Set files = new TreeSet<>(); for (String hash : commits.split("\\R")) { String out = GitCli.run( repo, git, "show", "--pretty=", "--name-only", hash.trim() ); for (String f : out.split("\\R")) { if (!f.trim().isEmpty()) { files.add(f.trim()); } } } if (files.isEmpty()) { log.accept("No files found."); return; } log.accept("Files committed under " + crOrIr + ":"); log.accept("----------------------------------------------"); for (String f : files) { log.accept(f); } }
