import java.nio.file.*; import java.nio.file.attribute.BasicFileAttributes; import java.util.*; import java.io.IOException; public class IRDisableProcessor { private final Path source; private final Path output; private final Set disabledIRs; private final String tag; public IRDisableProcessor(Path source, Path output, Set disabledIRs, String tag) { this.source = source; this.output = output; this.disabledIRs = disabledIRs; this.tag = tag; } public void process() throws IOException { Files.walkFileTree(source, new SimpleFileVisitor() { @Override public FileVisitResult visitFile(Path file, BasicFileAttributes attrs) throws IOException { if (!attrs.isRegularFile()) { return FileVisitResult.CONTINUE; } FileType type = FileTypeResolver.resolve(file); Path rel = source.relativize(file); Path outFile = output.resolve(rel); Files.createDirectories(outFile.getParent()); List lines = Files.readAllLines(file); List out = new ArrayList<>(); List buffer = new ArrayList<>(); IRContext ctx = new IRContext(disabledIRs); boolean insideJsBlockComment = false; for (String line : lines) { String trimmed = line.trim(); /* ---------------- JS block comment handling ---------------- */ if (type == FileType.JS) { if (trimmed.startsWith("/*")) { insideJsBlockComment = true; } if (insideJsBlockComment) { out.add(line); if (trimmed.endsWith("*/")) { insideJsBlockComment = false; } continue; } } /* ---------------- CR/IR boundary handling ---------------- */ String start = CRTagDetector.extractStartId(line); if (start != null) { flush(buffer, out, type); ctx.enter(start); out.add(line); continue; } String end = CRTagDetector.extractEndId(line); if (end != null) { flush(buffer, out, type); out.add(line); ctx.exit(end); continue; } /* ---------------- Disable logic ---------------- */ if (ctx.isInsideDisabled() && !CommentUtil.isBlank(line) && !CRTagDetector.isAnyCRTag(line) && !CommentUtil.isAlreadyCommented(line, type) && !line.contains("START OF " + tag)) { buffer.add(line); continue; } flush(buffer, out, type); out.add(line); } flush(buffer, out, type); Files.write(outFile, out); return FileVisitResult.CONTINUE; } }); } private void flush(List buffer, List out, FileType type) { if (buffer.isEmpty()) return; String indent = CommentUtil.indent(buffer.get(0)); out.add(CommentUtil.startMarker(tag, indent, type)); if (buffer.size() == 1) { out.add(CommentUtil.commentLine(buffer.get(0).trim(), type)); } else { if (type == FileType.JS) { out.add(indent + "/*"); for (String l : buffer) { out.add(l); } out.add(indent + "*/"); } else { for (String l : buffer) { out.add(CommentUtil.commentLine(l.trim(), type)); } } } out.add(CommentUtil.endMarker(tag, indent, type)); buffer.clear(); } }
