public static void chkStatus( String region, java.util.function.Consumer log) throws Exception { File repo = new File( Config.get("git.repo.path." + region)); log.accept("Repository: " + repo.getAbsolutePath()); log.accept("Checking status...\n"); String status = GitCli.run(repo, "status", "--porcelain"); if (status.trim().isEmpty()) { log.accept("No changes found. Working tree clean."); return; } log.accept("Files to be committed:\n"); String[] lines = status.split("\\r?\\n"); for (String line : lines) { if (line.length() < 3) continue; String flag = line.substring(0, 2).trim(); String file = line.substring(3); if ("??".equals(flag)) { log.accept("[NEW] " + file); } else if (flag.contains("M")) { log.accept("[MODIFIED] " + file); } else if (flag.contains("A")) { log.accept("[ADDED] " + file); } else if (flag.contains("D")) { log.accept("[DELETED] " + file); } else { log.accept("[CHANGED] " + file); } } }
