async function copyCRFiles(crId) { const cr = data.crs.find( c => c.id === crId ); if (!cr) { return; } const files = []; cr.functions.forEach(fn => { fn.files.forEach(file => { files.push(file.path); }); }); await navigator.clipboard.writeText( files.join("\n") ); } async function copyFunctionFiles( crId, functionName ) { const cr = data.crs.find( c => c.id === crId ); if (!cr) { return; } const fn = cr.functions.find( f => f.name === functionName ); if (!fn) { return; } const files = fn.files.map( file => file.path ); await navigator.clipboard.writeText( files.join("\n") ); }
