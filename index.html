package com.reminder; public class Task { private int id; private String description; private String status; public Task(int id, String description, String status) { this.id = id; this.description = description; this.status = status; } public int getId() { return id; } public String getDescription() { return description; } public void setDescription(String description) { this.description = description; } public String getStatus() { return status; } public void setStatus(String status) { this.status = status; } } package com.reminder; import javax.swing.*; import javax.swing.table.DefaultTableModel; import java.awt.*; import java.util.List; public class ReminderUI { public static void show(List tasks) { JDialog dialog = new JDialog((Frame) null, "Task Reminder", true); dialog.setSize(700, 400); dialog.setLayout(new BorderLayout()); String[] columns = {"ID", "Task", "Status"}; DefaultTableModel model = new DefaultTableModel(columns, 0) { @Override public boolean isCellEditable(int row, int col) { return col == 2; } }; for (Task t : tasks) { model.addRow(new Object[]{ t.getId(), t.getDescription(), t.getStatus() }); } JTable table = new JTable(model); table.setSelectionMode(ListSelectionModel.SINGLE_SELECTION); JComboBox statusBox = new JComboBox<>(new String[]{"PENDING", "DONE"}); table.getColumnModel() .getColumn(2) .setCellEditor(new DefaultCellEditor(statusBox)); JScrollPane scrollPane = new JScrollPane(table); /* ================= Buttons ================= */ JButton addBtn = new JButton("Add Task"); JButton editBtn = new JButton("Edit Task"); JButton deleteBtn = new JButton("Delete Task"); JButton saveBtn = new JButton("Save & Close"); /* ---------- Add Task ---------- */ addBtn.addActionListener(e -> { String desc = JOptionPane.showInputDialog( dialog, "Enter task description:" ); if (desc != null && !desc.trim().isEmpty()) { int newId = getNextId(model); model.addRow(new Object[]{ newId, desc.trim(), "PENDING" }); } }); /* ---------- Edit Task ---------- */ editBtn.addActionListener(e -> { int row = table.getSelectedRow(); if (row == -1) { showMsg(dialog, "Please select a task to edit"); return; } String current = model.getValueAt(row, 1).toString(); String updated = JOptionPane.showInputDialog( dialog, "Edit task description:", current ); if (updated != null && !updated.trim().isEmpty()) { model.setValueAt(updated.trim(), row, 1); } }); /* ---------- Delete Task ---------- */ deleteBtn.addActionListener(e -> { int row = table.getSelectedRow(); if (row == -1) { showMsg(dialog, "Please select a task to delete"); return; } int confirm = JOptionPane.showConfirmDialog( dialog, "Delete selected task?", "Confirm", JOptionPane.YES_NO_OPTION ); if (confirm == JOptionPane.YES_OPTION) { model.removeRow(row); } }); /* ---------- Save ---------- */ saveBtn.addActionListener(e -> { tasks.clear(); for (int i = 0; i < model.getRowCount(); i++) { tasks.add(new Task( Integer.parseInt(model.getValueAt(i, 0).toString()), model.getValueAt(i, 1).toString(), model.getValueAt(i, 2).toString() )); } TaskJsonStore.saveTasks(tasks); dialog.dispose(); }); JPanel btnPanel = new JPanel(new FlowLayout(FlowLayout.RIGHT)); btnPanel.add(addBtn); btnPanel.add(editBtn); btnPanel.add(deleteBtn); btnPanel.add(saveBtn); dialog.add(scrollPane, BorderLayout.CENTER); dialog.add(btnPanel, BorderLayout.SOUTH); dialog.setLocationRelativeTo(null); dialog.setVisible(true); } /* ================= Helpers ================= */ private static int getNextId(DefaultTableModel model) { int max = 0; for (int i = 0; i < model.getRowCount(); i++) { max = Math.max(max, Integer.parseInt(model.getValueAt(i, 0).toString())); } return max + 1; } private static void showMsg(Component parent, String msg) { JOptionPane.showMessageDialog(parent, msg); } }
