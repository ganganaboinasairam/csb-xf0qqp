package com.reminder; public class Task { private int id; private String description; private String status; public Task(int id, String description, String status) { this.id = id; this.description = description; this.status = status; } public int getId() { return id; } public String getDescription() { return description; } public String getStatus() { return status; } public void setStatus(String status) { this.status = status; } } package com.reminder; import com.google.gson.*; import java.io.*; import java.util.*; public class TaskJsonStore { private static final String FILE_NAME = "tasks.json"; private static final Gson gson = new GsonBuilder().setPrettyPrinting().create(); public static List loadTasks() { try { File file = new File(FILE_NAME); if (!file.exists()) { saveTasks(new ArrayList<>()); return new ArrayList<>(); } JsonObject root = JsonParser .parseReader(new FileReader(file)) .getAsJsonObject(); Task[] tasks = gson.fromJson( root.getAsJsonArray("tasks"), Task[].class ); return new ArrayList<>(Arrays.asList(tasks)); } catch (Exception e) { e.printStackTrace(); return new ArrayList<>(); } } public static void saveTasks(List tasks) { try (Writer writer = new FileWriter(FILE_NAME)) { JsonObject root = new JsonObject(); data[i][2] = t.getStatus(); } DefaultTableModel model = new DefaultTableModel(data, columns) { @Override public boolean isCellEditable(int row, int col) { return col == 2; } }; JTable table = new JTable(model); saveBtn.addActionListener(e -> { for (int i = 0; i < tasks.size(); i++) { tasks.get(i).setStatus( model.getValueAt(i, 2).toString() ); } TaskJsonStore.saveTasks(tasks); dialog.dispose(); }); dialog.add(new JScrollPane(table), BorderLayout.CENTER); dialog.add(saveBtn, BorderLayout.SOUTH); dialog.setLocationRelativeTo(null); dialog.setVisible(true); } } package com.reminder; import java.awt.*; import java.util.List; public class SystemTrayManager { public static void init() { if (!SystemTray.isSupported()) { return;
