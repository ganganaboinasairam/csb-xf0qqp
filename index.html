function generateFlow(){ const xml=document.getElementById("xmlInput").value; const parser=new DOMParser(); const xmlDoc=parser.parseFromString(xml,"text/xml"); const rows=xmlDoc.getElementsByTagName("ROW"); let nodes=new Map(); let edges=new Set(); for(let row of rows){ let step=getColumn(row,"C_STEP_NAME"); let funcId=getColumn(row,"C_FUNC_ID"); let funcDesc=getColumn(row,"C_FUNC_DESC"); let toStep=getColumn(row,"C_TO_STEP_NAME"); let toFuncId=getColumn(row,"C_TO_FUNC_ID"); let toFuncDesc=getColumn(row,"C_TO_FUNC_DESC"); let fromNode=nodeId(step,funcId); let toNode=nodeId(toStep,toFuncId); nodes.set(fromNode, `${fromNode}["${step}
${funcId}
${funcDesc}"]:::nodeStyle`); if(toStep!="End"){ nodes.set(toNode, `${toNode}["${toStep}
${toFuncId}
${toFuncDesc}"]:::nodeStyle`); } if(toStep=="End"){ edges.add(`${fromNode} --> End`); }else{ edges.add(`${fromNode} --> ${toNode}`); } } let mermaidText="graph TD\n"; nodes.forEach(n=>mermaidText+=n+"\n"); edges.forEach(e=>mermaidText+=e+"\n"); mermaidText+=` classDef nodeStyle fill:#ffffff,stroke:#4a6cf7,stroke-width:2px,color:#222,font-size:14px; `; document.getElementById("chart").innerHTML= `
${mermaidText}
`; mermaid.init(undefined, document.querySelectorAll(".mermaid")); }
