private boolean validateFields() { if (isEmpty(gitRepoEE)) return error("Git Repo Path (EE) is required"); if (isEmpty(gitRepoCE)) return error("Git Repo Path (CE) is required"); if (isEmpty(gitBranchEE)) return error("Git Branch (EE) is required"); if (isEmpty(gitBranchCE)) return error("Git Branch (CE) is required"); if (isEmpty(gitExe)) return error("Git Executable path is required"); if (isEmpty(jenkinsUrl)) return error("Jenkins URL is required"); if (isEmpty(jenkinsUser)) return error("Jenkins User is required"); if (jenkinsToken.getPassword().length == 0) return error("Jenkins Token is required"); if (isEmpty(jenkinsJobEE)) return error("Jenkins Job (EE) is required"); if (isEmpty(jenkinsJobCE)) return error("Jenkins Job (CE) is required"); return true; } private boolean isEmpty(JTextField tf) { return tf.getText() == null || tf.getText().trim().isEmpty(); } private boolean error(String msg) { JOptionPane.showMessageDialog( this, msg, "Validation Error", JOptionPane.ERROR_MESSAGE); return false; } private void onSave() { if (!validateFields()) { return; // ⛔ block save } try { Properties p = new Properties(); p.setProperty("git.repo.path.ee", gitRepoEE.getText().trim()); p.setProperty("git.repo.path.ce", gitRepoCE.getText().trim()); p.setProperty("git.branch.name.ee", gitBranchEE.getText().trim()); p.setProperty("git.branch.name.ce", gitBranchCE.getText().trim()); p.setProperty("git.executable", gitExe.getText().trim()); p.setProperty("jenkins.url", jenkinsUrl.getText().trim()); p.setProperty("jenkins.user", jenkinsUser.getText().trim()); p.setProperty("jenkins.token", new String(jenkinsToken.getPassword()).trim()); p.setProperty("jenkins.job.ee", jenkinsJobEE.getText().trim()); p.setProperty("jenkins.job.ce", jenkinsJobCE.getText().trim()); ConfigManager.save(p); saved = true; dispose(); } catch (Exception ex) { JOptionPane.showMessageDialog( this, "Failed to save configuration\n" + ex.getMessage(), "Error", JOptionPane.ERROR_MESSAGE ); } } public static void main(String[] args) { SSLUtil.disableSSLVerification(); SwingUtilities.invokeLater(() -> { ConfigDialog dialog = new ConfigDialog(null); dialog.setVisible(true); // ⛔ DO NOT proceed unless config is saved if (!dialog.isSaved()) { System.exit(0); } new MainUI().setVisible(true); }); }
