package com.cs.ceweb.servlets; import java.io.File; import java.io.FileOutputStream; import java.io.IOException; import java.io.ObjectOutputStream; import java.io.PrintWriter; import java.sql.Connection; import java.sql.PreparedStatement; import java.sql.ResultSet; import javax.servlet.ServletException; import javax.servlet.http.HttpServlet; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletResponse; import org.w3c.dom.Node; import com.cs.base.utility.ASPathConst; import com.cs.base.xml.XMLManager; import com.cs.ce.core.helper.CESecuFileHelper; import com.cs.core.dao.DSManager; import com.cs.core.utility.WSZipUtil; public class ImageDownloadServlet extends HttpServlet { private static final long serialVersionUID = 1L; @Override protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException { String indexList = request.getParameter("indexList"); String cMainRef = request.getParameter("cMainRef"); String reqType = request.getParameter("reqType"); String imgIndex = request.getParameter("imgIndex"); try (Connection conn = DSManager.getConnection("CET")) { if ("download".equals(reqType)) { handleDownloadRequest(indexList, cMainRef, conn, response); } else if ("viewDoc".equals(reqType)) { handleViewDocRequest(imgIndex, cMainRef, conn, response); } } catch (Exception e) { e.printStackTrace(); if (!response.isCommitted()) { response.sendError(HttpServletResponse.SC_INTERNAL_SERVER_ERROR, "An error occurred while processing the request."); } } } private void handleDownloadRequest(String indexList, String cMainRef, Connection conn, HttpServletResponse response) throws Exception { String[] array = indexList.split(";"); boolean success = false; for (String value : array) { try (PreparedStatement pstmt = conn.prepareStatement( "SELECT B_IMG_CONTENT, C_IMG_FILE_TYPE, C_IMG_DOC_TYPE FROM TRX_IMAGES_DTAL WHERE C_IMG_INDX = ? AND C_MAIN_REF = ?")) { pstmt.setString(1, value); pstmt.setString(2, cMainRef); try (ResultSet rs = pstmt.executeQuery()) { if (rs.next()) { processRecord(rs, value, response); success = true; } } } } if (!response.isCommitted() && success) { response.setContentType("application/json"); response.getWriter().write("{\"message\":\"Files Saved Successfully\"}"); } else if (!success && !response.isCommitted()) { response.sendError(HttpServletResponse.SC_NOT_FOUND, "No matching records found."); } } private void handleViewDocRequest(String imgIndex, String cMainRef, Connection conn, HttpServletResponse response) throws Exception { try (PreparedStatement pstmt = conn.prepareStatement( "SELECT B_IMG_CONTENT, C_IMG_FILE_TYPE, C_IMG_DOC_TYPE FROM TRX_IMAGES_DTAL WHERE C_IMG_INDX = ? AND C_MAIN_REF = ?")) { pstmt.setString(1, imgIndex); pstmt.setString(2, cMainRef); try (ResultSet rs = pstmt.executeQuery()) { if (rs.next()) { processRecord(rs, imgIndex, response); } else if (!response.isCommitted()) { response.sendError(HttpServletResponse.SC_NOT_FOUND, "Document not found."); } } } } private void processRecord(ResultSet rs, String index, HttpServletResponse response) throws Exception { String docType = rs.getString("C_IMG_DOC_TYPE"); String base64 = rs.getString("B_IMG_CONTENT"); String fileType = rs.getString("C_IMG_FILE_TYPE"); try { if ("form".equals(docType)) { processFileResponse(base64, fileType, response); } else { Object obj = WSZipUtil.decompressByBase64(base64); printContentInTheFile((String) obj, index, fileType); } } catch (Exception e) { e.printStackTrace(); if (!response.isCommitted()) { response.sendError(HttpServletResponse.SC_INTERNAL_SERVER_ERROR, "Error processing file."); } } } private void printContentInTheFile(String obj, String imgIndex, String fileType) throws Exception { String configFilePath = ASPathConst.getUserDirPath() + File.separator + "PARAM" + File.separator + "Default" + File.separator + "SYST" + File.separator + "sysparam.xml"; Node sImagePathNode = XMLManager.findNode(XMLManager.xmlFileToDom(configFilePath).getDocumentElement(), "WEB_IMAGE_ROOT_PATH"); String imagePath = XMLManager.getNodeAttribute(sImagePathNode, "VALUE"); File file = new File(imagePath + "/" + imgIndex + "." + fileType); try (FileOutputStream out = CESecuFileHelper.newFileOutputStream(file)) { byte[] data = obj.getBytes("ISO-8859-1"); out.write(data); } } private void processFileResponse(String base64, String fileType, HttpServletResponse response) throws IOException { try { Object obj = WSZipUtil.decompressByBase64(base64); response.setContentType(getContentType(fileType)); response.setHeader("Pragma", "No-cache"); response.setHeader("Cache-Control", "no-cache"); response.setDateHeader("Expires", 2002L); try (ObjectOutputStream out = new ObjectOutputStream(response.getOutputStream())) { out.writeObject(obj); } } catch (IllegalArgumentException e) { e.printStackTrace(); if (!response.isCommitted()) { response.sendError(HttpServletResponse.SC_BAD_REQUEST, "Invalid base64 content."); } } catch (Exception e) { e.printStackTrace(); if (!response.isCommitted()) { response.sendError(HttpServletResponse.SC_INTERNAL_SERVER_ERROR, "An error occurred while processing the file."); } } } private String getContentType(String fileType) { switch (fileType.toUpperCase()) { case "PDF": return "application/pdf"; case "TXT": return "text/plain"; case "XLS": case "XLSX": return "application/vnd.ms-excel"; case "JPG": case "JPEG": return "image/jpeg"; default: return "application/octet-stream"; } } }
