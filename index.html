package com.chat; import java.io.IOException; import java.io.PrintWriter; import javax.servlet.ServletException; import javax.servlet.annotation.WebServlet; import javax.servlet.http.HttpServlet; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletResponse; import com.google.gson.JsonObject; @WebServlet("/getJwtToken") public class JwtTokenServlet extends HttpServlet { @Override protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException { // For simplicity, assume authentication is done elsewhere, and we generate token based on customerId // In production, validate credentials here String customerId = request.getParameter("customerId"); String branchCode = request.getParameter("branchCode"); if (customerId == null || branchCode == null) { response.setStatus(HttpServletResponse.SC_BAD_REQUEST); response.getWriter().write("{\"error\":\"Missing customerId or branchCode\"}"); return; } // Generate JWT token with subject as customerId-branchCode String subject = customerId + "-" + branchCode; String token = JwtUtil.generateToken(subject); // Return token as JSON response.setContentType("application/json"); PrintWriter out = response.getWriter(); JsonObject json = new JsonObject(); json.addProperty("token", token); out.print(json.toString()); out.flush(); } } package com.chat; import io.jsonwebtoken.Claims; import io.jsonwebtoken.Jwts; import io.jsonwebtoken.SignatureAlgorithm; import io.jsonwebtoken.security.Keys; import java.security.Key; import java.util.Date; public class JwtUtil { // Secret key for signing tokens (in production, use environment variable or config) private static final Key SECRET_KEY = Keys.secretKeyFor(SignatureAlgorithm.HS256); // Token validity in milliseconds (e.g., 1 hour) private static final long TOKEN_VALIDITY = 3600000; // Generate JWT token for a given subject (e.g., customerId) public static String generateToken(String subject) { long nowMillis = System.currentTimeMillis(); Date now = new Date(nowMillis); Date expiryDate = new Date(nowMillis + TOKEN_VALIDITY); return Jwts.builder() .setSubject(subject) .setIssuedAt(now) .setExpiration(expiryDate) .signWith(SECRET_KEY) .compact(); } // Validate JWT token and return claims if valid, else null public static Claims validateToken(String token) { try { return Jwts.parserBuilder() .setSigningKey(SECRET_KEY) .build() .parseClaimsJws(token) .getBody(); } catch (Exception e) { System.err.println("Invalid JWT token: " + e.getMessage()); return null; } } }
