private void pushWithRebaseRetry(Git git) throws Exception { String remote = Config.get("git.remote.name"); String branch = Config.get("git.branch.name"); System.out.println("Pushing to " + remote + "/" + branch); Iterable results = git.push() .setRemote(remote) .setRefSpecs(new RefSpec( "refs/heads/" + branch + ":refs/heads/" + branch)) .setCredentialsProvider(credentials()) .call(); for (PushResult result : results) { for (RemoteRefUpdate update : result.getRemoteUpdates()) { RemoteRefUpdate.Status status = update.getStatus(); System.out.println("Push status: " + status); if (status == RemoteRefUpdate.Status.OK || status == RemoteRefUpdate.Status.UP_TO_DATE) { return; // success } if (status == RemoteRefUpdate.Status.REJECTED_NONFASTFORWARD) { System.out.println( "NONFASTFORWARD detected. Rebasing and retrying..."); PullResult pull = git.pull() .setRebase(true) .setCredentialsProvider(credentials()) .call(); if (!pull.isSuccessful()) { throw new RuntimeException( "Rebase failed. Resolve conflicts manually."); } // retry push ONCE pushWithRebaseRetry(git); return; } throw new RuntimeException( "Push failed: " + status + " " + update.getMessage()); } } }
