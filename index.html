import java.util.regex.*; public class Main { public static void main(String[] args) { String testContent = "1000.50USD"; String[] result = getForeignAmountAndCurrency(testContent); System.out.println("Foreign Amount: " + result[0]); // Output: 1000.50 System.out.println("Foreign Currency: " + result[1]); // Output: USD } private static String[] getForeignAmountAndCurrency(String t_Content) { String foreignAmount = ""; String foreignCurrency = ""; if (t_Content == null || t_Content.isEmpty()) { return new String[]{foreignAmount, foreignCurrency}; } try { // Regex patterns to extract content Pattern amtPattern = Pattern.compile("(.*?)"); Pattern ccyPattern = Pattern.compile("(.*?)"); Matcher amtMatcher = amtPattern.matcher(t_Content); Matcher ccyMatcher = ccyPattern.matcher(t_Content); // Extract the values if found if (amtMatcher.find()) { foreignAmount = amtMatcher.group(1); } if (ccyMatcher.find()) { foreignCurrency = ccyMatcher.group(1); } } catch (Exception e) { e.printStackTrace(); } return new String[]{foreignAmount, foreignCurrency}; } }
