function hasOfacHitCountIncreased() { var ofacCount = getColumn("OFAC_HIT_COUNT"); var shortName = getColumn("C_FUNC_SHORT_NAME"); var latest = null; var previous = null; for (var i = 0; i < shortName.length; i++) { if (shortName[i] === "IMCC_DEO_DIRPAY") { if (latest === null) { latest = parseInt(ofacCount[i], 10) || 0; } else { previous = parseInt(ofacCount[i], 10) || 0; break; } } } // Less than 2 matching events if (latest === null || previous === null) { return false; } return latest > previous; }
