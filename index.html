function checkMandatoryFields() { var fields = $('[class^="CHAR_M"]'); // Get all elements with class starting with "CHAR_M" var allDetailsFilled = true; // Assume all fields are filled initially fields.each(function() { if ($(this).val().trim() === '') { $(this).focus(); // Focus on the first empty field allDetailsFilled = false; // Mark as not filled return false; // Exit loop early } }); return allDetailsFilled; // Return true if all fields are filled, false otherwise }
