package com.cs.ceweb.servlets; import java.io.File; import java.io.FileOutputStream; import java.io.IOException; import java.io.PrintWriter; import java.sql.Connection; import java.sql.PreparedStatement; import java.sql.ResultSet; import java.util.Base64; import javax.servlet.ServletException; import javax.servlet.http.HttpServlet; import javax.servlet.http.HttpServletRequest; import javax.servlet.http.HttpServletResponse; import org.w3c.dom.Node; import com.cs.base.utility.ASPathConst; import com.cs.base.xml.XMLManager; import com.cs.ce.core.helper.CESecuFileHelper; import com.cs.core.dao.DSManager; import com.cs.core.utility.WSZipUtil; import org.slf4j.Logger; import org.slf4j.LoggerFactory; public class ImageDownloadServlet extends HttpServlet { private static final long serialVersionUID = 1L; private static final Logger LOGGER = LoggerFactory.getLogger(ImageDownloadServlet.class); protected void doPost(HttpServletRequest request, HttpServletResponse response) throws ServletException, IOException { String indexList = request.getParameter("indexList"); String c_main_ref = request.getParameter("cMainRef"); String reqType = request.getParameter("reqType"); String imgIndex = request.getParameter("imgIndex"); if (indexList == null || c_main_ref == null || reqType == null) { LOGGER.error("Invalid request parameters."); response.sendError(HttpServletResponse.SC_BAD_REQUEST, "Missing required parameters."); return; } try (Connection conn = DSManager.getConnection("CET")) { if (reqType.equals("download")) { processDownloadRequest(indexList, c_main_ref, conn, response); } else if (reqType.equals("viewDoc")) { processViewDocRequest(imgIndex, c_main_ref, conn, response); } else { LOGGER.warn("Invalid request type: {}", reqType); response.sendError(HttpServletResponse.SC_BAD_REQUEST, "Invalid request type."); } } catch (Exception e) { LOGGER.error("Error processing request", e); response.sendError(HttpServletResponse.SC_INTERNAL_SERVER_ERROR, "An error occurred while processing the request."); } } private void processDownloadRequest(String indexList, String c_main_ref, Connection conn, HttpServletResponse response) throws Exception { String[] array = indexList.split(";"); for (String value : array) { try (PreparedStatement pstmt = conn.prepareStatement( "SELECT B_IMG_CONTENT, C_IMG_DOC_TYPE, C_IMG_FILE_TYPE FROM TRX_IMAGES_DTAL WHERE C_IMG_INDX = ? AND C_MAIN_REF = ?")) { pstmt.setString(1, value); pstmt.setString(2, c_main_ref); try (ResultSet rs = pstmt.executeQuery()) { if (rs.next()) { handleResultSet(rs, value, response); } else { handleNoDataFound(response); } } } } response.setContentType("application/json"); response.getWriter().write("{\"message\":\"Files saved successfully\"}"); } private void processViewDocRequest(String imgIndex, String c_main_ref, Connection conn, HttpServletResponse response) throws Exception { try (PreparedStatement pstmt = conn.prepareStatement( "SELECT B_IMG_CONTENT, C_IMG_DOC_TYPE, C_IMG_FILE_TYPE FROM TRX_IMAGES_DTAL WHERE C_IMG_INDX = ? AND C_MAIN_REF = ?")) { pstmt.setString(1, imgIndex); pstmt.setString(2, c_main_ref); try (ResultSet rs = pstmt.executeQuery()) { if (rs.next()) { handleResultSet(rs, imgIndex, response); } else { handleNoDataFound(response); } } } } private void handleResultSet(ResultSet rs, String index, HttpServletResponse response) throws Exception { String docType = rs.getString("C_IMG_DOC_TYPE"); String base64Content = rs.getString("B_IMG_CONTENT"); String fileType = rs.getString("C_IMG_FILE_TYPE"); if (base64Content != null && !base64Content.isEmpty() && !"null".equalsIgnoreCase(base64Content)) { if ("image".equalsIgnoreCase(docType)) { LOGGER.info("Processing image with index: {}", index); byte[] decompressedBytes = WSZipUtil.decompressByBase64(base64Content).toString().getBytes("ISO-8859-1"); saveFile(decompressedBytes, index); } else if ("form".equalsIgnoreCase(docType)) { processFileResponse(base64Content, fileType, response); } } else { handleNoDataFound(response); } } private void saveFile(byte[] content, String fileName) throws Exception { String configFilePath = ASPathConst.getUserDirPath() + File.separator + "PARAM" + File.separator + "Default" + File.separator + "SYST" + File.separator + "sysparam.xml"; Node sImagepathNode = XMLManager.findNode(XMLManager.xmlFileToDom(configFilePath).getDocumentElement(), "WEB_IMAGE_ROOT_PATH"); String imagePath = XMLManager.getNodeAttribute(sImagepathNode, "VALUE"); File file = new File(imagePath, fileName + ".pdf"); try (FileOutputStream out = CESecuFileHelper.newFileOutputStream(file.getAbsolutePath())) { out.write(content); LOGGER.info("File saved successfully at {}", file.getAbsolutePath()); } } private void processFileResponse(String base64, String fileType, HttpServletResponse response) throws IOException { try { byte[] content = Base64.getDecoder().decode(base64); String contentType = switch (fileType.toUpperCase()) { case "PDF" -> "application/pdf"; case "TXT" -> "text/plain"; case "XLS", "XLSX" -> "application/vnd.ms-excel"; case "JPG", "JPEG" -> "image/jpeg"; default -> "application/octet-stream"; }; response.setContentType(contentType); response.getOutputStream().write(content); } catch (IllegalArgumentException e) { LOGGER.error("Invalid Base64 content", e); response.sendError(HttpServletResponse.SC_BAD_REQUEST, "Invalid Base64 content."); } } private void handleNoDataFound(HttpServletResponse response) throws IOException { LOGGER.warn("No data found in the database."); response.setContentType("text/html"); try (PrintWriter out = response.getWriter()) { out.println(""); out.println("
Report file not found...
"); out.println(""); } } }
