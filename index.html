function convertToJson(text) { return text .replace(/\r/g, "") .split("===================================") .map(b => b.trim()) .filter(Boolean) .map(parseBlock); } function parseBlock(block) { const obj = { versionLabels: {} }; const lines = block.split("\n").map(l => l.trim()).filter(Boolean); // First line after "Change History" const archiveLine = lines.find(l => l.startsWith("\\\\TCSSBITF")); if (archiveLine) { const match = archiveLine.match(/^(.*?)\-arc\((.*?)\)$/); if (match) { obj.archive = match[1] .replaceAll("\\", "/") .replace("//TCSSBITF/archives", ""); obj.workfile = match[2]; } } const revIndexes = []; for (let i = 0; i < lines.length; i++) { if (/^Rev\s+/.test(lines[i])) { revIndexes.push(i); } } obj.revisionCount = revIndexes.length; if (revIndexes.length) { obj.lastTrunkRevision = lines[revIndexes[0]].replace("Rev", "").trim(); } for (let i = 0; i < revIndexes.length; i++) { const start = revIndexes[i]; const end = (i + 1 < revIndexes.length) ? revIndexes[i + 1] : lines.length; const revision = lines[start].replace("Rev", "").trim(); for (let j = start + 1; j < end; j++) { const irMatch = lines[j].match(/^IR\s+(\d+)/); if (irMatch) { obj.versionLabels[irMatch[1]] = revision; break; // only first IR for this revision } } } return obj; }
