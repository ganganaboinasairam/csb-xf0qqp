function changeFieldClassNew(inputId, param) { var inputField = document.getElementById(inputId); var fieldType = inputField.className.split('_')[0]; if (!inputField) { console.error(`No input field found with ID: ${inputId}`); return; } var parentTd = inputField.parentElement; if (!parentTd) { console.error(`No parent found for input field with ID: ${inputId}`); return; } if(parentTd.className == 'select-wrapper' || parentTd.className == 'date-wrapper' || parentTd.className == 'textarea-wrapper'){ parentTd = inputField.parentElement.parentElement.firstChild; }else{ parentTd = inputField.parentElement.firstChild; } var previousTd = parentTd; if (!previousTd) { console.error(`No previous found for input field with ID: ${inputId}`); return; } if (param === "M") { //inputField.className = (fieldType == 'DATE') ? inputField.className = fieldType + "_M hasDatepicker air-datepicker-initialized" : inputField. if (fieldType == 'DATE'){ inputField.className = fieldType + "_M hasDatepicker air-datepicker-initialized"; addDatePicker(inputField); }else{ inputField.className = fieldType + "_M"; } className = fieldType + "_M"; previousTd.className = "FldLabel star"; inputField.readOnly = false; inputField.removeAttribute('disabled'); } else if (param === "O") { inputField.className = fieldType + "_O"; previousTd.className = "FldLabel"; inputField.readOnly = false; inputField.removeAttribute('disabled'); } else if (param === "P") { inputField.className = fieldType + "_P"; previousTd.className = "FldLabel"; inputField.readOnly = true; inputField.setAttribute('disabled','true'); } else { console.error(`Invalid parameter value: ${param}`); } }
