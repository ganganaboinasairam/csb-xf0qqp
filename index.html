package com.cs.eximap.utility; import java.io.File; import java.sql.Types; import java.util.ArrayList; import java.util.List; import org.apache.xerces.dom.DocumentImpl; import org.w3c.dom.Document; import org.w3c.dom.Element; import org.w3c.dom.Node; import com.cs.base.utility.ASPathConst; import com.cs.base.xml.XMLManager; import com.cs.eximweb.utility.WSTools; import com.google.common.base.Strings; public class DeleteDoRecordInDB { public static DocumentImpl deleteDoRecordInDB(String strMainRef2, String sFunctionId2, DocumentImpl paraDOM) throws Exception { CSSQLStatement deleteEntries = null; Document deleteSQLDom = null; List deleteSqlList = new ArrayList(); try { APLog.reportLog("deleteDoRecordInDB : function ID is "+sFunctionId2 + ",C_MAIN_REF is "+strMainRef2); String functionList = ASPathConst.getUserDirPath() + File.separator + "INT" + File.separator + "DeleteDoRecords.xml"; Document docPara = XMLManager.xmlFileToDom(functionList); Element eRoot = docPara.getDocumentElement(); Node txnNode = XMLManager.findNode(eRoot, sFunctionId2); String table_Name = XMLManager.getChildNodeValue(txnNode, "TABLE", true); String whereClause = XMLManager.getChildNodeValue(txnNode, "CLAUSE", true); String doName = XMLManager.getChildNodeValue(txnNode, "DO_NAME", true); APLog.reportLog("deleteDoRecordInDB : table_Name is "+table_Name + ",whereClause is "+whereClause+",doName is "+doName); Element domDataInfo = (Element)paraDOM.getElementsByTagName("domData").item(0); List columnsList = XMLManager.findChildNodes(domDataInfo, doName); APLog.reportLog("DO Records for "+doName+" are "+columnsList.size()); if(columnsList!=null) { for (int i = 0; i < columnsList.size(); i++) { Node column = columnsList.get(i); String Type = XMLManager.getNodeAttribute(column, "Type"); if(Type.equalsIgnoreCase("D")) { String sb_no = XMLManager.getChildNodeValue(column,"SB_NO",true); if (Strings.isNullOrEmpty(sFunctionId2)) { APLog.reportLog("Delete do record skipped due to FunctionId is is null for the following ref number - " + strMainRef2); }else { deleteEntries = new CSSQLStatement(CSSQLStatement.I_SQL_TYPE_DELETE, table_Name, "EXIMT"); Object[] cValues = { strMainRef2, sb_no}; Integer[] cTypes = { Types.VARCHAR,Types.VARCHAR}; CSSQLClause csClause = new CSSQLClause(); csClause.setClause(whereClause, cValues, cTypes); deleteEntries.setClause(csClause); deleteSqlList.add(deleteEntries); deleteSQLDom = generateSqlDom(deleteSqlList); APLog.reportLog("DO table insert dom SQL Data : " + XMLManager.convertDomToString(deleteSQLDom)); return (DocumentImpl) deleteSQLDom; } } } } } catch (Exception e) { APLog.reportLog(WSTools.getErrorStackTrace(e)); APLog.reportLog("error " + e.getMessage()); } return (DocumentImpl) deleteSQLDom; } private static Document generateSqlDom(List alCSSQL) throws Exception { Document sqlDom = XMLManager.createDocument("root"); Element root = sqlDom.getDocumentElement(); Element eleSQLStatement = sqlDom.createElement("SQLStatement"); root.appendChild(eleSQLStatement); if ((alCSSQL == null) || (alCSSQL.size() == 0)) { return sqlDom; } CSSQLStatement csSQL = (CSSQLStatement) alCSSQL.get(0); String strDS = csSQL.getDS(); Element eleDS = sqlDom.createElement("C_DS_ID"); eleDS.appendChild(sqlDom.createTextNode(strDS)); root.appendChild(eleDS); Element eleSQL = (Element) csSQL.genSqlNode(); eleSQLStatement.appendChild(sqlDom.importNode(eleSQL, true)); for (int i = 1; i < alCSSQL.size(); ++i) { csSQL = (CSSQLStatement) alCSSQL.get(i); eleSQLStatement.appendChild(eleSQLStatement.getOwnerDocument().importNode(csSQL.genSqlNode(), true)); } return sqlDom; } }
