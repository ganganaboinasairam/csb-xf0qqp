SELECT iine.LC_REF_NO AS LC_Reference, SUM(iine.TTL_NEGO_AMT) AS Total_Negotiated_Amount, inad.PRES_LC_BAL AS Presented_LC_Balance, (SUM(iine.TTL_NEGO_AMT) - inad.PRES_LC_BAL) AS Difference FROM INAD_MASTER inad JOIN IINE_MASTER iine ON inad.LC_REF_NO = iine.LC_REF_NO GROUP BY iine.LC_REF_NO, inad.PRES_LC_BAL;
