@Override protected void done() { setBusy(false, null); progressBar.setValue(0); try { CommitPrepResult result = get(); if (result.noChanges) { JOptionPane.showMessageDialog( MainUI.this, "No changes to commit" ); return; } boolean checkFiles = "true".equalsIgnoreCase( Config.get("check.files.before.commit") ); if (checkFiles) { showFileSelection(result); } else { // Direct commit of all changed files SelectedFiles = result.files.size(); showCommitConfirmation( result.defaultMsg, result.files ); } } catch (Exception ex) { log("ERROR: " + ex.getMessage() + "\n"); } }
