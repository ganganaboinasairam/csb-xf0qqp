async function getIRDetails(irNumber) { const url = `https://irtking.stnktimes.in/CGI/Utilities/GetSubjectline.cgi?ir=${irNumber}`; const response = await fetch(url); const html = await response.text(); // IR Number const irMatch = html.match(/IR Number:\s*(\d+)/i); // Status const statusMatch = html.match(/STATUS:\s*([^<]+)/i); // Description const descMatch = html.match(/Desc:([^\/<]+)/i); return { irNumber: irMatch ? irMatch[1].trim() : null, description: descMatch ? descMatch[1].trim() : null, status: statusMatch ? statusMatch[1].trim() : null }; } // Example getIRDetails(25110040).then(console.log);
