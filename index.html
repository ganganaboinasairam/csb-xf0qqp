function EMAIL_ID_check() { try { var email = document.MAINFORM.EMAIL_ADD.value.trim(); // Validate using custom regex var emailfilter = /^[a-zA-Z0-9](?:[a-zA-Z0-9._+=-]*[a-zA-Z0-9])?@(?:[a-zA-Z0-9-]+\.)+(com|net|co\.in|gov\.in)$/; var isValidEmail = emailfilter.test(email); if (isValidEmail) { return true; } else { alert("Please enter a valid email address."); document.MAINFORM.EMAIL_ADD.value = ''; return false; } } catch (e) { DisExcpt("NLRF_EMAIL_ID_Function.js", e); } }
