private static void exportToExcel(JTable table, Component parent) { JFileChooser chooser = new JFileChooser(); chooser.setDialogTitle("Export Tasks to Excel"); chooser.setSelectedFile(new java.io.File("tasks.csv")); if (chooser.showSaveDialog(parent) != JFileChooser.APPROVE_OPTION) { return; } java.io.File file = chooser.getSelectedFile(); try (java.io.PrintWriter pw = new java.io.PrintWriter(file)) { // Header for (int col = 0; col < table.getColumnCount(); col++) { pw.print(table.getColumnName(col)); if (col < table.getColumnCount() - 1) pw.print(","); } pw.println(); // Rows for (int row = 0; row < table.getRowCount(); row++) { for (int col = 0; col < table.getColumnCount(); col++) { Object val = table.getValueAt(row, col); pw.print("\"" + (val == null ? "" : val.toString()) + "\""); if (col < table.getColumnCount() - 1) pw.print(","); } pw.println(); } JOptionPane.showMessageDialog( parent, "Tasks exported successfully.\n\n" + file.getAbsolutePath(), "Export Complete", JOptionPane.INFORMATION_MESSAGE ); } catch (Exception ex) { JOptionPane.showMessageDialog( parent, "Failed to export tasks", "Error", JOptionPane.ERROR_MESSAGE ); ex.printStackTrace(); } }
